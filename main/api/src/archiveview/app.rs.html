16/// How long a middle-click waits for the selection (seconds).
17const SELECTION_WAIT: f64 = 2.0;
18
19/// Text selected in the search box becomes the primary selection once it
20/// has stayed the same this long (seconds): not at every shift+arrow,
21/// which over `ssh -X` costs a round trip each.
22const PRIMARY_DEBOUNCE: f64 = 0.15;
23
24/// How long the "Restored last session" notice shows (seconds).
25const RESTORED_SECS: f64 = 8.0;
26
27/// What the command line asks for.
28#[derive(Clone, Debug, Default)]
29pub struct Options {
30    /// `--archiver`. Without it the archiver is the restored view's, else
31    /// `Settings::archiver_url`, else none (live only): see
32    /// [`startup_archiver`].
33    pub archiver: Option<ArchiverClient>,
34    /// `--no-archiver`: live only, whatever the view or settings say.
35    pub no_archiver: bool,
36    /// PVs given on the command line.
37    pub pvs: Vec<String>,
38    /// `--view NAME`.
39    pub view: Option<String>,
40    /// `--range` (seconds back from now), overriding the view's.
41    pub range: Option<f64>,
42    /// Open `views/default.json` when neither PVs nor a view are given.
43    pub restore_default: bool,
44    /// The theme, overriding the settings (`--theme`; snapshots).
45    pub theme: Option<Theme>,
46    /// Scale the whole interface (`--ui-scale`, `ARCHIVEVIEW_UI_SCALE`).
47    pub ui_scale: Option<f32>,
48}
49
50/// A menu choice, applied once the plot has been drawn.
51enum Action {
52    Save(String),
53    Open(String),
54    Delete(String),
55    Theme(Theme),
56    /// Leave the restored view for an empty plot.
57    StartEmpty,
58}

60/// A message shown at the bottom for a while.
61#[derive(Clone)]
62struct Notice {
63    text: String,
64    error: bool,
65    /// When it goes (egui time): NaN until first shown, then `secs` on.
66    until: f64,
67    secs: f64,
68    /// It offers "Start empty" (the default view was restored).
69    start_empty: bool,
70}
71
72impl Notice {
73    fn new(text: impl Into<String>, error: bool, secs: f64) -> Self {
74        Self {
75            text: text.into(),
76            error,
77            until: f64::NAN,
78            secs,
79            start_empty: false,
80        }
81    }
82}

84/// The archiver to start with, in this order: `--archiver`, none for
85/// `--no-archiver`, the restored view's (`default`, or `--view NAME`),
86/// then `settings.json`'s `archiver_url`.
87pub fn startup_archiver<T>(
88    cli: Option<T>,
89    no_archiver: bool,
90    view: Option<T>,
91    settings: Option<T>,
92) -> Option<T> {
93    match cli {
94        Some(a) => Some(a),
95        None if no_archiver => None,
96        None => view.or(settings),
97    }
98}

144    /// Build the app on `ctx`: fonts, theme, a read-only `PvManager` whose
145    /// notifier repaints `ctx`, and the startup view.
146    pub fn new(ctx: &egui::Context, opts: Options) -> Self {
147        crate::style::install_fonts(ctx);
148        let settings = Settings::load();
149        apply_theme(ctx, opts.theme.unwrap_or(settings.theme));
150        ctx.options_mut(|o| o.zoom_with_keyboard = false);
151        if let Some(f) = opts.ui_scale {
152            ctx.set_zoom_factor(f);
153        }
154
155        let asked = Arc::new(AtomicBool::new(false));
156        let (c, a) = (ctx.clone(), asked.clone());
157        let pvs = PvManager::new();
158        pvs.set_read_only(true);
159        pvs.set_notifier(Arc::new(move || {
160            if !a.swap(true, Ordering::Relaxed) {
161                c.request_repaint_after(Duration::from_millis(100));
162            }
163        }));
164
165        // PVs on the command line give a new view; else `--view`, else the
166        // default view saved last time.
167        let mut notice = None;
168        let mut opened = None;
169        if opts.pvs.is_empty() {
170            let name = opts.view.clone().or_else(|| {
171                (opts.restore_default && config::view_exists(DEFAULT_VIEW))
172                    .then(|| DEFAULT_VIEW.to_string())
173            });
174            if let Some(name) = name {
175                match config::load_view(&name) {
176                    Ok(v) => opened = Some((name, v)),
177                    Err(e) => {
178                        log::warn!("not opening view {name}: {e}");
179                        notice = Some(Notice::new(e, true, f64::INFINITY));
180                    }
181                }
182            }
183        }
184
185        let url = startup_archiver(
186            opts.archiver.as_ref().map(|a| a.base().to_string()),
187            opts.no_archiver,
188            opened.as_ref().and_then(|(_, v)| v.archiver.clone()),
189            settings.archiver_url.clone(),
190        );
191        let archiver = url.and_then(|u| {
192            ArchiverClient::new(&u)
193                .inspect_err(|e| {
194                    log::warn!("{e}");
195                    notice = Some(Notice::new(e.to_string(), true, f64::INFINITY));
196                })
197                .ok()
198        });
199        let mut plot = Plot::new(&pvs, archiver);
200        plot.set_archiver_switching(true);
201        plot.set_recent_archivers(&settings.recent_archivers);
202        plot.set_search_history(config::load_history());
203        let kept_archiver = if opts.no_archiver {
204            match &opened {
205                Some((name, v)) if name == DEFAULT_VIEW => v.archiver.clone(),
206                _ => config::load_view(DEFAULT_VIEW)
207                    .ok()
208                    .and_then(|v| v.archiver),
209            }
210        } else {
211            None
212        };
213
214        plot.add_pvs(&opts.pvs);
215        if let Some((name, mut v)) = opened {
216            // Its archiver is chosen above, in its place among the others.
217            v.archiver = None;
218            plot.set_view(&v);
219            if opts.view.is_none() && !v.pvs.is_empty() && notice.is_none() {
220                notice = Some(Notice {
221                    start_empty: true,
222                    ..Notice::new("Restored last session", false, RESTORED_SECS)
223                });
224            }
225            // The view last opened or saved: the plot keeps it with each
226            // undo step, so undoing an opened view brings back the name.
227            plot.set_view_name(Some(&name));
228        }
229        if let Some(secs) = opts.range {
230            plot.set_range(&Range::Relative { secs });
231        }
232        let keep_default =
233            config::view_exists(DEFAULT_VIEW) && config::load_view(DEFAULT_VIEW).is_err();
234        let known_archiver = plot.archiver().map(|a| a.base().to_string());
235        Self {
236            plot,
237            settings,
238            asked,
239            views: config::list_views(),
240            save_as: String::new(),
241            notice,
242            title: String::new(),
243            forced_theme: opts.theme.is_some(),
244            keep_default,
245            history_unsaved: false,
246            range: opts.range,
247            kept_archiver,
248            known_archiver,
249            selection: None,
250            clicks: 0,
251            reading: None,
252            selected: None,
253        }
254    }
255
256    /// Remember an archiver the user switched to (in the pill, or by
257    /// opening a view that names one) in `settings.json`, as `--archiver`
258    /// is. Switching to none ("No archiver") is for this run only: the
259    /// settings keep the last archiver.
260    fn remember_switch(&mut self) {
261        self.plot.take_archiver_change();
262        let now = self.plot.archiver().map(|a| a.base().to_string());
263        if now == self.known_archiver {
264            return;
265        }
266        self.known_archiver = now.clone();
267        if let Some(url) = now {
268            self.settings.remember_archiver(&url);
269            self.settings.save();
270        }
271    }
272
273    /// Where middle-click reads the primary selection (or the clipboard),
274    /// and the search box's selected text goes: `make` builds it on a
275    /// thread of its own (see [`crate::selection::system`]), which wakes
276    /// `ctx` when a read is answered.
277    pub fn set_selection(&mut self, ctx: &egui::Context, make: Option<MakeSelection>) {
278        self.selection = make.map(|make| {
279            let ctx = ctx.clone();
280            Worker::spawn(make, move || ctx.request_repaint())
281        });
282        self.reading = None;
283    }
284
285    /// Answer the plot's middle-clicks from the selection (read on its
286    /// thread, so a slow owner never stalls the window), make text
287    /// selected in its search box the primary selection once it is steady,
288    /// and say what a paste added.
289    fn paste(&mut self, ctx: &egui::Context, now: f64) {
290        if let Some(worker) = &self.selection {
291            if let Some(click) = self.plot.take_middle_click() {
292                self.clicks += 1;
293                worker.read((self.clicks, click));
294                self.reading = Some((self.clicks, now));
295                ctx.request_repaint_after(Duration::from_secs_f64(SELECTION_WAIT));
296            }
297            while let Some(((n, click), text)) = worker.answer() {
298                // Only the latest click's answer, and only in time.
299                if self.reading.is_some_and(|(r, _)| r == n) {
300                    self.reading = None;
301                    let text = text.unwrap_or_default();
302                    match click {
303                        MiddleClick::Plot => {
304                            self.plot.paste_pvs(&text);
305                        }
306                        MiddleClick::Search => self.plot.paste_into_search(&text),
307                    }
308                }
309            }
310            if let Some((_, since)) = self.reading
311                && now - since >= SELECTION_WAIT
312            {
313                self.reading = None;
314                let mut notice = Notice::new("Could not read the selection", true, 4.0);
315                notice.until = now + notice.secs;
316                self.notice = Some(notice);
317            }
318            // Queued after any read, so a click reads what was selected
319            // before it.
320            if let Some(text) = self.plot.take_search_selection() {
321                self.selected = Some((text, now));
322                ctx.request_repaint_after(Duration::from_secs_f64(PRIMARY_DEBOUNCE));
323            }
324            if self
325                .selected
326                .as_ref()
327                .is_some_and(|(_, since)| now - since >= PRIMARY_DEBOUNCE)
328                && let Some((text, _)) = self.selected.take()
329            {
330                worker.set_primary(text);
331            }
332        }
333        if let Some(pasted) = self.plot.take_pasted() {
334            let error = pasted.added.is_empty() && pasted.present.is_empty();
335            let mut notice = Notice::new(pasted.message(), error, 3.0);
336            notice.until = now + notice.secs;
337            self.notice = Some(notice);
338        }
339    }

346    /// One frame of the whole window.
347    pub fn draw(&mut self, ui: &mut egui::Ui) {
348        self.asked.store(false, Ordering::Relaxed);
349        let ctx = ui.ctx().clone();
350        let now = ui.input(|i| i.time);
351
352        let mut action = None;
353        let frame = egui::Frame::new()
354            .fill(ui.visuals().panel_fill)
355            .inner_margin(egui::Margin::same(10));
356        let Self {
357            plot,
358            settings,
359            views,
360            save_as,
361            forced_theme,
362            ..
363        } = self;
364        let view_name = plot.view_name().map(str::to_string);
365        egui::CentralPanel::default().frame(frame).show(ui, |ui| {
366            let mut menu = |ui: &mut egui::Ui| {
367                action = view_menu(
368                    ui,
369                    settings,
370                    view_name.as_deref(),
371                    views,
372                    save_as,
373                    *forced_theme,
374                );
375            };
376            plot.ui_with_menu(ui, &mut menu);
377        });
378        if let Some(action) = action {
379            self.apply(&ctx, action, now);
380        }
381        self.remember_switch();
382        // Saved once per change; a failed save is tried again on exit,
383        // not every frame.
384        if self.plot.search_history_changed() {
385            self.history_unsaved = !config::save_history(self.plot.search_history());
386        }
387        self.paste(&ctx, now);
388        self.toast(&ctx, now);
389
390        let title = match self.plot.title() {
391            t if t.is_empty() => "archiveview".to_string(),
392            t => format!("{t} — archiveview"),
393        };
394        if title != self.title {
395            ctx.send_viewport_cmd(egui::ViewportCommand::Title(title.clone()));
396            self.title = title;
397        }
398    }

405    /// Save the default view when the settings say to, unless the one
406    /// there could not be read; and the recent searches, if saving them
407    /// last failed.
408    pub fn exit(&mut self) {
409        if self.history_unsaved {
410            self.history_unsaved = !config::save_history(self.plot.search_history());
411        }
412        if !self.settings.save_default_on_exit {
413            return;
414        }
415        if self.keep_default {
416            log::warn!("not saving the default view over one that could not be read");
417            return;
418        }
419        let mut view = self.plot.view();
420        if view.archiver.is_none() {
421            view.archiver = self.kept_archiver.clone();
422        }
423        if let Err(e) = config::save_view(DEFAULT_VIEW, &view) {
424            log::error!("{e}");
425        }
426    }

428    fn apply(&mut self, ctx: &egui::Context, action: Action, now: f64) {
429        let result = match action {
430            Action::Save(name) => {
431                let name = name.trim().to_string();
432                config::save_view(&name, &self.plot.view()).map(|()| {
433                    if name == DEFAULT_VIEW {
434                        self.keep_default = false;
435                    }
436                    self.plot.set_view_name(Some(&name));
437                    self.save_as.clear();
438                    format!("Saved view “{name}”")
439                })
440            }
441            Action::Open(name) => config::load_view(&name).map(|v| {
442                let before = self.plot.archiver().cloned();
443                self.plot.set_view(&v);
444                self.plot.set_view_name(Some(&name));
445                match self.plot.archiver() {
446                    Some(a) if before.as_ref() != Some(a) => {
447                        format!("Opened view “{name}” from {}", a.host())
448                    }
449                    _ => format!("Opened view “{name}”"),
450                }
451            }),
452            Action::Delete(name) => config::delete_view(&name).map(|()| {
453                if name == DEFAULT_VIEW {
454                    self.keep_default = false;
455                }
456                if self.plot.view_name() == Some(name.as_str()) {
457                    self.plot.set_view_name(None);
458                }
459                format!("Deleted view “{name}”")
460            }),
461            Action::Theme(theme) => {
462                self.settings.theme = theme;
463                self.settings.save();
464                apply_theme(ctx, theme);
465                return;
466            }
467            Action::StartEmpty => {
468                // An empty view keeps the archiver, and `--range`.
469                let range = self.range.map(|secs| Range::Relative { secs });
470                self.plot.set_view(&View {
471                    range: range.unwrap_or_default(),
472                    ..View::default()
473                });
474                self.plot.set_view_name(None);
475                self.notice = None;
476                return;
477            }
478        };
479        self.views = config::list_views();
480        let mut notice = match result {
481            Ok(msg) => Notice::new(msg, false, 3.0),
482            Err(e) => Notice::new(e, true, 8.0),
483        };
484        notice.until = now + notice.secs;
485        self.notice = Some(notice);
486    }

488    /// The notice, floating over the bottom of the window until it expires
489    /// (or is clicked).
490    fn toast(&mut self, ctx: &egui::Context, now: f64) {
491        let Some(notice) = &mut self.notice else {
492            return;
493        };
494        if notice.until.is_nan() {
495            notice.until = now + notice.secs;
496        }
497        let Notice {
498            text,
499            error,
500            until,
501            start_empty,
502            ..
503        } = notice.clone();
504        if now > until {
505            self.notice = None;
506            return;
507        }
508        let mut empty = false;
509        let shown = egui::Area::new(egui::Id::new("archiveview-notice"))
510            .order(egui::Order::Foreground)
511            .anchor(egui::Align2::CENTER_BOTTOM, egui::vec2(0.0, -56.0))
512            .show(ctx, |ui| {
513                egui::Frame::popup(ui.style())
514                    .corner_radius(16.0)
515                    .inner_margin(egui::Margin::symmetric(14, 8))
516                    .show(ui, |ui| {
517                        let color = if error {
518                            ui.visuals().error_fg_color
519                        } else {
520                            ui.visuals().text_color()
521                        };
522                        ui.horizontal(|ui| {
523                            let r = ui.add(
524                                egui::Label::new(egui::RichText::new(text).color(color))
525                                    .sense(egui::Sense::click()),
526                            );
527                            if start_empty
528                                && ui
529                                    .button("Start empty")
530                                    .on_hover_text("Clear the plot (the archiver stays)")
531                                    .clicked()
532                            {
533                                empty = true;
534                            }
535                            r
536                        })
537                        .inner
538                    })
539                    .inner
540            });
541        if empty {
542            self.apply(ctx, Action::StartEmpty, now);
543        } else if shown.inner.clicked() {
544            self.notice = None;
545        } else if until.is_finite() {
546            ctx.request_repaint_after(Duration::from_secs_f64((until - now).max(0.05)));
547        }
548    }
549}

571/// The app's part of the `≡` menu: views and the theme.
572fn view_menu(
573    ui: &mut egui::Ui,
574    settings: &mut Settings,
575    current: Option<&str>,
576    views: &[String],
577    save_as: &mut String,
578    forced_theme: bool,
579) -> Option<Action> {
580    let mut action = None;
581    ui.label(egui::RichText::new("Views").small().weak());
582    if let Some(name) = current
583        && ui.button(format!("Save “{name}”")).clicked()
584    {
585        action = Some(Action::Save(name.to_string()));
586    }
587    ui.horizontal(|ui| {
588        let r = ui.add(
589            egui::TextEdit::singleline(save_as)
590                .hint_text("Save as…")
591                .desired_width(140.0),
592        );
593        let ok = config::valid_name(save_as);
594        let enter = r.lost_focus() && ui.input(|i| i.key_pressed(egui::Key::Enter));
595        if (ui.add_enabled(ok, egui::Button::new("Save")).clicked() || enter) && ok {
596            action = Some(Action::Save(save_as.clone()));
597        }
598    });
599    if ui.button("Save as default").clicked() {
600        action = Some(Action::Save(DEFAULT_VIEW.to_string()));
601    }
602    ui.add_enabled_ui(!views.is_empty(), |ui| {
603        ui.menu_button("Open", |ui| {
604            for name in views.iter() {
605                if ui
606                    .selectable_label(Some(name.as_str()) == current, name)
607                    .clicked()
608                {
609                    action = Some(Action::Open(name.clone()));
610                }
611            }
612        });
613        ui.menu_button("Delete", |ui| {
614            for name in views.iter() {
615                if ui.button(name).clicked() {
616                    action = Some(Action::Delete(name.clone()));
617                }
618            }
619        });
620    });
621    if ui
622        .checkbox(&mut settings.save_default_on_exit, "Save default on exit")
623        .changed()
624    {
625        settings.save();
626    }
627    if !forced_theme {
628        ui.separator();
629        ui.label(egui::RichText::new("Theme").small().weak());
630        ui.horizontal(|ui| {
631            for (theme, name) in [
632                (Theme::System, "System"),
633                (Theme::Light, "Light"),
634                (Theme::Dark, "Dark"),
635            ] {
636                if ui.selectable_label(settings.theme == theme, name).clicked() {
637                    action = Some(Action::Theme(theme));
638                }
639            }
640        });
641    }
642    action
643}

677    /// A selection the test fills, sharing what was set with the test.
678    struct FakeSelection {
679        /// There is a primary selection (`primary` may still be `None`).
680        has: bool,
681        primary: Option<String>,
682        clipboard: Option<String>,
683        set: Arc<parking_lot::Mutex<Vec<String>>>,
684        /// Take this long to answer, as a hung selection owner does.
685        slow: Duration,
686    }
687
688    impl crate::selection::Selection for FakeSelection {
689        fn has_primary(&mut self) -> bool {
690            self.has
691        }
692        fn primary(&mut self) -> Option<String> {
693            std::thread::sleep(self.slow);
694            self.primary.clone()
695        }
696        fn clipboard(&mut self) -> Option<String> {
697            self.clipboard.clone()
698        }
699        fn set_primary(&mut self, text: &str) {
700            self.set.lock().push(text.to_string());
701        }
702    }
703
704    impl Drop for FakeSelection {
705        /// A slow one says when it is dropped (it should be leaked).
706        fn drop(&mut self) {
707            if !self.slow.is_zero() {
708                self.set.lock().push("dropped".into());
709            }
710        }
711    }

713    #[test]
714    fn middle_click_adds_the_selected_pvs_and_says_so() {
715        let ctx = egui::Context::default();
716        let mut app = ArchiveApp::new(&ctx, Options::default());
717        app.plot.set_archiver(None);
718        let t = std::cell::Cell::new(0.0);
719        let frame = |app: &mut ArchiveApp, events: Vec<egui::Event>| {
720            let input = egui::RawInput {
721                screen_rect: Some(egui::Rect::from_min_size(
722                    egui::Pos2::ZERO,
723                    egui::vec2(1000.0, 600.0),
724                )),
725                time: Some(t.get()),
726                events,
727                ..Default::default()
728            };
729            t.set(t.get() + 0.05);
730            let mut out = ctx.run_ui(input, |ui| app.draw(ui));
731            out.textures_delta.clear();
732        };
733        // A middle-click, then frames until the selection's answer is in.
734        let click = |app: &mut ArchiveApp, pos| {
735            let ev = |pressed| egui::Event::PointerButton {
736                pos,
737                button: egui::PointerButton::Middle,
738                pressed,
739                modifiers: egui::Modifiers::NONE,
740            };
741            frame(app, vec![egui::Event::PointerMoved(pos), ev(true)]);
742            frame(app, vec![ev(false)]);
743            assert!(app.reading.is_some(), "asked for the selection");
744            let start = std::time::Instant::now();
745            while app.reading.is_some() {
746                assert!(start.elapsed().as_secs() < 5, "no answer");
747                std::thread::sleep(Duration::from_millis(10));
748                frame(app, vec![]);
749            }
750            frame(app, vec![]);
751        };
752        let set = Arc::new(parking_lot::Mutex::new(Vec::new()));
753        let select =
754            |app: &mut ArchiveApp, has: bool, primary: Option<&str>, clip: Option<&str>| {
755                let fake = FakeSelection {
756                    has,
757                    primary: primary.map(Into::into),
758                    clipboard: clip.map(Into::into),
759                    set: set.clone(),
760                    slow: Duration::ZERO,
761                };
762                app.set_selection(&ctx, Some(Box::new(move || Box::new(fake))));
763            };
764        let on_plot = egui::pos2(500.0, 300.0);
765        frame(&mut app, vec![]);
766
767        // The primary selection's PVs, with a toast.
768        let two = Some("BL16I-EA-DET-01:COUNT\nSR:X");
769        select(&mut app, true, two, Some("CLIP:BOARD"));
770        click(&mut app, on_plot);
771        assert_eq!(app.plot.names(), ["BL16I-EA-DET-01:COUNT", "SR:X"]);
772        assert_eq!(app.notice.as_ref().unwrap().text, "Added 2 PVs");
773        // An empty primary selection is not the clipboard's cue.
774        select(&mut app, true, None, Some("CLIP:BOARD"));
775        click(&mut app, on_plot);
776        assert_eq!(app.plot.names().len(), 2);
777        assert!(app.notice.as_ref().unwrap().text.contains("empty"));
778        // No primary selection at all: the clipboard.
779        select(&mut app, false, None, Some("CLIP:BOARD"));
780        click(&mut app, on_plot);
781        assert_eq!(app.plot.names().last().unwrap(), "CLIP:BOARD");
782        assert_eq!(app.notice.as_ref().unwrap().text, "Added CLIP:BOARD");
783        // Nothing usable: the toast says so.
784        let junk = Some("42 12:30:00 https://example.com/x");
785        select(&mut app, true, junk, None);
786        click(&mut app, on_plot);
787        assert_eq!(app.plot.names().len(), 3);
788        let notice = app.notice.as_ref().unwrap();
789        assert!(notice.error && notice.text.contains("No PV names"));
790
791        // In the search box: pasted there, not added.
792        select(&mut app, true, Some("SR-DI-DCCT-01:SIGNAL"), None);
793        click(&mut app, egui::pos2(80.0, 22.0));
794        assert_eq!(app.plot.names().len(), 3);
795        // Selecting it there makes it the primary selection, once steady.
796        let select_all = egui::Event::Key {
797            key: egui::Key::A,
798            physical_key: None,
799            pressed: true,
800            repeat: false,
801            modifiers: egui::Modifiers::COMMAND,
802        };
803        frame(&mut app, vec![select_all]);
804        frame(&mut app, vec![]);
805        assert!(set.lock().is_empty(), "not yet");
806        for _ in 0..6 {
807            frame(&mut app, vec![]);
808        }
809        let start = std::time::Instant::now();
810        while set.lock().is_empty() && start.elapsed().as_secs() < 5 {
811            std::thread::sleep(Duration::from_millis(10));
812        }
813        assert_eq!(*set.lock(), ["SR-DI-DCCT-01:SIGNAL"]);
814
815        // A selection owner that never answers: the window goes on, and
816        // after a while says so.
817        let late = Arc::new(parking_lot::Mutex::new(Vec::new()));
818        let fake = FakeSelection {
819            has: true,
820            primary: Some("LATE:PV".into()),
821            clipboard: None,
822            set: late.clone(),
823            slow: Duration::from_millis(1500),
824        };
825        app.set_selection(&ctx, Some(Box::new(move || Box::new(fake))));
826        let ev = |pressed| egui::Event::PointerButton {
827            pos: on_plot,
828            button: egui::PointerButton::Middle,
829            pressed,
830            modifiers: egui::Modifiers::NONE,
831        };
832        let before = std::time::Instant::now();
833        frame(&mut app, vec![egui::Event::PointerMoved(on_plot), ev(true)]);
834        frame(&mut app, vec![ev(false)]);
835        // 2.5 s of egui time pass in frames, quickly.
836        for _ in 0..50 {
837            frame(&mut app, vec![]);
838        }
839        assert!(before.elapsed().as_secs_f64() < 1.0, "never blocked");
840        assert!(app.reading.is_none());
841        let notice = app.notice.as_ref().unwrap();
842        assert!(notice.error && notice.text.contains("Could not read"));
843        assert_eq!(app.plot.names().len(), 3, "the late answer is dropped");
844        // Still stuck when dropped (as at exit): given up on, and when it
845        // wakes it leaks the selection rather than dropping it.
846        app.set_selection(&ctx, None);
847        std::thread::sleep(Duration::from_millis(1500));
848        assert!(late.lock().is_empty(), "{:?}", late.lock());
849    }

851    #[test]
852    fn the_default_view_comes_back_with_its_archiver() {
853        let dir = std::env::temp_dir().join(format!("archiveview-app-{}", std::process::id()));
854        std::fs::remove_dir_all(&dir).ok();
855        // SAFETY: the only test that sets the environment; the others
856        // read their config from directories they are given.
857        unsafe { std::env::set_var("XDG_CONFIG_HOME", &dir) };
858        let base = |o: &ArchiveApp| o.plot.archiver().map(|a| a.base().to_string());
859        let view_url = "http://view:17668/retrieval";
860        let settings_url = "http://settings:17668/retrieval";
861        let mut settings = Settings::default();
862        settings.remember_archiver(settings_url);
863        settings.save();
864        let view = View {
865            pvs: vec![crate::view::TraceView {
866                name: "sim://sine".into(),
867                color: None,
868                visible: true,
869                axis: None,
870            }],
871            archiver: Some(view_url.into()),
872            ..View::default()
873        };
874        config::save_view(DEFAULT_VIEW, &view).unwrap();
875        let ctx = egui::Context::default();
876        let restore = Options {
877            restore_default: true,
878            ..Options::default()
879        };
880
881        // Restored, with the view's archiver, and said so.
882        let app = ArchiveApp::new(&ctx, restore.clone());
883        assert_eq!(base(&app), Some(view_url.into()));
884        assert_eq!(app.plot.names(), ["sim://sine"]);
885        let notice = app.notice.as_ref().expect("a notice");
886        assert!(notice.start_empty && notice.text.contains("Restored"));
887        assert_eq!(
888            app.plot.recent_archivers(),
889            [view_url, settings_url],
890            "the pill offers both"
891        );
892        // A run with --no-archiver doesn't forget the default's archiver.
893        let mut app = ArchiveApp::new(
894            &ctx,
895            Options {
896                no_archiver: true,
897                range: Some(600.0),
898                ..restore.clone()
899            },
900        );
901        assert_eq!(base(&app), None);
902        app.exit();
903        let saved = config::load_view(DEFAULT_VIEW).unwrap();
904        assert_eq!(saved.archiver.as_deref(), Some(view_url));
905        // "Start empty" keeps --range.
906        app.apply(&ctx, Action::StartEmpty, 0.0);
907        assert_eq!(app.plot.view().range, Range::Relative { secs: 600.0 });
908        // Opening a view with another archiver is remembered, as a
909        // switch in the pill is; switching to none is not.
910        let other = View {
911            archiver: Some("http://other:17668/retrieval".into()),
912            ..View::default()
913        };
914        config::save_view("other", &other).unwrap();
915        app.apply(&ctx, Action::Open("other".into()), 0.0);
916        app.remember_switch();
917        assert_eq!(
918            Settings::load().archiver_url.as_deref(),
919            Some("http://other:17668/retrieval")
920        );
921        app.plot.set_archiver(None);
922        app.remember_switch();
923        assert_eq!(
924            Settings::load().archiver_url.as_deref(),
925            Some("http://other:17668/retrieval")
926        );
927        config::delete_view("other").unwrap();
928        settings.save();
929
930        // --archiver wins, and --no-archiver.
931        let cli = ArchiverClient::new("http://cli:17668").unwrap();
932        let app = ArchiveApp::new(
933            &ctx,
934            Options {
935                archiver: Some(cli.clone()),
936                ..restore.clone()
937            },
938        );
939        assert_eq!(base(&app), Some(cli.base().into()));
940        let app = ArchiveApp::new(
941            &ctx,
942            Options {
943                no_archiver: true,
944                ..restore.clone()
945            },
946        );
947        assert_eq!(base(&app), None);
948        // PVs on the command line replace the view: the settings' archiver.
949        let mut app = ArchiveApp::new(
950            &ctx,
951            Options {
952                pvs: vec!["sim://ramp".into()],
953                ..restore.clone()
954            },
955        );
956        assert_eq!(base(&app), Some(settings_url.into()));
957        assert_eq!(app.plot.names(), ["sim://ramp"]);
958        assert!(app.notice.is_none());
959        // Exit saves the default with the archiver in use.
960        app.exit();
961        let saved = config::load_view(DEFAULT_VIEW).unwrap();
962        assert_eq!(saved.archiver.as_deref(), Some(settings_url));
963        assert_eq!(saved.pvs[0].name, "sim://ramp");
964        // "Start empty" keeps the archiver.
965        app.apply(&ctx, Action::StartEmpty, 0.0);
966        assert!(app.plot.names().is_empty());
967        assert_eq!(base(&app), Some(settings_url.into()));
968        std::fs::remove_dir_all(&dir).ok();
969    }
