4/// Tick values for a linear scale over `lo..=hi`, at a "nice" step
5/// (1, 2 or 5 times a power of ten) giving at most `max_ticks` ticks.
6/// Returns the ticks and the step.
7pub fn ticks(lo: f64, hi: f64, max_ticks: usize) -> (Vec<f64>, f64) {
8    let (lo, hi) = if lo <= hi { (lo, hi) } else { (hi, lo) };
9    let span = hi - lo;
10    if !lo.is_finite() || !hi.is_finite() {
11        return (Vec::new(), 0.0);
12    }
13    if span <= 0.0 {
14        return (vec![lo], 0.0);
15    }
16    let step = nice_step(span, max_ticks);
17    let first = (lo / step - 1e-9).ceil();
18    let values = (0..)
19        .map(|i| (first + i as f64) * step)
20        .take_while(|v| *v <= hi + step * 1e-9)
21        .take(1000)
22        .map(|v| if v.abs() < step * 1e-9 { 0.0 } else { v })
23        .collect();
24    (values, step)
25}

27/// As [`ticks`], for whole-number values (enum states, counts): the step
28/// is never under 1, so no tick falls between two values.
29pub fn whole_ticks(lo: f64, hi: f64, max_ticks: usize) -> (Vec<f64>, f64) {
30    let (t, step) = ticks(lo, hi, max_ticks);
31    if step >= 1.0 || step == 0.0 {
32        return (t, step);
33    }
34    let (lo, hi) = if lo <= hi { (lo, hi) } else { (hi, lo) };
35    let values: Vec<f64> = (0..1000)
36        .map(|i| lo.ceil() + i as f64)
37        .take_while(|v| *v <= hi)
38        .collect();
39    (values, 1.0)
40}

42/// An autoscaled range for whole-number values: at least two wide, so a
43/// value that holds steady sits between whole ticks (0..2 for a constant
44/// 1), not on a fractional axis from 0.9 to 1.1.
45pub fn whole_range(lo: f64, hi: f64) -> (f64, f64) {
46    if hi - lo < 2.0 && lo.is_finite() && hi.is_finite() {
47        let mid = ((lo + hi) / 2.0).round();
48        (mid - 1.0, mid + 1.0)
49    } else {
50        (lo, hi)
51    }
52}

54/// The smallest 1/2/5 step that splits `span` into at most `max_ticks - 1` gaps.
55pub fn nice_step(span: f64, max_ticks: usize) -> f64 {
56    let gaps = (max_ticks.max(2) - 1) as f64;
57    let mag = 10f64.powf((span / gaps).log10().floor());
58    [1.0, 2.0, 5.0, 10.0]
59        .iter()
60        .map(|m| m * mag)
61        .find(|s| span / s <= gaps + 1e-9)
62        .unwrap_or(10.0 * mag)
63}

65/// Decade ticks for a log scale over `lo..=hi` (both must be positive).
66pub fn log_ticks(lo: f64, hi: f64) -> Vec<f64> {
67    if lo <= 0.0 || hi <= lo || !hi.is_finite() {
68        return Vec::new();
69    }
70    let (a, b) = (lo.log10().ceil() as i32, hi.log10().floor() as i32);
71    (a..=b).take(100).map(|e| 10f64.powi(e)).collect()
72}

74/// Widen `lo..hi` outwards to whole ticks, as Phoebus does when autoscaling.
75/// An empty range is opened up around its value.
76pub fn nice_range(lo: f64, hi: f64, max_ticks: usize) -> (f64, f64) {
77    if !lo.is_finite() || !hi.is_finite() {
78        return (0.0, 100.0);
79    }
80    if hi <= lo {
81        let d = if lo == 0.0 { 1.0 } else { lo.abs() * 0.1 };
82        return nice_range(lo - d, lo + d, max_ticks);
83    }
84    let step = nice_step(hi - lo, max_ticks);
85    ((lo / step).floor() * step, (hi / step).ceil() * step)
86}

88/// A tick label, with as many decimals as the step needs.
89pub fn label(v: f64, step: f64) -> String {
90    let big = v.abs().max(step.abs());
91    if big >= 1e6 || (step > 0.0 && step < 1e-4) {
92        if v == 0.0 {
93            // Not "0e0" among 2e9, 4e9.
94            return "0".into();
95        }
96        // Enough mantissa decimals to reach the step's digit, so 1.30e8 and
97        // 1.32e8 don't both read 1.3e8.
98        let decimals = if step > 0.0 && v != 0.0 {
99            let e = (v.abs().log10() + 1e-9).floor();
100            (e - (step.log10() + 1e-9).floor()).clamp(0.0, 15.0) as usize
101        } else {
102            1
103        };
104        let s = format!("{v:.decimals$e}");
105        return s.replace(".0e", "e");
106    }
107    let decimals = if step > 0.0 {
108        (-step.log10().floor()).max(0.0) as usize
109    } else {
110        // A lone tick or a log decade: just enough to show the value, small
111        // ones (pressures in mbar) in exponent form.
112        let s = format!("{v}");
113        return match s.len() {
114            0..=8 => s,
115            _ if v != 0.0 && v.abs() < 1e-3 => exponent(v),
116            _ => format!("{v:.3}"),
117        };
118    };
119    format!("{v:.decimals$}")
120}
121
122/// `v` as `1.5e-9`, with no trailing zeros.
123fn exponent(v: f64) -> String {
124    let s = format!("{v:.2e}");
125    match s.split_once('e') {
126        Some((m, e)) if m.contains('.') => {
127            format!("{}e{e}", m.trim_end_matches('0').trim_end_matches('.'))
128        }
129        _ => s,
130    }
131}

133/// Maps values to positions along a scale.
134#[derive(Clone, Copy, Debug)]
135pub struct Scale {
136    pub lo: f64,
137    pub hi: f64,
138    pub log: bool,
139    /// Screen positions of `lo` and `hi`.
140    pub from: f32,
141    pub to: f32,
142}
143
144impl Scale {
145    fn t(&self, v: f64) -> f64 {
146        if self.log && self.lo > 0.0 {
147            let l = |x: f64| x.max(f64::MIN_POSITIVE).log10();
148            (l(v) - l(self.lo)) / (l(self.hi) - l(self.lo))
149        } else {
150            (v - self.lo) / (self.hi - self.lo)
151        }
152    }
153
154    /// Screen position of `v`; values outside the range land outside `from..to`.
155    pub fn pos(&self, v: f64) -> f32 {
156        let t = self.t(v);
157        let t = if t.is_finite() { t } else { 0.0 };
158        self.from + (self.to - self.from) * t as f32
159    }
160
161    /// As [`Scale::ticks`], at whole numbers only on a linear scale.
162    pub fn whole_ticks(&self, label_len: f32) -> (Vec<f64>, f64) {
163        if self.log {
164            return self.ticks(label_len);
165        }
166        let length = (self.to - self.from).abs();
167        let n = (length / label_len.max(1.0)).floor().max(2.0) as usize;
168        whole_ticks(self.lo, self.hi, n)
169    }
170
171    /// Ticks for a scale `length` pixels long with labels about `label_len` apart.
172    pub fn ticks(&self, label_len: f32) -> (Vec<f64>, f64) {
173        if self.log && self.lo > 0.0 {
174            let t = log_ticks(self.lo, self.hi);
175            if t.len() >= 2 {
176                return (t, 0.0);
177            }
178        }
179        let length = (self.to - self.from).abs();
180        let n = (length / label_len.max(1.0)).floor().max(2.0) as usize;
181        ticks(self.lo, self.hi, n)
182    }
183}

185#[cfg(test)]

189    #[test]
190    fn ticks_use_nice_steps() {
191        let (t, step) = ticks(-25.0, 25.0, 6);
192        assert_eq!(step, 10.0);
193        assert_eq!(t, vec![-20.0, -10.0, 0.0, 10.0, 20.0]);
194        let (t, step) = ticks(-360.0, 360.0, 10);
195        assert_eq!(step, 100.0);
196        assert_eq!(t.first(), Some(&-300.0));
197        assert_eq!(t.last(), Some(&300.0));
198        let (t, step) = ticks(0.0, 1.0, 6);
199        assert_eq!(step, 0.2);
200        assert_eq!(t.len(), 6);
201        assert_eq!(ticks(0.0, 100.0, 11).0.len(), 11);
202    }
203
204    #[test]
205    fn ticks_handle_degenerate_ranges() {
206        assert_eq!(ticks(5.0, 5.0, 5), (vec![5.0], 0.0));
207        assert!(ticks(f64::NAN, 1.0, 5).0.is_empty());
208        // Reversed ranges are fine.
209        assert_eq!(ticks(10.0, 0.0, 3).0, vec![0.0, 5.0, 10.0]);
210    }
211
212    #[test]
213    fn labels_match_the_step() {
214        assert_eq!(label(20.0, 10.0), "20");
215        assert_eq!(label(-0.4, 0.2), "-0.4");
216        assert_eq!(label(0.25, 0.05), "0.25");
217        assert_eq!(label(2e6, 1e6), "2e6");
218        assert_eq!(label(1.5e6, 5e5), "1.5e6");
219        assert_eq!(label(0.0, 1e9), "0");
220        assert_eq!(label(0.0, 1e-5), "0");
221        // A narrow range of big values keeps adjacent ticks apart.
222        assert_eq!(label(1.3e8, 2e6), "1.30e8");
223        assert_eq!(label(1.32e8, 2e6), "1.32e8");
224        assert_eq!(label(1.5e-5, 5e-6), "1.5e-5");
225        // Log decades have no step.
226        assert_eq!(label(1e-9, 0.0), "1e-9");
227        assert_eq!(label(2.5e-7, 0.0), "2.5e-7");
228        assert_eq!(label(100.0, 0.0), "100");
229    }

231    #[test]
232    fn whole_values_get_whole_ticks() {
233        // A steady enum: 0..2 with ticks on the states, not 0.92, 0.94, ...
234        assert_eq!(whole_range(0.9, 1.1), (0.0, 2.0));
235        assert_eq!(whole_range(0.0, 3.0), (0.0, 3.0));
236        assert_eq!(whole_ticks(0.0, 2.0, 10), (vec![0.0, 1.0, 2.0], 1.0));
237        assert_eq!(whole_ticks(-0.5, 3.5, 20).0, vec![0.0, 1.0, 2.0, 3.0]);
238        // Wide ranges keep their nice steps.
239        assert_eq!(whole_ticks(0.0, 100.0, 6).1, 20.0);
240    }

242    #[test]
243    fn nice_range_rounds_outwards() {
244        assert_eq!(nice_range(0.0, 1023.0, 8), (0.0, 1200.0));
245        assert_eq!(nice_range(3.2, 9.7, 8), (3.0, 10.0));
