1//! Background archive requests. Each runs as a blocking task on the pv
2//! crate's tokio runtime (the caller's `PvManager`), so a plot starts no
3//! threads of its own; results come back through a shared inbox, and the
4//! plot's context is asked to repaint.
5
6use std::sync::Arc;
7use std::sync::atomic::{AtomicUsize, Ordering};

9use archiver::{ArchiverClient, Operator, PvInfo, Query, Search, TimeSeries};
10use parking_lot::Mutex;
11use pv::PvManager;

13/// The padding fetched either side of the visible window, as a fraction of
14/// its span, so a small pan or zoom has data at once.
15pub const PAD: f64 = 0.25;
16
17/// A data request for one trace.
18pub struct Job {
19    pub trace: u64,
20    pub generation: u64,
21    pub pv: String,
22    pub from: f64,
23    pub to: f64,
24    pub op: Operator,
25    /// Where to start instead if the metadata says to fetch raw: no more
26    /// than [`RAW_MAX_SPAN`] (padded) of a wide window.
27    pub raw_from: f64,
28    /// Ask `getMetadata` first, to learn whether to fetch raw.
29    pub probe: bool,
30    /// Only the tail of the history (a live refresh): its samples are
31    /// added to the history, not put in its place.
32    pub append: bool,
33}

35/// A data job as sent: (trace, from, to, append).
36#[cfg(test)]
37pub type Sent = (u64, f64, f64, bool);

39/// A finished data request.
40pub struct Done {
41    pub trace: u64,
42    pub generation: u64,
43    /// The window fetched (or asked for, on an error).
44    pub from: f64,
45    pub to: f64,
46    pub op: Operator,
47    /// As [`Job::append`].
48    pub append: bool,
49    pub info: Option<PvInfo>,
50    pub result: Result<Option<TimeSeries>, archiver::Error>,
51}

53/// A finished search: its number and the names (or an error).
54pub type Found = (u64, Result<Vec<String>, String>);

56/// Where to ask for a repaint when a request finishes: the context and the
57/// viewport the plot was last drawn in. A host such as bobview may show
58/// the plot in its own viewport, and a plain `request_repaint` from a
59/// worker thread repaints only the root one.
60#[derive(Clone)]
61pub struct Repaint {
62    pub ctx: egui::Context,
63    pub viewport: egui::ViewportId,
64}
65
66impl Repaint {
67    fn now(&self) {
68        self.ctx.request_repaint_of(self.viewport);
69    }
70}

72/// The plot's requests in flight and the results not yet taken in.
73#[derive(Clone, Default)]
74pub struct Fetcher {
75    inbox: Arc<Mutex<Vec<Done>>>,
76    found: Arc<Mutex<Option<Found>>>,
77    busy: Arc<AtomicUsize>,
78    /// Every data job sent, for tests.
79    #[cfg(test)]
80    pub sent: Arc<Mutex<Vec<Sent>>>,
81}
82
83impl Fetcher {
84    /// Requests in flight (data and search).
85    pub fn busy(&self) -> bool {
86        self.busy.load(Ordering::Relaxed) > 0
87    }
88
89    /// How many requests are in flight.
90    #[cfg(test)]
91    pub fn in_flight(&self) -> usize {
92        self.busy.load(Ordering::Relaxed)
93    }
94
95    /// Requests in flight, or results not yet taken in.
96    pub fn pending(&self) -> bool {
97        self.busy() || !self.inbox.lock().is_empty() || self.found.lock().is_some()
98    }
99
100    pub fn take(&self) -> Vec<Done> {
101        std::mem::take(&mut *self.inbox.lock())
102    }
103
104    pub fn take_found(&self) -> Option<Found> {
105        self.found.lock().take()
106    }

108    /// Run `job` against `client` in the background.
109    pub fn fetch(
110        &self,
111        pvs: &PvManager,
112        client: &ArchiverClient,
113        job: Job,
114        repaint: Option<Repaint>,
115    ) {
116        #[cfg(test)]
117        self.sent
118            .lock()
119            .push((job.trace, job.from, job.to, job.append));
120        let client = client.clone();
121        let inbox = self.inbox.clone();
122        let (trace, generation, from, to, op, append) = (
123            job.trace,
124            job.generation,
125            job.from,
126            job.to,
127            job.op,
128            job.append,
129        );
130        let panicked = {
131            let inbox = inbox.clone();
132            move || {
133                inbox.lock().push(Done {
134                    trace,
135                    generation,
136                    from,
137                    to,
138                    op,
139                    append,
140                    info: None,
141                    result: Err(archiver::Error::Transport(
142                        "the request failed unexpectedly".into(),
143                    )),
144                });
145            }
146        };
147        self.run(pvs, repaint, panicked, move || {
148            let info = if job.probe {
149                match client.metadata(&job.pv) {
150                    Ok(info) => Some(info),
151                    // Not every server answers getMetadata, and a 404 from
152                    // it doesn't prove the PV isn't archived (a proxy may
153                    // not pass it on, or a cluster member may not know the
154                    // PV): the data request decides.
155                    Err(e) => {
156                        log::debug!("{}: no metadata: {e}", job.pv);
157                        None
158                    }
159                }
160            } else {
161                None
162            };
163            let (op, from) = match &info {
164                Some(i) if !i.binnable() => (Operator::Raw, job.raw_from.max(job.from)),
165                _ => (job.op, job.from),
166            };
167            // An array is not plotted: the metadata says it is archived,
168            // and that is all the plot needs. A day of a busy waveform
169            // could be gigabytes.
170            let result = if let Some(series) = info.as_ref().and_then(|i| array_series(&job.pv, i))
171            {
172                Ok(Some(series))
173            } else {
174                client.get_data(&Query::new(&job.pv, from, job.to, op))
175            };
176            let result = result.map_err(|e| {
177                log::debug!("{}: {e}", job.pv);
178                if e.is_not_archived() {
179                    check_404(&client, &job.pv, e)
180                } else {
181                    e
182                }
183            });
184            inbox.lock().push(Done {
185                trace: job.trace,
186                generation: job.generation,
187                from,
188                to: job.to,
189                op,
190                append: job.append,
191                info,
192                result,
193            });
194        });
195    }

197    /// Search `client` for `search`, as request number `n`.
198    pub fn search(
199        &self,
200        pvs: &PvManager,
201        client: &ArchiverClient,
202        search: Search,
203        n: u64,
204        repaint: Option<Repaint>,
205    ) {
206        let client = client.clone();
207        let found = self.found.clone();
208        let store = move |names: Result<Vec<String>, String>| {
209            // Searches can finish out of order: keep only the newest answer
210            // not yet taken in, so an older one never hides a newer one.
211            let mut slot = found.lock();
212            if slot.as_ref().is_none_or(|(m, _)| n > *m) {
213                *slot = Some((n, names));
214            }
215        };
216        let panicked = {
217            let store = store.clone();
218            move || store(Err("the search failed unexpectedly".into()))
219        };
220        self.run(pvs, repaint, panicked, move || {
221            store(client.search(&search).map_err(|e| e.to_string()));
222        });
223    }
224
225    /// Run `work` as a blocking task, `panicked` instead reporting its
226    /// failure if it panics. The busy count goes back down and the plot
227    /// is repainted however `work` ends.
228    fn run(
229        &self,
230        pvs: &PvManager,
231        repaint: Option<Repaint>,
232        panicked: impl FnOnce() + Send + 'static,
233        work: impl FnOnce() + Send + 'static,
234    ) {
235        /// Undoes the busy count (and repaints) on drop, panic or not.
236        struct Busy(Arc<AtomicUsize>, Option<Repaint>);
237        impl Drop for Busy {
238            fn drop(&mut self) {
239                self.0.fetch_sub(1, Ordering::Relaxed);
240                if let Some(r) = &self.1 {
241                    r.now();
242                }
243            }
244        }
245        self.busy.fetch_add(1, Ordering::Relaxed);
246        let busy = Busy(self.busy.clone(), repaint);
247        let task = move || {
248            let _busy = busy;
249            if std::panic::catch_unwind(std::panic::AssertUnwindSafe(work)).is_err() {
250                log::error!("an archiver request panicked");
251                panicked();
252            }
253        };
254        #[cfg(not(target_arch = "wasm32"))]
255        drop(pvs.runtime().spawn_blocking(task));
256        #[cfg(target_arch = "wasm32")]
257        {
258            // No blocking HTTP in a browser (yet): live only.
259            let _ = (pvs, panicked);
260            drop(task);
261        }
262    }
263}

265/// A 404 means "not archived" only when the archiver itself said so. With
266/// a wrong `--archiver` base (port or context path) Tomcat answers 404 to
267/// every URL, so check with a search for the exact name, which any
268/// archiver answers: only if that is a 404 too is the URL wrong, and that
269/// is reported as an error. Any other failure of the check (a timeout, a
270/// server error) proves nothing, so the PV stays "not archived".
271fn check_404(client: &ArchiverClient, pv: &str, e: archiver::Error) -> archiver::Error {
272    match client.search(&Search::exact(pv, 1)) {
273        Err(check) if check.is_not_archived() => {
274            log::debug!("{pv}: {e}; the PV search: {check}");
275            archiver::Error::BadUrl(format!(
276                "{}: not an archiver (HTTP 404 for data and search)",
277                client.base()
278            ))
279        }
280        Err(check) => {
281            log::debug!("{pv}: {e}; the PV search failed too: {check}");
282            e
283        }
284        Ok(_) => e,
285    }
286}

288/// The window to fetch for a visible `from..to`: padded by [`PAD`] of the
289/// span either side.
290pub fn padded(from: f64, to: f64) -> (f64, f64) {
291    let pad = (to - from) * PAD;
292    (from - pad, to + pad)
293}
294
295/// The operator for a plot `width_px` device pixels wide showing part of a
296/// padded window: about one bin per device pixel of the padded window (at
297/// most `MAX_BINS`), or raw. Finer bins only alias: the means zig-zag into
298/// a solid block where the envelope already shows the extremes.
299pub fn operator(width_px: f32, raw: bool) -> Operator {
300    if raw {
301        Operator::Raw
302    } else {
303        Operator::for_width(width_px.max(1.0) * (1.0 + 2.0 * PAD as f32))
304    }
305}

307/// The longest span fetched raw, for traces that cannot be binned
308/// (waveforms, text, enums): a week of a busy waveform could be gigabytes.
309/// Wider windows show only the latest part of their history.
310pub const RAW_MAX_SPAN: f64 = 86_400.0;
311
312/// The start of the history a trace needs for a visible `from..to`: all of
313/// it, or for a raw-only trace no more than [`RAW_MAX_SPAN`].
314pub fn needed_from(from: f64, to: f64, raw: bool) -> f64 {
315    if raw {
316        from.max(to - RAW_MAX_SPAN)
317    } else {
318        from
319    }
320}
321
322/// Refresh binned history no more often than every this many bin widths
323/// while following now.
324const BINS_PER_REFRESH: f64 = 3.0;
325
326/// How long after fetching `got` with `op` to refresh it while following
327/// now: `base` for raw history (only its tail is fetched, which is
328/// cheap), but for bins no sooner than [`BINS_PER_REFRESH`] bin widths.
329/// Refetching a week of bins every 30 s would load the appliance for no
330/// visible change: a bin is minutes wide, and the right edge is drawn from
331/// live data anyway.
332pub fn refresh_after(base: f64, got: (f64, f64), op: Operator, binned: bool) -> f64 {
333    match op {
334        Operator::Optimized(n) if binned && n > 0 => {
335            let bin = (got.1 - got.0) / f64::from(n);
336            (BINS_PER_REFRESH * bin).max(base)
337        }
338        _ => base,
339    }
340}

342/// Whether raw history fetched as `optimized_N` (because there were at
343/// most N samples) has grown, through live tail refreshes, past N samples:
344/// then it should be fetched whole again, to come back binned, rather than
345/// grow without end.
346pub fn outgrown(op: Option<Operator>, len: usize) -> bool {
347    matches!(op, Some(Operator::Optimized(n)) if len > n as usize)
348}
349
350/// How long to wait before asking again after `failures` errors in a row:
351/// `base`, doubling each time, at most five minutes.
352pub fn backoff(base: f64, failures: u32) -> f64 {
353    (base * 2f64.powi(failures.saturating_sub(1).min(10) as i32)).min(300.0)
354}
355
356/// Whether data fetched for `got` (and binned or not) still serves the
357/// visible `from..to`: it covers it, and (for bins) it is not zoomed in so
358/// far that the bins show as steps.
359pub fn covers(got: (f64, f64), binned: bool, from: f64, to: f64) -> bool {
360    let (a, b) = got;
361    let inside = a <= from && b >= to;
362    let fine_enough = !binned || (to - from) / (b - a) >= 0.35;
363    inside && fine_enough
364}

366/// For an array PV (more than one element, by its metadata): an empty
367/// waveform series standing for its history, which is not fetched.
368fn array_series(pv: &str, info: &PvInfo) -> Option<TimeSeries> {
369    info.element_count
370        .is_some_and(|n| n > 1)
371        .then(|| TimeSeries {
372            meta: archiver::Meta {
373                name: pv.to_string(),
374                egu: info.units.clone(),
375                prec: info.precision,
376                ..Default::default()
377            },
378            kind: archiver::ValueKind::Waveform,
379            binned: false,
380            samples: Vec::new(),
381        })
382}

384#[cfg(test)]
385mod tests {
386    use super::*;

388    #[test]
389    fn windows_are_padded_and_binned_to_the_width() {
390        assert_eq!(padded(100.0, 200.0), (75.0, 225.0));
391        // 1000 visible pixels in a window 1.5 times as wide: 1500 bins,
392        // one per pixel of the padded window.
393        assert_eq!(operator(1000.0, false), Operator::Optimized(1500));
394        assert_eq!(operator(4000.0, false), Operator::Optimized(4000));
395        assert_eq!(operator(1000.0, true), Operator::Raw);
396    }
397
398    #[test]
399    fn refetch_when_out_of_cover_or_too_coarse() {
400        let got = padded(100.0, 200.0);
401        assert!(covers(got, true, 100.0, 200.0));
402        // Panned within the padding.
403        assert!(covers(got, true, 120.0, 220.0));
404        assert!(!covers(got, true, 150.0, 250.0));
405        // Zoomed in on bins: refetch for finer ones; raw data never needs it.
406        assert!(!covers(got, true, 140.0, 160.0));
407        assert!(covers(got, false, 140.0, 160.0));
408    }
409
410    #[test]
411    fn raw_history_is_capped_and_errors_back_off() {
412        let week = 7.0 * 86_400.0;
413        assert_eq!(needed_from(0.0, week, false), 0.0);
414        assert_eq!(needed_from(0.0, week, true), week - RAW_MAX_SPAN);
415        assert_eq!(needed_from(100.0, 200.0, true), 100.0);
416        assert_eq!(backoff(10.0, 1), 10.0);
417        assert_eq!(backoff(10.0, 3), 40.0);
418        assert_eq!(backoff(10.0, 40), 300.0);
419    }
420
421    #[test]
422    fn binned_history_refreshes_at_the_bin_pace() {
423        let week = padded(0.0, 7.0 * 86_400.0);
424        // 1500 bins over 10.5 days: 605 s each, refreshed every 3 of them.
425        let every = refresh_after(30.0, week, Operator::Optimized(1500), true);
426        assert!((every - 1814.4).abs() < 1e-6, "{every}");
427        // Fine bins, raw history, and raw answers to `optimized`: 30 s.
428        let hour = padded(0.0, 3600.0);
429        assert_eq!(
430            refresh_after(30.0, hour, Operator::Optimized(1500), true),
431            30.0
432        );
433        assert_eq!(refresh_after(30.0, week, Operator::Raw, false), 30.0);
434        assert_eq!(
435            refresh_after(30.0, week, Operator::Optimized(1500), false),
436            30.0
437        );
438    }

440    #[test]
441    fn raw_history_past_n_samples_is_rebinned() {
442        let op = Some(Operator::Optimized(100));
443        assert!(!outgrown(op, 100));
444        assert!(outgrown(op, 101));
445        assert!(!outgrown(Some(Operator::Raw), 1_000_000));
446        assert!(!outgrown(None, 1_000_000));
