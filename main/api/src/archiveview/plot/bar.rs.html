13/// Wait this long after typing before searching (seconds).
14const SEARCH_DELAY: f64 = 0.2;

19/// Below this width the bar is compact: the pill is a menu and the

23/// The search box.
24#[derive(Default)]
25pub struct SearchState {
26    text: String,
27    regex: bool,
28    /// The text as last searched for, and when it was last edited.
29    searched: String,
30    edited_at: f64,
31    /// The latest search's number; older answers are dropped.
32    asked: u64,
33    waiting: bool,
34    pub(super) results: Vec<String>,
35    error: Option<String>,
36    /// The highlighted row of the popup.

39    /// Give the box the keyboard on the next frame (`/`).

53pub fn top_bar(
54    plot: &mut Plot,
55    ui: &mut egui::Ui,
56    menu: Option<&mut dyn FnMut(&mut egui::Ui)>,
57    now: f64,
58) {
59    let compact = ui.available_width() < COMPACT_WIDTH;
60    ui.spacing_mut().button_padding = Vec2::new(10.0, 5.0);
61    ui.spacing_mut().item_spacing.x = 6.0;
62    let h = icons::height(ui);
63    let row = Vec2::new(ui.available_width(), h);
64    ui.allocate_ui_with_layout(row, Layout::right_to_left(Align::Center), |ui| {
65        let menu_button = icons::button(ui, Icon::Menu, false).on_hover_text("More");
66        egui::Popup::menu(&menu_button).show(|ui| {
67            if compact {
68                if ui.button("Autoscale values").clicked() {
69                    plot.y.autoscale = true;
70                }
71                if ui.button("Reset view").clicked() {
72                    plot.reset();
73                }
74                ui.separator();
75            }
76            if ui.checkbox(&mut plot.y.log, "Log scale").changed() {
77                plot.fit_log_range();
78            }
79            if let Some(menu) = menu {
80                ui.separator();
81                menu(ui);
82            }
83            ui.separator();
84            ui.label(
85                egui::RichText::new(
86                    "drag: pan · wheel: zoom time · shift+wheel: zoom values\n\
87                     shift+drag: zoom to box · double-click: reset\n\
88                     / search · ←→ pan · +/- zoom · L live · A autoscale",
89                )
90                .small()
91                .weak(),
92            );
93        });
94        if !compact {
95            if icons::button(ui, Icon::Home, false)
96                .on_hover_text(
97                    "Reset: the preset span up to now, values fitted (Home, double-click)",
98                )
99                .clicked()
100            {
101                plot.reset();
102            }
103            if icons::button(ui, Icon::Fit, plot.y.autoscale)
104                .on_hover_text("Autoscale values (A)")
105                .clicked()
106            {
107                plot.y.autoscale = true;
108            }
109        }
110        live_pill(plot, ui, now);
111        if compact {
112            range_combo(plot, ui);
113        } else {
114            range_pill(plot, ui, now);
115        }
116        // The search box takes the rest, from the left.
117        let rest = Vec2::new(ui.available_width(), h);
118        ui.allocate_ui_with_layout(rest, Layout::left_to_right(Align::Center), |ui| {
119            let w = ui.available_width().min(460.0);
120            search_box(plot, ui, w, now);
121        });
122    });
123}

139fn search_box(plot: &mut Plot, ui: &mut egui::Ui, width: f32, now: f64) {
140    let h = icons::height(ui);
141    let (rect, _) = ui.allocate_exact_size(Vec2::new(width, h), Sense::hover());
142    let edit_id = plot.id.with("search");
143    let focused = ui.ctx().memory(|m| m.has_focus(edit_id));
144    pill_frame(ui, rect);
145    if focused {
146        ui.painter().rect_stroke(
147            rect,
148            h / 2.0,
149            Stroke::new(1.5, ui.visuals().selection.bg_fill),
150            StrokeKind::Inside,
151        );
152    }
153    let icon = Rect::from_min_size(rect.min + Vec2::new(8.0, 0.0), Vec2::new(h * 0.6, h));
154    Icon::Search.paint(ui.painter(), icon, h * 0.5, ui.visuals().weak_text_color());
155
156    // Keys for the result list, taken before the text field sees them.
157    let s = &mut plot.search;
158    let rows = rows(s, plot.archiver.is_some());
159    if focused && s.open {
160        let (down, up) = ui.input_mut(|i| {
161            (
162                i.consume_key(egui::Modifiers::NONE, Key::ArrowDown),
163                i.consume_key(egui::Modifiers::NONE, Key::ArrowUp),
164            )
165        });
166        if down && s.selected + 1 < rows.len() {
167            s.selected += 1;
168        }
169        if up {
170            s.selected = s.selected.saturating_sub(1);
171        }
172    }
173
174    // On the right inside the box: ✕ to clear when there is text, and the
175    // regex toggle while the box has the focus or the pointer (or regex is
176    // on), so an idle box is just a search field.
177    let regex_w = 28.0;
178    let clear_w = 20.0;
179    let hovered = ui.rect_contains_pointer(rect);
180    let show_regex = plot.archiver.is_some() && (focused || hovered || s.regex);
181    let show_clear = !s.text.is_empty();
182    let right = rect.right()
183        - 6.0
184        - if show_regex { regex_w } else { 0.0 }
185        - if show_clear { clear_w } else { 0.0 };
186    let inner = Rect::from_min_max(
187        Pos2::new(icon.right() + 2.0, rect.top()),
188        Pos2::new(right.max(icon.right() + 40.0), rect.bottom()),
189    );
190    let mut edit_ui = ui.new_child(
191        egui::UiBuilder::new()
192            .max_rect(inner)
193            .layout(Layout::left_to_right(Align::Center)),
194    );
195    let hint = if plot.archiver.is_some() {
196        "Search PVs…"
197    } else {
198        "Add a PV…"
199    };
200    if !focused && s.text.is_empty() {
201        // The shortcut, as a key cap.
202        let cap = Rect::from_center_size(
203            Pos2::new(inner.right() - 10.0, rect.center().y),
204            Vec2::new(16.0, 16.0),
205        );
206        let v = ui.visuals();
207        ui.painter().rect(
208            cap,
209            4.0,
210            Color32::TRANSPARENT,
211            Stroke::new(1.0, v.weak_text_color().gamma_multiply(0.6)),
212            StrokeKind::Inside,
213        );
214        ui.painter().text(
215            cap.center(),
216            egui::Align2::CENTER_CENTER,
217            "/",
218            egui::FontId::proportional(11.0),
219            v.weak_text_color(),
220        );
221    }
222    let resp = edit_ui.add(
223        egui::TextEdit::singleline(&mut s.text)
224            .id(edit_id)
225            .frame(egui::Frame::NONE)
226            .hint_text(hint)
227            .desired_width(inner.width()),
228    );
229    if std::mem::take(&mut s.focus) {
230        resp.request_focus();
231    }
232    if resp.changed() {
233        s.edited_at = now;
234        s.selected = 0;
235        s.open = !s.text.trim().is_empty();
236    }
237    if resp.gained_focus() && !s.text.trim().is_empty() {
238        s.open = true;
239    }
240
241    // The regex toggle, inside the box on the right.
242    let toggle = Rect::from_center_size(
243        Pos2::new(rect.right() - regex_w / 2.0 - 5.0, rect.center().y),
244        Vec2::new(regex_w, h - 8.0),
245    );
246    if show_clear {
247        let x = if show_regex {
248            toggle.left() - clear_w / 2.0
249        } else {
250            rect.right() - 5.0 - clear_w / 2.0
251        };
252        let button = Rect::from_center_size(Pos2::new(x, rect.center().y), Vec2::splat(18.0));
253        let r = ui
254            .interact(button, plot.id.with("clear"), Sense::click())
255            .on_hover_text("Clear");
256        let v = ui.visuals();
257        if r.hovered() {
258            ui.painter()
259                .circle_filled(button.center(), 9.0, v.widgets.hovered.weak_bg_fill);
260        }
261        let k = Stroke::new(
262            1.4,
263            if r.hovered() {
264                v.text_color()
265            } else {
266                v.weak_text_color()
267            },
268        );
269        let (m, d) = (button.center(), 3.5);
270        ui.painter()
271            .line_segment([m + Vec2::new(-d, -d), m + Vec2::new(d, d)], k);
272        ui.painter()
273            .line_segment([m + Vec2::new(d, -d), m + Vec2::new(-d, d)], k);
274        if r.clicked() {
275            s.text.clear();
276            s.searched.clear();
277            s.results.clear();
278            s.error = None;
279            s.waiting = false;
280            s.open = false;
281            s.focus = true;
282            ui.ctx().request_repaint();
283        }
284    }
285    if show_regex {
286        let r = ui
287            .interact(toggle, plot.id.with("regex"), Sense::click())
288            .on_hover_text("Match a regular expression (else * and ? wildcards)");
289        if r.clicked() {
290            s.regex = !s.regex;
291            s.searched.clear();
292            s.edited_at = now;
293            s.focus = true;
294            ui.ctx().request_repaint();
295        }
296        let v = ui.visuals();
297        let (fill, fg) = if s.regex {
298            (v.selection.bg_fill, v.selection.stroke.color)
299        } else if r.hovered() {
300            (v.widgets.hovered.weak_bg_fill, v.text_color())
301        } else {
302            (Color32::TRANSPARENT, v.weak_text_color())
303        };
304        ui.painter()
305            .rect_filled(toggle, toggle.height() / 2.0, fill);
306        ui.painter().text(
307            toggle.center(),
308            egui::Align2::CENTER_CENTER,
309            ".*",
310            egui::FontId::monospace(12.0),
311            fg,
312        );
313    }
314
315    // Search once typing pauses.
316    let text = s.text.trim().to_string();
317    if let Some(client) = &plot.archiver
318        && !text.is_empty()
319        && text != s.searched
320        && now - s.edited_at >= SEARCH_DELAY
321    {
322        s.searched = text.clone();
323        s.asked += 1;
324        s.waiting = true;
325        let search = if s.regex {
326            Search::regex(text.clone(), SEARCH_LIMIT)
327        } else {
328            Search::from_user(&text, SEARCH_LIMIT)
329        };
330        plot.fetcher
331            .search(&plot.pvs, client, search, s.asked, plot.repaint.clone());
332    } else if !text.is_empty() && text != s.searched && plot.archiver.is_some() {
333        ui.ctx()
334            .request_repaint_after(std::time::Duration::from_millis(60));
335    }
336
337    // Enter adds the highlighted row, or what was typed.
338    let mut add: Vec<String> = Vec::new();
339    let enter = resp.lost_focus() && ui.input(|i| i.key_pressed(Key::Enter));
340    if enter {
341        if let Some(row) = rows.get(s.selected) {
342            add = row.names();
343        }
344        resp.request_focus();
345    }
346    if focused && ui.input(|i| i.key_pressed(Key::Escape)) {
347        if s.open {
348            s.open = false;
349        } else {
350            s.text.clear();
351            ui.ctx().memory_mut(|m| m.surrender_focus(edit_id));
352        }
353    }
354
355    // The popup of results.
356    if s.open && !text.is_empty() {
357        let pos = rect.left_bottom() + Vec2::new(0.0, 4.0);
358        let area = egui::Area::new(plot.id.with("search-popup"))
359            .order(egui::Order::Foreground)
360            .fixed_pos(pos)
361            .show(ui.ctx(), |ui| {
362                egui::Frame::popup(ui.style())
363                    .corner_radius(10.0)
364                    .show(ui, |ui| {
365                        ui.set_width(width - 14.0);
366                        ui.spacing_mut().item_spacing.y = 2.0;
367                        egui::ScrollArea::vertical()
368                            .max_height(320.0)
369                            .min_scrolled_height((rows.len() as f32 * 26.0).min(320.0))
370                            .show(ui, |ui| {
371                                for (i, row) in rows.iter().enumerate() {
372                                    let picked = i == s.selected;
373                                    let label = row.label();
374                                    let r = ui.add(
375                                        egui::Button::selectable(picked, label)
376                                            .min_size(Vec2::new(ui.available_width(), 24.0)),
377                                    );
378                                    if picked && enter {
379                                        r.scroll_to_me(None);
380                                    }
381                                    if r.hovered() && ui.input(|i| i.pointer.delta() != Vec2::ZERO)
382                                    {
383                                        s.selected = i;
384                                    }
385                                    if r.clicked() {
386                                        add = row.names();
387                                    }
388                                }
389                            });
390                        let note = if s.waiting {
391                            Some("Searching the archiver…".to_string())
392                        } else if let Some(e) = &s.error {
393                            Some(format!("Search failed: {e}"))
394                        } else if plot.archiver.is_some()
395                            && s.searched == text
396                            && s.results.is_empty()
397                        {
398                            Some("No archived PVs match".to_string())
399                        } else if s.results.len() as i32 >= SEARCH_LIMIT {
400                            Some(format!("First {SEARCH_LIMIT} matches: type more to narrow"))
401                        } else {
402                            None
403                        };
404                        if let Some(note) = note {
405                            ui.add_space(2.0);
406                            ui.horizontal(|ui| {
407                                if s.waiting {
408                                    ui.spinner();
409                                }
410                                ui.label(egui::RichText::new(note).small().weak());
411                            });
412                        }
413                    });
414            });
415        // A click anywhere else closes it.
416        let clicked_away = ui.input(|i| {
417            i.pointer.any_pressed()
418                && i.pointer
419                    .interact_pos()
420                    .is_some_and(|p| !area.response.rect.contains(p) && !rect.contains(p))
421        });
422        if clicked_away {
423            s.open = false;
424        }
425    }
426
427    if !add.is_empty() {
428        s.text.clear();
429        s.searched.clear();
430        s.results.clear();
431        s.open = false;
432        plot.add_pvs(&add);
433    }
434}

436/// A row of the search popup.
437enum Row {
438    /// What was typed, added as it is (one or more names).
439    Typed(Vec<String>),
440    Found(String),
441}
442
443impl Row {
444    fn names(&self) -> Vec<String> {
445        match self {
446            Row::Typed(n) => n.clone(),
447            Row::Found(n) => vec![n.clone()],
448        }
449    }
450
451    fn label(&self) -> egui::RichText {
452        match self {
453            Row::Typed(n) => egui::RichText::new(format!("+  Add “{}” as typed", n.join("”, “"))),
454            Row::Found(n) => egui::RichText::new(n),
455        }
456    }
457}

459/// The popup's rows, the first highlighted (what Enter adds): a match
460/// that is exactly the text typed, then the other archiver matches, then
461/// the text as typed (unless it is a pattern, or exactly a match). Text
462/// that can only be added as typed (a `pva://` or `sim://` name, or
463/// several names) comes first instead.
464fn rows(s: &SearchState, archiver: bool) -> Vec<Row> {
465    let text = s.text.trim();
466    let mut found: Vec<&String> = if s.searched == text {
467        s.results.iter().collect()
468    } else {
469        Vec::new()
470    };
471    if let Some(i) = found.iter().position(|r| *r == text) {
472        let exact = found.remove(i);
473        found.insert(0, exact);
474    }
475    let typed: Vec<String> = text
476        .split([' ', ',', '\t'])
477        .filter(|n| !n.is_empty())
478        .map(str::to_string)
479        .collect();
480    let pattern = archiver && (s.regex || text.contains(['*', '?']));
481    let mut rows: Vec<Row> = found.into_iter().cloned().map(Row::Found).collect();
482    if !typed.is_empty() && !pattern && !s.results.iter().any(|r| r == text) {
483        let as_typed = typed.len() > 1 || text.contains("://");
484        let row = Row::Typed(typed);
485        if as_typed {
486            rows.insert(0, row);
487        } else {
488            rows.push(row);
489        }
490    }
491    rows
492}

494/// Whether typing has paused for a search not yet sent.
495pub fn search_due(plot: &Plot) -> bool {
496    let text = plot.search.text.trim();
497    plot.archiver.is_some() && !text.is_empty() && text != plot.search.searched
498}

500/// Take in a finished search.
501pub fn take_found(plot: &mut Plot, _now: f64) {
502    if let Some((n, found)) = plot.fetcher.take_found() {
503        let s = &mut plot.search;
504        if n != s.asked {
505            return;
506        }
507        s.waiting = false;
508        match found {
509            Ok(names) => {
510                s.results = names;
511                s.error = None;
512            }
513            Err(e) => {
514                s.results.clear();
515                s.error = Some(e);
516            }
517        }
518        s.selected = 0;
519    }
520}

524/// The segmented control: 1h 8h 1d 7d and a custom segment showing a span
525/// that is not a preset.
526fn range_pill(plot: &mut Plot, ui: &mut egui::Ui, now: f64) {
527    let h = icons::height(ui);
528    let font = egui::TextStyle::Button.resolve(ui.style());
529    let current = plot.window.preset();
530    let custom_label = if current.is_none() {
531        format!("{}  ⋯", format_span(plot.window.span))
532    } else {
533        "⋯".to_string()
534    };
535    let labels: Vec<String> = PRESETS
536        .iter()
537        .map(|(_, l)| l.to_string())
538        .chain(std::iter::once(custom_label))
539        .collect();
540    let widths: Vec<f32> = labels
541        .iter()
542        .map(|l| super::render::text_size(ui.painter(), l, &font).x + 20.0)
543        .collect();
544    let total: f32 = widths.iter().sum::<f32>() + 4.0;
545    let (rect, _) = ui.allocate_exact_size(Vec2::new(total, h), Sense::hover());
546    pill_frame(ui, rect);
547    let mut x = rect.left() + 2.0;
548    let v = ui.visuals().clone();
549    let mut custom_rect = rect;
550    for (i, (label, w)) in labels.iter().zip(&widths).enumerate() {
551        let seg = Rect::from_min_size(Pos2::new(x, rect.top() + 2.0), Vec2::new(*w, h - 4.0));
552        x += w;
553        let is_custom = i == PRESETS.len();
554        let on = if is_custom {
555            current.is_none()
556        } else {
557            current == Some(i)
558        };
559        let r = ui.interact(seg, plot.id.with(("range", i)), Sense::click());
560        let r = if is_custom {
561            custom_rect = seg;
562            r.on_hover_text("Another span, or a fixed range")
563        } else {
564            r.on_hover_text(format!("The last {label}"))
565        };
566        let (fill, fg) = if on {
567            (v.selection.bg_fill, v.selection.stroke.color)
568        } else if r.hovered() {
569            (v.widgets.hovered.weak_bg_fill, v.text_color())
570        } else {
571            (Color32::TRANSPARENT, v.text_color())
572        };
573        ui.painter().rect_filled(seg, seg.height() / 2.0, fill);
574        ui.painter().text(
575            seg.center(),
576            egui::Align2::CENTER_CENTER,
577            label,
578            font.clone(),
579            fg,
580        );
581        if r.clicked() {
582            if is_custom {
583                open_custom(plot, now);
584            } else {
585                plot.preset(PRESETS[i].0);
586            }
587        }
588    }
589    custom_popover(plot, ui, custom_rect, now);
590}

592/// The range as a menu, for a narrow bar: one pill showing the span, which
593/// opens the presets.
594fn range_combo(plot: &mut Plot, ui: &mut egui::Ui) {
595    let label = match plot.window.preset() {
596        Some(i) => PRESETS[i].1.to_string(),
597        None => format_span(plot.window.span),
598    };
599    let label = format!("{label}  ▾");
600    let h = icons::height(ui);
601    let font = egui::TextStyle::Button.resolve(ui.style());
602    let w = super::render::text_size(ui.painter(), &label, &font).x + 24.0;
603    let (rect, r) = ui.allocate_exact_size(Vec2::new(w, h), Sense::click());
604    pill_frame(ui, rect);
605    let v = ui.visuals();
606    if r.hovered() {
607        let inner = rect.shrink(2.0);
608        ui.painter()
609            .rect_filled(inner, inner.height() / 2.0, v.widgets.hovered.weak_bg_fill);
610    }
611    ui.painter().text(
612        rect.center(),
613        egui::Align2::CENTER_CENTER,
614        &label,
615        font,
616        v.text_color(),
617    );
618    let r = r.on_hover_text("Time range");
619    egui::Popup::menu(&r).show(|ui| {
620        let current = plot.window.preset();
621        for (i, (span, name)) in PRESETS.iter().enumerate() {
622            if ui.selectable_label(current == Some(i), *name).clicked() {
623                plot.preset(*span);
624            }
625        }
626        if ui.selectable_label(current.is_none(), "Custom…").clicked() {
627            open_custom(plot, now());
628        }
629    });
630    if plot.custom.open {
631        custom_popover(plot, ui, rect, now());
632    }
633}

635fn open_custom(plot: &mut Plot, now: f64) {
636    let c = &mut plot.custom;
637    c.open = !c.open;
638    let (from, to) = plot.window.range(now);
639    c.span = format_span(plot.window.span);
640    c.from = format_local(from);
641    c.to = if plot.window.live {
642        "now".into()
643    } else {
644        format_local(to)
645    };
646    c.error = None;
647}

649/// The custom range popover under `anchor`: a span up to now, or fixed
650/// start and end times.
651fn custom_popover(plot: &mut Plot, ui: &mut egui::Ui, anchor: Rect, now: f64) {
652    if !plot.custom.open {
653        return;
654    }
655    let width = 300.0;
656    let pos = Pos2::new(
657        (anchor.right() - width).max(ui.ctx().content_rect().left() + 4.0),
658        anchor.bottom() + 6.0,
659    );
660    let mut apply: Option<Window> = None;
661    let c = &mut plot.custom;
662    let area = egui::Area::new(plot.id.with("custom-range"))
663        .order(egui::Order::Foreground)
664        .fixed_pos(pos)
665        .show(ui.ctx(), |ui| {
666            egui::Frame::popup(ui.style())
667                .corner_radius(10.0)
668                .inner_margin(12.0)
669                .show(ui, |ui| {
670                    ui.set_width(width - 24.0);
671                    ui.spacing_mut().item_spacing.y = 8.0;
672                    ui.label(egui::RichText::new("Last").strong());
673                    ui.horizontal(|ui| {
674                        let r = ui.add(
675                            egui::TextEdit::singleline(&mut c.span)
676                                .hint_text("90m, 3h, 2 days")
677                                .desired_width(150.0),
678                        );
679                        let go = ui.button("Show").clicked()
680                            || (r.lost_focus() && ui.input(|i| i.key_pressed(Key::Enter)));
681                        if go {
682                            match parse_span(&c.span) {
683                                Some(s) => apply = Some(Window::live(s)),
684                                None => c.error = Some(format!("Not a span: {:?}", c.span)),
685                            }
686                        }
687                    });
688                    ui.separator();
689                    ui.label(egui::RichText::new("From … to").strong());
690                    egui::Grid::new("custom-range-grid")
691                        .num_columns(2)
692                        .spacing([8.0, 6.0])
693                        .show(ui, |ui| {
694                            ui.label("From");
695                            ui.add(
696                                egui::TextEdit::singleline(&mut c.from)
697                                    .hint_text("2026-10-17 09:00")
698                                    .desired_width(170.0),
699                            );
700                            ui.end_row();
701                            ui.label("To");
702                            ui.add(
703                                egui::TextEdit::singleline(&mut c.to)
704                                    .hint_text("now")
705                                    .desired_width(170.0),
706                            );
707                            ui.end_row();
708                        });
709                    if ui.button("Show range").clicked() {
710                        let to =
711                            if c.to.trim().eq_ignore_ascii_case("now") || c.to.trim().is_empty() {
712                                Some(now)
713                            } else {
714                                parse_local(&c.to)
715                            };
716                        match (parse_local(&c.from), to) {
717                            (Some(a), Some(b)) if a < b => {
718                                let mut w = Window::fixed(a, b);
719                                if b >= now {
720                                    w = Window::live(b - a);
721                                }
722                                apply = Some(w);
723                            }
724                            (Some(_), Some(_)) => c.error = Some("From must be before to".into()),
725                            _ => c.error = Some("Times read YYYY-MM-DD HH:MM[:SS] (local)".into()),
726                        }
727                    }
728                    if let Some(e) = &c.error {
729                        ui.colored_label(ui.visuals().error_fg_color, e);
730                    }
731                });
732        });
733    let away = ui.input(|i| {
734        i.pointer.any_pressed()
735            && i.pointer
736                .interact_pos()
737                .is_some_and(|p| !area.response.rect.contains(p) && !anchor.contains(p))
738    });
739    if away || ui.input(|i| i.key_pressed(Key::Escape)) {
740        plot.custom.open = false;
741    }
742    if let Some(w) = apply {
743        plot.custom.open = false;
744        plot.window = w;
745        plot.home_span = w.span;
746        plot.glide = 0.0;
747    }
748}

752/// `● Live`: the dot shows while following now, pulsing for a few seconds
753/// after it starts and while the pointer is over the plot; click to pause
754/// or to jump to now.
755fn live_pill(plot: &mut Plot, ui: &mut egui::Ui, now: f64) {
756    let h = icons::height(ui);
757    let font = egui::TextStyle::Button.resolve(ui.style());
758    let text_w = super::render::text_size(ui.painter(), "Live", &font).x;
759    let size = Vec2::new(text_w + h * 0.9 + 18.0, h);
760    let (rect, r) = ui.allocate_exact_size(size, Sense::click());
761    let live = plot.window.live;
762    let r = r.on_hover_text(if live {
763        "Following now: click to pause (L)"
764    } else {
765        "Jump to now and follow it (L)"
766    });
767    let v = ui.visuals();
768    let dark = v.dark_mode;
769    let accent = crate::style::live(dark);
770    let fill = if live {
771        accent.gamma_multiply(if dark { 0.22 } else { 0.10 })
772    } else if r.hovered() {
773        v.widgets.hovered.weak_bg_fill
774    } else {
775        v.extreme_bg_color
776    };
777    let stroke = if live {
778        Stroke::new(1.0, accent.gamma_multiply(0.5))
779    } else {
780        v.widgets.noninteractive.bg_stroke
781    };
782    ui.painter()
783        .rect(rect, h / 2.0, fill, stroke, StrokeKind::Inside);
784    let dot = Pos2::new(rect.left() + 12.0 + 4.0, rect.center().y);
785    if live && plot.pulse {
786        // A slow breath: a halo growing and fading every two seconds.
787        let phase = (ui.input(|i| i.time) % 2.0 / 2.0) as f32;
788        ui.painter().circle_filled(
789            dot,
790            4.0 + 5.0 * phase,
791            accent.gamma_multiply(0.45 * (1.0 - phase)),
792        );
793        ui.painter().circle_filled(dot, 4.0, accent);
794    } else if live {
795        // Still, once it has said so: no animation in an idle window.
796        ui.painter()
797            .circle_filled(dot, 6.5, accent.gamma_multiply(0.18));
798        ui.painter().circle_filled(dot, 4.0, accent);
799    } else {
800        ui.painter()
801            .circle_stroke(dot, 3.5, Stroke::new(1.5, v.weak_text_color()));
802    }
803    let fg = if live {
804        v.strong_text_color()
805    } else {
806        v.text_color()
807    };
808    ui.painter().text(
809        Pos2::new(dot.x + 10.0, rect.center().y),
810        egui::Align2::LEFT_CENTER,
811        "Live",
812        font,
813        fg,
814    );
815    if r.clicked() {
816        plot.toggle_live(now);
817    }
818}

824    fn state(text: &str, searched: &str, results: &[&str]) -> SearchState {
825        SearchState {

833    #[test]
834    fn rows_list_matches_then_the_text() {
835        let s = state("SINE", "SINE", &["MOCK:SINE", "MOCK:SINE2"]);
836        let r = rows(&s, true);
837        assert_eq!(r.len(), 3);
838        assert_eq!(r[0].names(), ["MOCK:SINE"]);
839        assert_eq!(r[2].names(), ["SINE"]);
840        // Stale results are not offered.
841        let s = state("SIN", "SINE", &["MOCK:SINE"]);
842        assert_eq!(rows(&s, true).len(), 1);
843        // A pattern is not a PV name; several names add together.
844        assert_eq!(rows(&state("MOCK:*", "", &[]), true).len(), 0);
845        let r = rows(&state("a:b, c:d", "", &[]), false);
846        assert_eq!(r[0].names(), ["a:b", "c:d"]);
847        // An exact match is not offered twice.
848        assert_eq!(
849            rows(&state("MOCK:SINE", "MOCK:SINE", &["MOCK:SINE"]), true).len(),
850            1
851        );
852    }

854    #[test]
855    fn enter_adds_an_exact_match_before_other_matches() {
856        // BL:M1:RBV sorts first, but M1:RBV is what was typed.
857        let s = state("M1:RBV", "M1:RBV", &["BL:M1:RBV", "M1:RBV", "M1:RBV2"]);
858        let r = rows(&s, true);
859        assert_eq!(r[0].names(), ["M1:RBV"]);
860        assert_eq!(r.len(), 3, "and not again as typed");
861        // A name only addable as typed comes first.
862        let s = state("pva://M1:RBV", "pva://M1:RBV", &["X:pva"]);
863        assert_eq!(rows(&s, true)[0].names(), ["pva://M1:RBV"]);
