16/// Wait this long after typing before searching (seconds).
17const SEARCH_DELAY: f64 = 0.2;
18
19/// Names a search asks for.

22/// Below this width the bar is compact: the pill is a menu and the
23/// buttons move into the `≡` menu.
24const COMPACT_WIDTH: f32 = 600.0;

26/// The search box.
27#[derive(Default)]
28pub struct SearchState {
29    pub(super) text: String,
30    pub(super) regex: bool,
31    /// The text as last searched for, and when it was last edited.
32    pub(super) searched: String,
33    edited_at: f64,
34    /// The latest search's number; older answers are dropped.
35    asked: u64,
36    waiting: bool,
37    pub(super) results: Vec<String>,
38    error: Option<String>,
39    /// The highlighted row of the popup, once the arrows or the pointer
40    /// have moved it (else `default_row`).
41    selected: Option<usize>,
42    pub(super) open: bool,
43    /// Give the box the keyboard on the next frame (`/`).
44    pub focus: bool,
45    /// As `focus`, but keeping the cursor where it is: the box lent the
46    /// focus to the popup for a click.
47    refocus: bool,
48    /// Select all the text, until the click that focused the box is let go.
49    select_all: bool,
50}
51
52impl SearchState {
53    /// Search again (another archiver): answers to earlier searches are
54    /// dropped.
55    pub fn restart(&mut self) {
56        self.asked += 1;
57        self.waiting = false;
58        self.searched.clear();
59        self.results.clear();
60        self.error = None;
61    }
62}

86pub fn top_bar(
87    plot: &mut Plot,
88    ui: &mut egui::Ui,
89    menu: Option<&mut dyn FnMut(&mut egui::Ui)>,
90    now: f64,
91) {
92    let compact = ui.available_width() < COMPACT_WIDTH;
93    ui.spacing_mut().button_padding = Vec2::new(10.0, 5.0);
94    ui.spacing_mut().item_spacing.x = 6.0;
95    let h = icons::height(ui);
96    let row = Vec2::new(ui.available_width(), h);
97    ui.allocate_ui_with_layout(row, Layout::right_to_left(Align::Center), |ui| {
98        let menu_button = icons::button(ui, Icon::Menu, false).on_hover_text("More");
99        egui::Popup::menu(&menu_button).show(|ui| {
100            if compact {
101                if ui.button("Autoscale values").clicked() {
102                    plot.autoscale();
103                }
104                if ui.button("Reset view").clicked() {
105                    plot.reset();
106                }
107                ui.separator();
108            }
109            let mut log = plot.all_log();
110            if ui
111                .checkbox(&mut log, "Log scale")
112                .on_hover_text("Every value axis; right-click an axis for its own")
113                .changed()
114            {
115                plot.set_log(log);
116            }
117            let any = !plot.history.is_empty();
118            if ui
119                .add_enabled(any, egui::Button::new("Clear search history"))
120                .clicked()
121            {
122                plot.history.clear();
123                plot.history_changed = true;
124            }
125            if let Some(menu) = menu {
126                ui.separator();
127                menu(ui);
128            }
129            ui.separator();
130            ui.label(
131                egui::RichText::new(
132                    "drag: pan · wheel: zoom time · shift+wheel: zoom values\n\
133                     shift+drag: zoom to box · double-click: reset\n\
134                     over an axis: drag or shift+wheel zooms it alone\n\
135                     chip's dot or drag a chip onto an axis: move its trace\n\
136                     / search · ←→ pan · +/- zoom · L live · A autoscale",
137                )
138                .small()
139                .weak(),
140            );
141        });
142        if !compact {
143            if icons::button(ui, Icon::Home, false)
144                .on_hover_text(
145                    "Reset: the preset span up to now, values fitted (Home, double-click)",
146                )
147                .clicked()
148            {
149                plot.reset();
150            }
151            if icons::button(ui, Icon::Fit, plot.autoscaling())
152                .on_hover_text("Autoscale values (A)")
153                .clicked()
154            {
155                plot.autoscale();
156            }
157        }
158        archiver_pill(plot, ui, compact, now);
159        live_pill(plot, ui, now);
160        if compact {
161            range_combo(plot, ui);
162        } else {
163            range_pill(plot, ui, now);
164        }
165        // The search box takes the rest, from the left.
166        let rest = Vec2::new(ui.available_width(), h);
167        ui.allocate_ui_with_layout(rest, Layout::left_to_right(Align::Center), |ui| {
168            let w = ui.available_width().min(460.0);
169            search_box(plot, ui, w, now);
170        });
171    });
172}

188fn search_box(plot: &mut Plot, ui: &mut egui::Ui, width: f32, now: f64) {
189    let h = icons::height(ui);
190    let (rect, _) = ui.allocate_exact_size(Vec2::new(width, h), Sense::hover());
191    let edit_id = plot.id.with("search");
192    let focused = ui.ctx().memory(|m| m.has_focus(edit_id));
193    pill_frame(ui, rect);
194    if focused {
195        ui.painter().rect_stroke(
196            rect,
197            h / 2.0,
198            Stroke::new(1.5, ui.visuals().selection.bg_fill),
199            StrokeKind::Inside,
200        );
201    }
202    let icon = Rect::from_min_size(rect.min + Vec2::new(8.0, 0.0), Vec2::new(h * 0.6, h));
203    Icon::Search.paint(ui.painter(), icon, h * 0.5, ui.visuals().weak_text_color());
204
205    // The plotted PVs, to tick them in the list.
206    let plotted: Vec<String> = plot.traces.iter().map(|t| t.pv.name().key()).collect();
207
208    // Keys for the result list, taken before the text field sees them.
209    let s = &mut plot.search;
210    let rows = rows(s, plot.archiver.is_some(), &plot.history);
211    let mut moved = false;
212    if focused {
213        let (down, up) = ui.input_mut(|i| {
214            (
215                i.consume_key(egui::Modifiers::NONE, Key::ArrowDown),
216                s.open && i.consume_key(egui::Modifiers::NONE, Key::ArrowUp),
217            )
218        });
219        let n = rows.len();
220        let at = s.selected.or_else(|| default_row(&rows));
221        if !s.open {
222            // ↓ in a closed box brings the list back.
223            s.open = down;
224        } else if n > 0 && (down || up) {
225            s.selected = Some(match at {
226                Some(i) if down => (i + 1).min(n - 1),
227                Some(i) => i.saturating_sub(1),
228                None => 0,
229            });
230            moved = true;
231        }
232    }
233    // The highlighted row: what Enter picks.
234    let picked = s
235        .selected
236        .or_else(|| default_row(&rows))
237        .filter(|&i| i < rows.len());
238
239    // On the right inside the box: ✕ to clear when there is text, and the
240    // regex toggle while the box has the focus or the pointer (or regex is
241    // on), so an idle box is just a search field.
242    let regex_w = 28.0;
243    let clear_w = 20.0;
244    let hovered = ui.rect_contains_pointer(rect);
245    let show_regex = plot.archiver.is_some() && (focused || hovered || s.regex);
246    let show_clear = !s.text.is_empty();
247    let right = rect.right()
248        - 6.0
249        - if show_regex { regex_w } else { 0.0 }
250        - if show_clear { clear_w } else { 0.0 };
251    let inner = Rect::from_min_max(
252        Pos2::new(icon.right() + 2.0, rect.top()),
253        Pos2::new(right.max(icon.right() + 40.0), rect.bottom()),
254    );
255    let mut edit_ui = ui.new_child(
256        egui::UiBuilder::new()
257            .max_rect(inner)
258            .layout(Layout::left_to_right(Align::Center)),
259    );
260    let hint = if plot.archiver.is_some() {
261        "Search PVs…"
262    } else {
263        "Add a PV…"
264    };
265    if !focused && s.text.is_empty() {
266        // The shortcut, as a key cap.
267        let cap = Rect::from_center_size(
268            Pos2::new(inner.right() - 10.0, rect.center().y),
269            Vec2::new(16.0, 16.0),
270        );
271        let v = ui.visuals();
272        ui.painter().rect(
273            cap,
274            4.0,
275            Color32::TRANSPARENT,
276            Stroke::new(1.0, v.weak_text_color().gamma_multiply(0.6)),
277            StrokeKind::Inside,
278        );
279        ui.painter().text(
280            cap.center(),
281            egui::Align2::CENTER_CENTER,
282            "/",
283            egui::FontId::proportional(11.0),
284            v.weak_text_color(),
285        );
286    }
287    let resp = edit_ui.add(
288        egui::TextEdit::singleline(&mut s.text)
289            .id(edit_id)
290            .frame(egui::Frame::NONE)
291            .hint_text(hint)
292            .desired_width(inner.width())
293            // Esc is ours: it closes the list before it leaves the box.
294            .event_filter(egui::EventFilter {
295                horizontal_arrows: true,
296                vertical_arrows: true,
297                escape: true,
298                ..Default::default()
299            }),
300    );
301    let refocus = std::mem::take(&mut s.refocus);
302    if std::mem::take(&mut s.focus) || refocus {
303        resp.request_focus();
304    }
305    // A click brings back a list closed with Esc.
306    if resp.gained_focus() || resp.clicked() {
307        s.open = true;
308    }
309    if resp.gained_focus() {
310        // Coming back to the box selects its text, so typing replaces it
311        // (→ or End to edit it instead); not when the list took the focus
312        // for a moment.
313        s.select_all = !refocus;
314    }
315    if resp.changed() {
316        s.edited_at = now;
317        s.selected = None;
318        s.open = true;
319        s.select_all = false;
320    }
321    if s.select_all {
322        // Until the click that focused the box is let go, unless it drags
323        // out a selection of its own.
324        if resp.dragged() {
325            s.select_all = false;
326        } else {
327            let all = CCursorRange::two(CCursor::new(0), CCursor::new(s.text.chars().count()));
328            set_cursor(ui.ctx(), edit_id, all);
329            s.select_all = ui.input(|i| i.pointer.any_down());
330        }
331    }
332
333    // The regex toggle, inside the box on the right.
334    let toggle = Rect::from_center_size(
335        Pos2::new(rect.right() - regex_w / 2.0 - 5.0, rect.center().y),
336        Vec2::new(regex_w, h - 8.0),
337    );
338    if show_clear {
339        let x = if show_regex {
340            toggle.left() - clear_w / 2.0
341        } else {
342            rect.right() - 5.0 - clear_w / 2.0
343        };
344        let button = Rect::from_center_size(Pos2::new(x, rect.center().y), Vec2::splat(18.0));
345        let r = ui
346            .interact(button, plot.id.with("clear"), Sense::click())
347            .on_hover_text("Clear");
348        let v = ui.visuals();
349        if r.hovered() {
350            ui.painter()
351                .circle_filled(button.center(), 9.0, v.widgets.hovered.weak_bg_fill);
352        }
353        let color = if r.hovered() {
354            v.text_color()
355        } else {
356            v.weak_text_color()
357        };
358        Icon::Cross.paint(ui.painter(), button, 11.0, color);
359        if r.clicked() {
360            clear(s);
361            s.refocus = true;
362            ui.ctx().request_repaint();
363        }
364    }
365    if show_regex {
366        let r = ui
367            .interact(toggle, plot.id.with("regex"), Sense::click())
368            .on_hover_text("Match a regular expression (else * and ? wildcards)");
369        if r.clicked() {
370            s.regex = !s.regex;
371            s.searched.clear();
372            s.edited_at = now;
373            s.refocus = true;
374            ui.ctx().request_repaint();
375        }
376        let v = ui.visuals();
377        let (fill, fg) = if s.regex {
378            (v.selection.bg_fill, v.selection.stroke.color)
379        } else if r.hovered() {
380            (v.widgets.hovered.weak_bg_fill, v.text_color())
381        } else {
382            (Color32::TRANSPARENT, v.weak_text_color())
383        };
384        ui.painter()
385            .rect_filled(toggle, toggle.height() / 2.0, fill);
386        ui.painter().text(
387            toggle.center(),
388            egui::Align2::CENTER_CENTER,
389            ".*",
390            egui::FontId::monospace(12.0),
391            fg,
392        );
393    }
394
395    // Search once typing pauses.
396    let text = s.text.trim().to_string();
397    if let Some(client) = &plot.archiver
398        && !text.is_empty()
399        && text != s.searched
400        && now - s.edited_at >= SEARCH_DELAY
401    {
402        s.searched = text.clone();
403        s.asked += 1;
404        s.waiting = true;
405        let search = if s.regex {
406            Search::regex(text.clone(), SEARCH_LIMIT)
407        } else {
408            Search::from_user(&text, SEARCH_LIMIT)
409        };
410        plot.fetcher
411            .search(&plot.pvs, client, search, s.asked, plot.repaint.clone());
412    } else if !text.is_empty() && text != s.searched && plot.archiver.is_some() {
413        ui.ctx()
414            .request_repaint_after(std::time::Duration::from_millis(60));
415    }
416
417    // Enter picks the highlighted row. The box keeps the focus, the text
418    // and the list, to add more from the same search.
419    let mut pick = None;
420    let enter = resp.lost_focus() && ui.input(|i| i.key_pressed(Key::Enter));
421    if enter {
422        pick = picked;
423        resp.request_focus();
424    }
425    // The list shows when open and it has something to show (the empty
426    // box with no recent searches has nothing).
427    let shown = s.open && (!rows.is_empty() || !text.is_empty());
428    if focused && ui.input(|i| i.key_pressed(Key::Escape)) {
429        if shown {
430            s.open = false;
431        } else {
432            clear(s);
433            ui.ctx().memory_mut(|m| m.surrender_focus(edit_id));
434        }
435    }
436
437    // The popup: recent searches, then the archiver's matches.
438    let mut forget = None;
439    if shown && s.open {
440        let pos = rect.left_bottom() + Vec2::new(0.0, 4.0);
441        let area = egui::Area::new(plot.id.with("search-popup"))
442            .order(egui::Order::Foreground)
443            .fixed_pos(pos)
444            .show(ui.ctx(), |ui| {
445                egui::Frame::popup(ui.style())
446                    .corner_radius(10.0)
447                    .show(ui, |ui| {
448                        ui.set_width(width - 14.0);
449                        ui.spacing_mut().item_spacing.y = 2.0;
450                        egui::ScrollArea::vertical()
451                            .max_height(320.0)
452                            .min_scrolled_height((rows.len() as f32 * 26.0).min(320.0))
453                            .show(ui, |ui| {
454                                for (i, row) in rows.iter().enumerate() {
455                                    let recent = matches!(row, Row::Recent(_));
456                                    if recent && i == 0 {
457                                        ui.label(egui::RichText::new("Recent").small().weak());
458                                    } else if !recent
459                                        && i > 0
460                                        && matches!(rows[i - 1], Row::Recent(_))
461                                    {
462                                        ui.separator();
463                                    }
464                                    let on = Some(i) == picked;
465                                    let id = plot.id.with(("search-row", i));
466                                    let (r, x) = popup_row(ui, row, on, row.plotted(&plotted), id);
467                                    if on && (enter || moved) {
468                                        r.scroll_to_me(None);
469                                    }
470                                    if r.hovered() && ui.input(|i| i.pointer.delta() != Vec2::ZERO)
471                                    {
472                                        s.selected = Some(i);
473                                    }
474                                    if r.clicked() {
475                                        pick = Some(i);
476                                        s.refocus = true;
477                                    }
478                                    if x {
479                                        forget = Some(i);
480                                        s.refocus = true;
481                                    }
482                                }
483                            });
484                        let note = if text.is_empty() {
485                            None
486                        } else if s.waiting {
487                            Some("Searching the archiver…".to_string())
488                        } else if let Some(e) = &s.error {
489                            Some(format!("Search failed: {e}"))
490                        } else if plot.archiver.is_some()
491                            && s.searched == text
492                            && s.results.is_empty()
493                        {
494                            Some("No archived PVs match".to_string())
495                        } else if s.results.len() as i32 >= SEARCH_LIMIT {
496                            Some(format!("First {SEARCH_LIMIT} matches: type more to narrow"))
497                        } else {
498                            None
499                        };
500                        if let Some(note) = note {
501                            ui.add_space(2.0);
502                            ui.horizontal(|ui| {
503                                if s.waiting {
504                                    ui.spinner();
505                                }
506                                ui.label(egui::RichText::new(note).small().weak());
507                            });
508                        }
509                    });
510            });
511        // A click anywhere else closes it, keeping the text.
512        let clicked_away = ui.input(|i| {
513            i.pointer.any_pressed()
514                && i.pointer
515                    .interact_pos()
516                    .is_some_and(|p| !area.response.rect.contains(p) && !rect.contains(p))
517        });
518        if clicked_away {
519            s.open = false;
520        }
521    }
522
523    if let Some(Row::Recent(r)) = forget.map(|i| &rows[i])
524        && plot.history.forget(r)
525    {
526        s.selected = None;
527        plot.history_changed = true;
528    }
529    match pick.map(|i| &rows[i]) {
530        // A recent search: back in the box, searched straight away.
531        Some(Row::Recent(r)) => {
532            s.text = r.text.clone();
533            s.regex = r.regex;
534            // Ask again, as the regex toggle does: the last answer may be
535            // for the same text in the other mode.
536            s.searched.clear();
537            s.results.clear();
538            s.error = None;
539            s.edited_at = now - SEARCH_DELAY;
540            s.selected = None;
541            s.open = true;
542            let end = CCursor::new(s.text.chars().count());
543            set_cursor(ui.ctx(), edit_id, CCursorRange::one(end));
544            ui.ctx().request_repaint();
545        }
546        // Already plotted: nothing to do.
547        Some(row) if !row.plotted(&plotted) => {
548            // A search that found something to add is worth remembering.
549            if plot.history.record(&text, s.regex) {
550                plot.history_changed = true;
551            }
552            plot.add_pvs(&row.names());
553        }
554        _ => {}
555    }
556}

558/// Empty the box, the list and the search.
559fn clear(s: &mut SearchState) {
560    s.text.clear();
561    s.searched.clear();
562    s.results.clear();
563    s.error = None;
564    s.waiting = false;
565    s.selected = None;
566    s.open = false;
567}

569/// Put the search box's cursor (or selection) at `range`, from the next
570/// frame.
571fn set_cursor(ctx: &egui::Context, id: egui::Id, range: CCursorRange) {
572    let mut state = egui::text_edit::TextEditState::load(ctx, id).unwrap_or_default();
573    state.cursor.set_char_range(Some(range));
574    state.store(ctx, id);
575}

577/// One row of the popup, drawn as a selectable button: a recent search
578/// has a clock, and a ✕ on hover (returning true when clicked) to forget
579/// it; a PV already plotted is dimmed and ticked.
580fn popup_row(
581    ui: &mut egui::Ui,
582    row: &Row,
583    picked: bool,
584    plotted: bool,
585    id: egui::Id,
586) -> (egui::Response, bool) {
587    let h = 24.0;
588    let (rect, r) = ui.allocate_exact_size(Vec2::new(ui.available_width(), h), Sense::click());
589    let hovered = ui.rect_contains_pointer(rect);
590    let v = ui.visuals().clone();
591    if picked {
592        ui.painter().rect_filled(rect, 6.0, v.selection.bg_fill);
593    } else if hovered {
594        ui.painter()
595            .rect_filled(rect, 6.0, v.widgets.hovered.weak_bg_fill);
596    }
597    let fg = if picked {
598        v.selection.stroke.color
599    } else {
600        v.text_color()
601    };
602    let weak = if picked {
603        fg.gamma_multiply(0.7)
604    } else {
605        v.weak_text_color()
606    };
607    let mut left = rect.left() + 8.0;
608    let mut right = rect.right() - 4.0;
609    let mut forget = false;
610    let slot = |x: f32| Rect::from_center_size(Pos2::new(x, rect.center().y), Vec2::splat(18.0));
611    match row {
612        Row::Recent(recent) => {
613            Icon::Clock.paint(ui.painter(), slot(left + 7.0), 13.0, weak);
614            left += 22.0;
615            let button = slot(right - 9.0);
616            right -= 22.0;
617            let x = ui
618                .interact(button, id.with("forget"), Sense::click())
619                .on_hover_text("Forget this search");
620            if hovered || x.hovered() {
621                if x.hovered() {
622                    let fill = if picked {
623                        fg.gamma_multiply(0.25)
624                    } else {
625                        v.widgets.active.weak_bg_fill
626                    };
627                    ui.painter().circle_filled(button.center(), 9.0, fill);
628                }
629                let color = if x.hovered() { fg } else { weak };
630                Icon::Cross.paint(ui.painter(), button, 11.0, color);
631            }
632            forget = x.clicked();
633            if recent.regex {
634                ui.painter().text(
635                    Pos2::new(right, rect.center().y),
636                    egui::Align2::RIGHT_CENTER,
637                    ".*",
638                    egui::FontId::monospace(12.0),
639                    weak,
640                );

667/// A row of the search popup.
668enum Row {
669    /// A recent search: picking it puts it back in the box.
670    Recent(RecentSearch),
671    /// What was typed, added as it is (one or more names).
672    Typed(Vec<String>),
673    Found(String),
674}
675
676impl Row {
677    fn names(&self) -> Vec<String> {
678        match self {
679            Row::Recent(_) => Vec::new(),
680            Row::Typed(n) => n.clone(),
681            Row::Found(n) => vec![n.clone()],
682        }
683    }
684
685    fn text(&self) -> String {
686        match self {
687            Row::Recent(r) => r.text.clone(),
688            Row::Typed(n) => format!("+  Add “{}” as typed", n.join("”, “")),
689            Row::Found(n) => n.clone(),
690        }
691    }
692
693    /// Whether all its PVs are on the plot already (`plotted` holds their
694    /// `PvName::key`s).
695    fn plotted(&self, plotted: &[String]) -> bool {
696        let names = self.names();
697        !names.is_empty()
698            && names
699                .iter()
700                .all(|n| plotted.contains(&pv::PvName::parse(n.trim()).key()))
701    }
702}
703
704/// The recent searches shown above the matches while typing.
705const RECENT_SHOWN: usize = 5;
706
707/// The popup's rows. With the box empty, the recent searches. Else up to
708/// [`RECENT_SHOWN`] recent searches holding the text, then a match that
709/// is exactly the text typed, then the other archiver matches, then the
710/// text as typed (unless it is a pattern, or exactly a match). Text that
711/// can only be added as typed (a `pva://` or `sim://` name, or several
712/// names) comes before the matches instead.
713fn rows(s: &SearchState, archiver: bool, history: &SearchHistory) -> Vec<Row> {
714    let text = s.text.trim();
715    if text.is_empty() {
716        return history
717            .searches()
718            .iter()
719            .cloned()
720            .map(Row::Recent)
721            .collect();
722    }
723    let mut found: Vec<&String> = if s.searched == text {
724        s.results.iter().collect()
725    } else {
726        Vec::new()
727    };
728    if let Some(i) = found.iter().position(|r| *r == text) {
729        let exact = found.remove(i);
730        found.insert(0, exact);
731    }
732    let typed: Vec<String> = text
733        .split([' ', ',', '\t'])
734        .filter(|n| !n.is_empty())
735        .map(str::to_string)
736        .collect();
737    let pattern = archiver && (s.regex || text.contains(['*', '?']));
738    let mut rows: Vec<Row> = found.into_iter().cloned().map(Row::Found).collect();
739    if !typed.is_empty() && !pattern && !s.results.iter().any(|r| r == text) {
740        let as_typed = typed.len() > 1 || text.contains("://");
741        let row = Row::Typed(typed);
742        if as_typed {
743            rows.insert(0, row);
744        } else {
745            rows.push(row);
746        }
747    }
748    let recent = history.matching(text).into_iter().take(RECENT_SHOWN);
749    recent.cloned().map(Row::Recent).chain(rows).collect()
750}
751
752/// The row highlighted until the arrows move it: the first that adds
753/// something, so Enter adds the best match rather than refilling the box
754/// with a recent search. None with only recent searches.
755fn default_row(rows: &[Row]) -> Option<usize> {
756    rows.iter().position(|r| !matches!(r, Row::Recent(_)))
757}

762    let rows = rows(&plot.search, plot.archiver.is_some(), &plot.history);

772/// Take in a finished search, and what it says of the archiver: true
773/// when the archiver is back after failing (see `status::Monitor::ok`).
774pub fn take_found(plot: &mut Plot, now: f64) -> bool {
775    let Some((n, found)) = plot.fetcher.take_found() else {
776        return false;
777    };
778    let s = &mut plot.search;
779    if n != s.asked {
780        return false;
781    }
782    s.waiting = false;
783    s.selected = None;
784    match found {
785        Ok(names) => {
786            s.results = names;
787            s.error = None;
788            plot.monitor.ok(now)
789        }
790        Err(e) => {
791            s.results.clear();
792            s.error = Some(e.to_string());
793            if super::status::archiver_fault(&e, true) {
794                plot.monitor.suspect(&e, now);
795            }
796            false
797        }
798    }
799}

1193/// `text` cut short with "…" to fit `max` points.
1194fn elide(painter: &egui::Painter, text: &str, font: &egui::FontId, max: f32) -> String {
1195    let chars: Vec<char> = text.chars().collect();
1196    let mut shown = text.to_string();
1197    let mut keep = chars.len();
1198    while super::render::text_size(painter, &shown, font).x > max && keep > 4 {
1199        keep -= 1;
1200        shown = format!("{}…", chars[..keep].iter().collect::<String>());
1201    }
1202    shown
1203}

1240                        let go = ui.button("Use").clicked()
1241                            || (r.lost_focus() && ui.input(|i| i.key_pressed(Key::Enter)));
1242                        if go {
1243                            match ArchiverClient::new(&e.url) {
1244                                Ok(a) => pick = Some(Some(a)),
1245                                Err(err) => e.error = Some(err.to_string()),
1246                            }
1247                        }
1248                    });
1249                    if let Some(err) = &e.error {
1250                        ui.colored_label(ui.visuals().error_fg_color, err);
1251                    }
1252                    if !plot.recent.is_empty() {
1253                        ui.separator();
1254                        ui.label(egui::RichText::new("Recent").small().weak());
1255                        ui.spacing_mut().item_spacing.y = 2.0;
1256                        for url in &plot.recent {
1257                            let on = current.as_deref() == Some(url.as_str());
1258                            if ui.selectable_label(on, url).clicked() {
1259                                match ArchiverClient::new(url) {
1260                                    Ok(a) => pick = Some(Some(a)),
1261                                    Err(err) => e.error = Some(err.to_string()),
1262                                }
1263                            }
1264                        }
1265                        ui.spacing_mut().item_spacing.y = 8.0;
1266                    }
1267                    if current.is_some() {
1268                        ui.separator();
1269                        if ui
1270                            .button("No archiver (live only)")
1271                            .on_hover_text("Show live data only")
1272                            .clicked()
1273                        {
1274                            pick = Some(None);
1275                        }
1276                    }
1277                });
1278        });
1279    let away = ui.input(|i| {
1280        i.pointer.any_pressed()
1281            && i.pointer
1282                .interact_pos()
1283                .is_some_and(|p| !area.response.rect.contains(p) && !anchor.contains(p))
1284    });
1285    if away || ui.input(|i| i.key_pressed(Key::Escape)) {
1286        plot.edit.open = false;
1287    }
1288    if let Some(archiver) = pick {
1289        plot.edit.open = false;
1290        if archiver.as_ref().map(|a| a.base()) != current.as_deref() {
1291            plot.set_archiver(archiver);
1292            plot.archiver_changed = true;
1293        } else if archiver.is_some() {
1294            // The same one again: check it now (it may be back).
1295            plot.monitor.due = true;
1296        }
1297    }
1298}

1302/// `● Live`: the dot shows while following now, pulsing for a few seconds
1303/// after it starts and while the pointer is over the plot; click to pause
1304/// or to jump to now.
1305fn live_pill(plot: &mut Plot, ui: &mut egui::Ui, now: f64) {
1306    let h = icons::height(ui);
1307    let font = egui::TextStyle::Button.resolve(ui.style());
1308    let text_w = super::render::text_size(ui.painter(), "Live", &font).x;
1309    let size = Vec2::new(text_w + h * 0.9 + 18.0, h);
1310    let (rect, r) = ui.allocate_exact_size(size, Sense::click());
1311    let live = plot.window.live;
1312    let r = r.on_hover_text(if live {
1313        "Following now: click to pause (L)"
1314    } else {
1315        "Jump to now and follow it (L)"
1316    });
1317    let v = ui.visuals();
1318    let dark = v.dark_mode;
1319    let accent = crate::style::live(dark);
1320    let fill = if live {
1321        accent.gamma_multiply(if dark { 0.22 } else { 0.10 })
1322    } else if r.hovered() {
1323        v.widgets.hovered.weak_bg_fill
1324    } else {
1325        v.extreme_bg_color
1326    };
1327    let stroke = if live {
1328        Stroke::new(1.0, accent.gamma_multiply(0.5))
1329    } else {
1330        v.widgets.noninteractive.bg_stroke
1331    };
1332    ui.painter()
1333        .rect(rect, h / 2.0, fill, stroke, StrokeKind::Inside);
1334    let dot = Pos2::new(rect.left() + 12.0 + 4.0, rect.center().y);
1335    if live && plot.pulse {
1336        // A slow breath: a halo growing and fading every two seconds.
1337        let phase = (ui.input(|i| i.time) % 2.0 / 2.0) as f32;
1338        ui.painter().circle_filled(
1339            dot,
1340            4.0 + 5.0 * phase,
1341            accent.gamma_multiply(0.45 * (1.0 - phase)),
1342        );
1343        ui.painter().circle_filled(dot, 4.0, accent);
1344    } else if live {
1345        // Still, once it has said so: no animation in an idle window.
1346        ui.painter()
1347            .circle_filled(dot, 6.5, accent.gamma_multiply(0.18));
1348        ui.painter().circle_filled(dot, 4.0, accent);
1349    } else {
1350        ui.painter()
1351            .circle_stroke(dot, 3.5, Stroke::new(1.5, v.weak_text_color()));
1352    }
1353    let fg = if live {
1354        v.strong_text_color()
1355    } else {
1356        v.text_color()
1357    };
1358    ui.painter().text(
1359        Pos2::new(dot.x + 10.0, rect.center().y),
1360        egui::Align2::LEFT_CENTER,
1361        "Live",
1362        font,
1363        fg,
1364    );
1365    if r.clicked() {
1366        plot.toggle_live(now);
1367    }
1368}
