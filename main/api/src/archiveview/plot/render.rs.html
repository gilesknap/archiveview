15/// Where things go in a plot rect, for painting and for hit testing.
16pub struct Frame {
17    /// The plot rect.
18    pub outer: Rect,
19    /// Where data is drawn.
20    pub data: Rect,
21    pub xs: Scale,
22    pub ys: Scale,
23    y_labels: Vec<(f64, String)>,
24    title: String,
25    font: FontId,
26}
27
28impl Frame {
29    /// The value-axis strip left of the data, for dragging values.
30    pub fn value_axis(&self) -> Rect {
31        Rect::from_min_max(
32            Pos2::new(self.outer.left(), self.data.top()),
33            Pos2::new(self.data.left(), self.data.bottom()),
34        )
35    }
36
37    /// The time at screen `x`.
38    pub fn time_at(&self, x: f32) -> f64 {
39        let f = ((x - self.xs.from) / (self.xs.to - self.xs.from)) as f64;
40        self.xs.lo + f * (self.xs.hi - self.xs.lo)
41    }
42
43    /// The value at screen `y`.
44    pub fn value_at(&self, y: f32) -> f64 {
45        let f = ((y - self.ys.from) / (self.ys.to - self.ys.from)) as f64;
46        if self.ys.log && self.ys.lo > 0.0 {
47            let (l, h) = (self.ys.lo.log10(), self.ys.hi.log10());
48            10f64.powf(l + f * (h - l))
49        } else {
50            self.ys.lo + f * (self.ys.hi - self.ys.lo)
51        }
52    }
53}

55const PAD: f32 = 8.0;
56const TICK_GAP: f32 = 6.0;
57
58/// Lay out the plot in `rect` for times `from..to` and values `lo..hi`,
59/// with `title` (units) up the value axis.
60pub fn frame(
61    painter: &egui::Painter,
62    rect: Rect,
63    (from, to): (f64, f64),
64    (lo, hi): (f64, f64),
65    log: bool,
66    title: &str,
67    font: &FontId,
68) -> Frame {
69    let label_h = text_size(painter, "0", font).y;
70    let bottom = rect.bottom() - PAD - label_h - TICK_GAP;
71    let top = rect.top() + PAD + label_h / 2.0;
72    let ys = Scale {
73        lo,
74        hi,
75        log,
76        from: bottom,
77        to: top,
78    };
79    let (y_ticks, y_step) = ys.ticks(label_h * 3.0);
80    let y_labels: Vec<(f64, String)> = y_ticks
81        .iter()
82        .map(|v| (*v, axis::label(*v, y_step)))
83        .collect();
84    let label_w = y_labels
85        .iter()
86        .map(|(_, l)| text_size(painter, l, font).x)
87        .fold(text_size(painter, "0000", font).x, f32::max);
88    let title_w = if title.is_empty() {
89        0.0
90    } else {
91        label_h + PAD / 2.0
92    };
93    let left = rect.left() + PAD + title_w + label_w + TICK_GAP;
94    let right = rect.right() - PAD * 2.0;
95    let data = Rect::from_min_max(
96        Pos2::new(left, top),
97        Pos2::new(right.max(left + 1.0), bottom),
98    );
99    let xs = Scale {
100        lo: from,
101        hi: to,
102        log: false,
103        from: data.left(),
104        to: data.right(),
105    };
106    Frame {
107        outer: rect,
108        data,
109        xs,
110        ys,
111        y_labels,
112        title: title.to_string(),
113        font: font.clone(),
114    }
115}

124/// Paint the background, grid, labels and `lines`.
125pub fn paint(painter: &egui::Painter, f: &Frame, lines: &[Line], visuals: &egui::Visuals) {
126    let painter = &painter.with_clip_rect(painter.clip_rect().intersect(f.outer));
127    painter.rect_filled(f.outer, 10.0, visuals.extreme_bg_color);
128    let text = visuals.text_color();
129    let weak = visuals.weak_text_color();
130    let grid = Stroke::new(1.0, text.gamma_multiply(0.07));
131    let data = f.data;
132    if data.width() < 4.0 || data.height() < 4.0 {
133        return;
134    }
135
136    // Time ticks, labels measured so none overlap.
137    let sample_w = text_size(painter, "00:00", &f.font).x;
138    let max_ticks = (data.width() / (sample_w + 48.0)).floor().max(2.0) as usize;
139    let (t_ticks, t_step) = time_ticks(f.xs.lo, f.xs.hi, max_ticks, &chrono::Local);
140    for t in &t_ticks {
141        let x = f.xs.pos(*t).round() + 0.5;
142        painter.line_segment(
143            [Pos2::new(x, data.top()), Pos2::new(x, data.bottom())],
144            grid,
145        );
146    }
147    for (v, _) in &f.y_labels {
148        let y = f.ys.pos(*v).round() + 0.5;
149        painter.line_segment(
150            [Pos2::new(data.left(), y), Pos2::new(data.right(), y)],
151            grid,
152        );
153    }
154    // A baseline under the data.
155    painter.line_segment(
156        [data.left_bottom(), data.right_bottom()],
157        Stroke::new(1.0, text.gamma_multiply(0.18)),
158    );
159
160    let clipped = painter.with_clip_rect(painter.clip_rect().intersect(data.expand(1.0)));
161    for line in lines {
162        draw_line(&clipped, f, line);
163    }
164
165    // Time labels below, the first with its date.
166    let mut right_of = f32::NEG_INFINITY;
167    for (i, t) in t_ticks.iter().enumerate() {
168        let x = f.xs.pos(*t);
169        let label = time_label(*t, t_step, i == 0);
170        let galley = painter.layout_no_wrap(label, f.font.clone(), weak);
171        let w = galley.size().x;
172        let left = (x - w / 2.0).clamp(f.outer.left() + 2.0, f.outer.right() - w - 2.0);
173        if left < right_of + 10.0 {
174            continue;
175        }
176        right_of = left + w;
177        painter.galley(Pos2::new(left, data.bottom() + TICK_GAP), galley, weak);
178    }
179    for (v, label) in &f.y_labels {
180        let y = f.ys.pos(*v);
181        painter.text(
182            Pos2::new(data.left() - TICK_GAP, y),
183            Align2::RIGHT_CENTER,
184            label,
185            f.font.clone(),
186            weak,
187        );
188    }
189    if !f.title.is_empty() {
190        let h = text_size(painter, &f.title, &f.font).y;
191        vertical_text(
192            painter,
193            Pos2::new(f.outer.left() + PAD + h / 2.0, data.center().y),
194            &f.title,
195            f.font.clone(),
196            weak,
197        );
198    }
199}

201/// A thin dashed line at `now` with a small tag, when it is in view. In
202/// neutral grey: red is a trace colour (and an alarm), and the Live pill's
203/// dot alone says the plot follows now.
204pub fn now_marker(painter: &egui::Painter, f: &Frame, now: f64, v: &egui::Visuals, font: &FontId) {
205    if now < f.xs.lo || now > f.xs.hi + (f.xs.hi - f.xs.lo) * 1e-3 {
206        return;
207    }
208    let color = v.weak_text_color();
209    let x = f.xs.pos(now).min(f.data.right());
210    let stroke = Stroke::new(1.0, color.gamma_multiply(0.7));
211    painter.extend(Shape::dashed_line(
212        &[Pos2::new(x, f.data.top()), Pos2::new(x, f.data.bottom())],
213        stroke,
214        4.0,
215        4.0,
216    ));
217    let galley = painter.layout_no_wrap("now".into(), font.clone(), color);
218    let size = galley.size() + Vec2::new(10.0, 2.0);
219    let tag = Rect::from_min_size(
220        Pos2::new(
221            (x - size.x / 2.0).min(f.outer.right() - size.x - 2.0),
222            f.outer.top() + 2.0,
223        ),
224        size,
225    );
226    painter.rect(
227        tag,
228        size.y / 2.0,
229        v.extreme_bg_color,
230        Stroke::new(1.0, color.gamma_multiply(0.5)),
231        egui::StrokeKind::Inside,
232    );
233    painter.galley(tag.min + Vec2::new(5.0, 1.0), galley, color);
234}

236fn draw_line(painter: &egui::Painter, f: &Frame, line: &Line) {
237    let ok = |v: f64| v.is_finite() && (!f.ys.log || v > 0.0);
238    // Far-off points are pulled in so the maths stays finite.
239    let bounds = f.data.expand(f.data.width().max(f.data.height()) * 4.0);
240    let x = |t: f64| f.xs.pos(t).clamp(bounds.left(), bounds.right());
241    let y = |v: f64| f.ys.pos(v).clamp(bounds.top(), bounds.bottom());
242
243    // Runs between gaps.
244    for run in line.pts.split(|p| !ok(p.v)).filter(|r| !r.is_empty()) {
245        // The min..max envelope of bins, as quads between neighbours.
246        if run.iter().any(Pt::is_bin) {
247            let fill = line.color.gamma_multiply(0.22);
248            let mut mesh = Mesh::default();
249            // One (x, lo, hi) per pixel column: quads narrower than a pixel
250            // would shimmer into moiré.
251            let cols: Vec<(f32, f32, f32)> = run
252                .iter()
253                .map(|p| {
254                    let (lo, hi) = if ok(p.lo) && ok(p.hi) {
255                        (p.lo, p.hi)
256                    } else {
257                        (p.v, p.v)
258                    };
259                    (x(p.t), y(lo), y(hi))
260                })
261                .collect::<Vec<_>>()
262                .chunk_by(|a, b| a.0.floor() == b.0.floor())
263                .map(|c| {
264                    let x = c.iter().map(|p| p.0).sum::<f32>() / c.len() as f32;
265                    let lo = c.iter().map(|p| p.1).fold(f32::MIN, f32::max);
266                    let hi = c.iter().map(|p| p.2).fold(f32::MAX, f32::min);
267                    (x, lo, hi)
268                })
269                .collect();
270            for w in cols.windows(2) {
271                let ((x0, lo0, hi0), cur) = (w[0], w[1]);
272                if cur.0 > x0 {
273                    let i = mesh.vertices.len() as u32;
274                    mesh.colored_vertex(Pos2::new(x0, lo0), fill);
275                    mesh.colored_vertex(Pos2::new(x0, hi0), fill);
276                    mesh.colored_vertex(Pos2::new(cur.0, cur.2), fill);
277                    mesh.colored_vertex(Pos2::new(cur.0, cur.1), fill);
278                    mesh.add_triangle(i, i + 1, i + 2);
279                    mesh.add_triangle(i, i + 2, i + 3);
280                }
281            }
282            if !mesh.is_empty() {
283                painter.add(Shape::mesh(mesh));
284            }
285        }
286        // The value line: steps for raw samples, slopes between bin means.
287        // Each point is marked whether it is a bin mean, which may be
288        // averaged with its pixel column's others; raw samples and step
289        // corners never are, so live values after binned history keep
290        // their steps.
291        let mut pts: Vec<(Pos2, bool)> = Vec::with_capacity(run.len() * 2);
292        let mut step = false;
293        for p in run {
294            let pos = Pos2::new(x(p.t), y(p.v));
295            if step && let Some((last, _)) = pts.last().copied() {
296                pts.push((Pos2::new(pos.x, last.y), false));
297            }
298            pts.push((pos, p.is_bin()));
299            step = p.step;
300        }
301        // Bin means packed tighter than a pixel apart zigzag into a solid
302        // block (a signal faster than the bins aliases); one mean per pixel
303        // column draws the trend, and the band above shows the extremes.
304        let pts = if run.iter().any(Pt::is_bin) {
305            column_means(&pts)
306        } else {
307            decimate(pts.into_iter().map(|(p, _)| p).collect())
308        };
309        stroke_line(painter, pts, Stroke::new(line.width, line.color));
310    }
311}

313/// Average the bin means (`true`) in each pixel column, when there is more
314/// than one; other points are kept as they are.
315fn column_means(run: &[(Pos2, bool)]) -> Vec<Pos2> {
316    run.chunk_by(|a, b| a.1 && b.1 && a.0.x.floor() == b.0.x.floor())
317        .map(|c| {
318            let n = c.len() as f32;
319            let sum = c.iter().fold(egui::Vec2::ZERO, |s, p| s + p.0.to_vec2());
320            (sum / n).to_pos2()
321        })
322        .collect()
323}

325/// The range that shows `data`: whole decades on a log scale, else rounded
326/// out to whole ticks. `None` when there is nothing to show (bobview's
327/// `plot::autoscale`).
328pub fn autoscale(log: bool, data: impl Iterator<Item = f64>) -> Option<(f64, f64)> {
329    let ok = |v: &f64| v.is_finite() && (!log || *v > 0.0);
330    let (lo, hi) = data
331        .filter(ok)
332        .fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), v| {
333            (a.min(v), b.max(v))
334        });
335    if lo > hi {
336        return None;
337    }
338    Some(if log {
339        let (a, b) = (
340            lo.log10().floor(),
341            hi.log10().ceil().max(lo.log10().floor() + 1.0),
342        );
343        (10f64.powf(a), 10f64.powf(b))
344    } else {
345        // A little air above and below (none below a floor of exactly zero,
346        // or above a ceiling of zero), then out to half ticks: rounding to
347        // whole ticks can waste an eighth of the height.
348        let pad = (hi - lo) * 0.04;
349        let a = if lo >= 0.0 {
350            (lo - pad).max(0.0)
351        } else {
352            lo - pad
353        };
354        let b = if hi <= 0.0 {
355            (hi + pad).min(0.0)
356        } else {
357            hi + pad
358        };
359        if b <= a {
360            return Some(axis::nice_range(a, b, 8));
361        }
362        let half = axis::nice_step(b - a, 8) / 2.0;
363        ((a / half).floor() * half, (b / half).ceil() * half)
364    })
365}

367pub fn text_size(painter: &egui::Painter, text: &str, font: &FontId) -> Vec2 {
368    painter
369        .layout_no_wrap(text.to_string(), font.clone(), Color32::BLACK)
370        .size()
371}

373/// Text turned to read upwards, centred on `center`.
374fn vertical_text(painter: &egui::Painter, center: Pos2, text: &str, font: FontId, c: Color32) {
375    let galley = painter.layout_no_wrap(text.to_string(), font, c);
376    let size = galley.size();
377    let pos = Pos2::new(center.x - size.y / 2.0, center.y + size.x / 2.0);
378    painter.add(epaint::TextShape::new(pos, galley, c).with_angle(-FRAC_PI_2));
379}

381/// Thin a line with many points per pixel column to the first, lowest,
382/// highest and last point of each column, which draws the same.
383fn decimate(run: Vec<Pos2>) -> Vec<Pos2> {
384    if run.len() < 4096 {
385        return run;
386    }
387    let mut out = Vec::new();
388    for column in run.chunk_by(|a, b| a.x.floor() == b.x.floor()) {
389        let lowest = column.iter().max_by(|a, b| a.y.total_cmp(&b.y));
390        let highest = column.iter().min_by(|a, b| a.y.total_cmp(&b.y));
391        let (first, last) = (column.first(), column.last());
392        let mut keep: Vec<Pos2> = [first, lowest, highest, last]
393            .into_iter()
394            .flatten()
395            .copied()
396            .collect();
397        keep.dedup();
398        out.extend(keep);
399    }
400    out
401}

403fn stroke_line(painter: &egui::Painter, mut pts: Vec<Pos2>, stroke: Stroke) {
404    // egui's tessellator turns a zero-length segment into NaN vertices,
405    // which can paint one triangle over the whole plot.
406    pts.dedup_by(|b, a| a.distance_sq(*b) < 1e-6);
407    if pts.len() < 2 {
408        if let Some(p) = pts.first() {
409            painter.circle_filled(*p, stroke.width, stroke.color);
410        }
411        return;
412    }
413    painter.add(Shape::line(pts, stroke));
414}

420    #[test]
421    fn autoscale_pads_and_rounds() {
422        let (lo, hi) = autoscale(false, [1.0, 9.0, f64::NAN].into_iter()).unwrap();
423        assert!(lo <= 0.68 && hi >= 9.32, "{lo}..{hi}");
424        assert_eq!(
425            autoscale(true, [3.0, 450.0, -1.0].into_iter()),
426            Some((1.0, 1000.0))
427        );
428        assert_eq!(autoscale(false, [f64::NAN].into_iter()), None);
429        let (lo, hi) = autoscale(false, [5.0, 5.0].into_iter()).unwrap();
430        assert!(lo < 5.0 && hi > 5.0);
431    }
432
433    #[test]
434    fn autoscale_wastes_little() {
435        // From zero: no negative tick under it.
436        let (lo, hi) = autoscale(false, [0.0, 300.0].into_iter()).unwrap();
437        assert_eq!(lo, 0.0);
438        assert!((300.0..=325.0).contains(&hi), "{hi}");
439        let (lo, hi) = autoscale(false, [-300.0, 0.0].into_iter()).unwrap();
440        assert!(lo <= -300.0 && hi == 0.0);
441    }
442
443    #[test]
444    fn column_means_keep_raw_steps() {
445        let p = |x: f32, y: f32| Pos2::new(x, y);
446        // Two bins in column 0 average; the step corner and the live
447        // sample in column 5 stay apart.
448        let run = [
449            (p(0.2, 10.0), true),
450            (p(0.7, 20.0), true),
451            (p(5.1, 20.0), false),
452            (p(5.3, 30.0), false),
453        ];
454        let out = column_means(&run);
455        assert_eq!(out, [p(0.45, 15.0), p(5.1, 20.0), p(5.3, 30.0)]);
456    }
457
458    #[test]
459    fn decimation_keeps_column_extremes() {
460        let pts: Vec<Pos2> = (0..10_000)
461            .map(|i| Pos2::new(i as f32 / 100.0, (i % 7) as f32))
462            .collect();
463        let d = decimate(pts);
464        assert!(d.len() <= 4 * 100);
465        assert!(d.iter().any(|p| p.y == 6.0) && d.iter().any(|p| p.y == 0.0));
466    }
