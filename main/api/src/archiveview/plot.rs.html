1//! [`Plot`]: the embeddable archived + live time-series widget.
2//!
3//! See DESIGN.md ("Embedding", "Data flow" and "UI") for the interaction
4//! model, fetch policy and live merge. The pieces:
5//!
6//! - `series`: a trace's live samples, archived history, and their merge;
7//! - `fetch`: background archive requests on the pv crate's runtime;
8//! - `window`: time window and value range maths (pan, zoom, presets);
9//! - `timeaxis`: ticks and adaptive labels in local time;
10//! - `render`: painting the plot area;
11//! - `bar`: the slim top bar (search, range pill, live pill, buttons);
12//! - `legend`: the chips under the plot;
13//! - `hover`: the crosshair readout, empty state and status line.

38/// Wait this long after the last pan or zoom before fetching (seconds).
39const DEBOUNCE: f64 = 0.3;
40
41/// While following now, refresh the archived data this often (seconds).
42const LIVE_REFRESH: f64 = 30.0;

44/// After an error, ask again no sooner than this (seconds), doubling with
45/// each failure in a row.
46const RETRY: f64 = 10.0;

59/// An archived + live plot of a set of PVs on one shared time axis.
60///
61/// Owns its traces, background fetches and live subscriptions; dropping it
62/// unsubscribes and abandons in-flight fetches. Its only shared state is
63/// the counter [`Plot::new`] salts ids with, so any number may exist at
64/// once.
65pub struct Plot {
66    pvs: PvManager,
67    archiver: Option<ArchiverClient>,
68    /// Salt for every egui id this plot uses.
69    id: egui::Id,
70    traces: Vec<Trace>,
71    next_trace: u64,
72    window: Window,
73    /// The span a reset (double-click, Home) returns to.
74    home_span: f64,
75    /// The range drawn, easing toward `window`'s.
76    shown: Option<(f64, f64)>,
77    y: YState,
78    fetcher: Fetcher,
79    /// When (seconds since 1970) the user last panned or zoomed, for the
80    /// fetch debounce.
81    changed_at: f64,
82    /// A fetch is waiting for the debounce.
83    pending: bool,
84    /// Where finished requests ask for a repaint: set every frame.
85    repaint: Option<fetch::Repaint>,
86    /// When (egui time) the plot last started following now, for the
87    /// Live pill's pulse.
88    live_since: Option<f64>,
89    /// The Live pill's dot pulses this frame.
90    pulse: bool,
91    /// egui's time at the last frame, for animation steps.
92    last_frame: Option<f64>,
93    /// Draw the next frame without easing (direct manipulation).
94    snap: bool,
95    /// Kinetic panning after a drag, in seconds of time per second.
96    glide: f64,
97    drag: Option<Drag>,
98    /// The trace whose chip is hovered: drawn bold, the others dimmed.
99    highlight: Option<u64>,
100    /// Keyboard shortcuts apply: the plot was clicked last.
101    active: bool,
102    search: bar::SearchState,
103    custom: bar::CustomState,
104}

141    /// A plot subscribing through `pvs` (the caller's manager: a `tagged`
142    /// handle from bobview repaints only the window holding this plot) and
143    /// fetching history from `archiver`, or live-only with `None`. Archive
144    /// fetches run as blocking tasks on `pvs.runtime()`.
145    pub fn new(pvs: &PvManager, archiver: Option<ArchiverClient>) -> Self {
146        let n = NEXT_PLOT.fetch_add(1, Ordering::Relaxed);
147        Self::with_id(pvs, archiver, egui::Id::new(("archiveview-plot", n)))
148    }

191    /// Change (or remove) the archiver; refetches every trace.
192    pub fn set_archiver(&mut self, archiver: Option<ArchiverClient>) {
193        self.archiver = archiver;
194        for t in &mut self.traces {
195            t.history = None;
196            // The generation carries on, so a late answer from the old
197            // archiver is still recognised as stale.
198            t.fetch = series::FetchState {
199                generation: t.fetch.generation + 1,
200                ..Default::default()
201            };
202        }
203    }

217    fn add(&mut self, name: &str) -> Option<&mut Trace> {
218        let name = name.trim();
219        let key = pv::PvName::parse(name).key();
220        if name.is_empty() || self.traces.iter().any(|t| t.pv.name().key() == key) {
221            return None;
222        }
223        // The first palette colour no other trace has.
224        let slot = (0..)
225            .find(|s| !self.traces.iter().any(|t| t.slot == *s))
226            .unwrap_or(0);
227        let mut trace = Trace::new(self.next_trace, name, self.pvs.subscribe(name));
228        trace.slot = slot;
229        self.next_trace += 1;
230        self.traces.push(trace);
231        self.traces.last_mut()
232    }

234    /// Remove a trace by the name it was added with.
235    pub fn remove_pv(&mut self, name: &str) {
236        let key = pv::PvName::parse(name).key();
237        self.traces
238            .retain(|t| t.name != name && t.pv.name().key() != key);
239    }

246    /// A window title: the PV names, shortened (as bobview's
247    /// `Trend::title`); empty with none.
248    pub fn title(&self) -> String {
249        let names: Vec<String> = self.traces.iter().map(Trace::label).collect();
250        match names.as_slice() {
251            [] => String::new(),
252            [one] => one.clone(),
253            [a, b] => format!("{a}, {b}"),
254            [first, rest @ ..] => format!("{first} and {} more", rest.len()),
255        }
256    }

258    /// The current view (traces, range, y axis), to save.
259    pub fn view(&self) -> View {
260        let range = if self.window.live {
261            view::Range::Relative {
262                secs: self.window.span,
263            }
264        } else {
265            let (from, to) = self.window.range(now());
266            view::Range::Absolute {
267                start: view::format_time(from),
268                end: view::format_time(to),
269            }
270        };
271        View {
272            version: view::VERSION,
273            pvs: self
274                .traces
275                .iter()
276                .map(|t| TraceView {
277                    name: t.name.clone(),
278                    color: t.color.clone(),
279                    visible: t.visible,
280                })
281                .collect(),
282            range,
283            y: YAxis {
284                autoscale: self.y.autoscale,
285                min: (!self.y.autoscale).then_some(self.y.lo),
286                max: (!self.y.autoscale).then_some(self.y.hi),
287                log: self.y.log,
288            },
289        }
290    }

292    /// Show `view`, replacing the traces and range.
293    pub fn set_view(&mut self, view: &View) {
294        self.traces.clear();
295        for tv in &view.pvs {
296            if let Some(t) = self.add(&tv.name) {
297                t.color = tv.color.clone();
298                t.visible = tv.visible;
299            }
300        }
301        self.set_range(&view.range);
302        self.y = YState {
303            autoscale: view.y.autoscale || view.y.min.is_none() || view.y.max.is_none(),
304            lo: view.y.min.unwrap_or(0.0),
305            hi: view.y.max.unwrap_or(1.0),
306            log: view.y.log,
307            shown: None,
308        };
309        self.snap = true;
310    }

312    /// Show `range`.
313    pub fn set_range(&mut self, range: &view::Range) {
314        self.window = match range {
315            view::Range::Relative { secs } => Window::live(*secs),
316            view::Range::Absolute { start, end } => {
317                match (view::parse_time(start), view::parse_time(end)) {
318                    (Some(a), Some(b)) => Window::fixed(a, b),
319                    _ => {
320                        log::warn!("bad view range {start}..{end}: showing the last hour");
321                        Window::live(3600.0)
322                    }
323                }
324            }
325        };
326        self.home_span = self.window.span;
327        self.snap = true;
328    }

330    /// True while any archive fetch or search is in flight, or one is due
331    /// (snapshots wait for this).
332    pub fn is_loading(&self) -> bool {
333        self.fetcher.pending()
334            || self.pending
335            || bar::search_due(self)
336            || (self.archiver.is_some()
337                && self.traces.iter().any(|t| {
338                    t.archive_name().is_some()
339                        && !t.fetch.not_archived
340                        && t.fetch.requested.is_none()

356    fn show(&mut self, ui: &mut egui::Ui, menu: Option<&mut dyn FnMut(&mut egui::Ui)>) {
357        let ctx = ui.ctx().clone();
358        self.repaint = Some(fetch::Repaint {
359            ctx: ctx.clone(),
360            viewport: ctx.viewport_id(),
361        });
362        let now = now();
363        let time = ui.input(|i| i.time);
364        let dt = self.last_frame.map_or(0.0, |t| (time - t).clamp(0.0, 0.1));
365        self.last_frame = Some(time);
366        self.take_results(now);
367        self.pulse = self.pulsing(ui, time);
368        let whole = ui.available_rect_before_wrap();
369        self.keyboard(ui, whole, now);
370
371        bar::top_bar(self, ui, menu, now);
372        ui.add_space(6.0);
373
374        // The legend's height first, so the plot fills the rest; never
375        // smaller than MIN_SIZE (an auto-sized window offers little).
376        let width = ui.available_width().max(MIN_SIZE.x);
377        let mut chips = legend::layout(self, ui, width);
378        let gap = if chips.height > 0.0 { 6.0 } else { 0.0 };
379        let plot_h = (ui.available_height() - chips.height - gap).max(MIN_SIZE.y);
380        let (rect, resp) =
381            ui.allocate_exact_size(Vec2::new(width, plot_h), Sense::click_and_drag());
382        ui.add_space(gap);
383        legend::interact(self, ui, &mut chips);
384
385        // Kinetic panning after a drag.
386        if self.glide != 0.0 && self.drag.is_none() {
387            self.window.pan(self.glide * dt, now);
388            self.glide *= (-5.0 * dt).exp();
389            self.changed_at = now;
390            self.snap = true;
391            if self.glide.abs() < self.window.span * 0.01 || self.window.live {
392                self.glide = 0.0;
393            }
394        }
395
396        let (from, to) = self.shown_range(now, dt);
397        let dark = ui.visuals().dark_mode;
398        let data: Vec<Vec<Pt>> = self
399            .traces
400            .iter()
401            .map(|t| {
402                if t.visible {
403                    t.points(from, to, now)
404                } else {
405                    Vec::new()
406                }
407            })
408            .collect();
409        let (lo, hi) = self.shown_values(&data, from, to, dt);
410
411        let font = egui::FontId::proportional(11.5);
412        let units = self.common_units();
413        let f = render::frame(
414            ui.painter(),
415            rect,
416            (from, to),
417            (lo, hi),
418            self.y.log,
419            &units,
420            &font,
421        );
422        self.interact(ui, &resp, &f, now);
423        let ppp = ui.ctx().pixels_per_point();
424        self.schedule_fetches(now, f.data.width() * ppp);
425
426        // Paint.
427        let lines: Vec<render::Line> = self
428            .traces
429            .iter()
430            .zip(&data)
431            .filter(|(t, _)| t.visible)
432            .map(|(t, pts)| {
433                let color = self.color(t, dark);
434                let (color, width) = match self.highlight {
435                    Some(h) if h == t.id => (color, 2.5),
436                    Some(_) => (color.gamma_multiply(0.25), 1.5),
437                    None => (color, 1.5),
438                };
439                render::Line { pts, color, width }
440            })
441            .collect();
442        let painter = ui.painter_at(rect);
443        if self.traces.is_empty() {
444            painter.rect_filled(rect, 10.0, ui.visuals().extreme_bg_color);
445        } else {
446            render::paint(&painter, &f, &lines, ui.visuals());
447            render::now_marker(&painter, &f, now, ui.visuals(), &font);
448        }
449        if let Some(Drag::Box(origin)) = self.drag
450            && let Some(p) = ui.input(|i| i.pointer.latest_pos())
451        {
452            let r = Rect::from_two_pos(origin, p).intersect(f.data);
453            let accent = ui.visuals().selection.bg_fill;
454            painter.rect(
455                r,
456                2.0,
457                accent.gamma_multiply(0.15),
458                egui::Stroke::new(1.0, accent),
459                egui::StrokeKind::Inside,
460            );
461        }
462        if self.traces.is_empty() {
463            hover::empty_state(&painter, &f, ui.visuals());
464        } else if !matches!(self.drag, Some(Drag::Pan | Drag::Axis)) {
465            hover::crosshair(self, ui, &f, &data, dark);
466        }
467        hover::status(self, ui, &f);
468        legend::paint(self, ui, &chips, &data, dark);
469
470        // Keep moving: easing, gliding, and time passing while live.
471        let target = self.window.range(now);
472        let settled = self.shown.is_some_and(|(a, b)| {
473            let tol = self.window.span * 1e-3;
474            (a - target.0).abs() <= tol && (b - target.1).abs() <= tol
475        });
476        if self.glide != 0.0 || !settled || self.pending {
477            ctx.request_repaint();
478        } else if self.window.live {
479            let per_pixel = self.window.span / f.data.width().max(1.0) as f64;
480            ctx.request_repaint_after(std::time::Duration::from_secs_f64(
481                per_pixel.clamp(1.0 / 15.0, 1.0),
482            ));
483            if self.pulse {
484                ctx.request_repaint_after(std::time::Duration::from_millis(66));
485            }
486        }
487    }

489    /// Whether the Live pill's dot pulses this frame: for a few seconds
490    /// after following now starts, and while the pointer is over the plot.
491    /// Otherwise the dot is still, and the plot repaints only at the
492    /// scroll pace (no 15 fps animation in an idle window).
493    fn pulsing(&mut self, ui: &egui::Ui, time: f64) -> bool {
494        if !self.window.live {
495            self.live_since = None;
496            return false;
497        }
498        let since = *self.live_since.get_or_insert(time);
499        time - since < PULSE_SECS || ui.ui_contains_pointer()
500    }

512    fn common_units(&self) -> String {
513        let units: Vec<String> = self
514            .traces
515            .iter()
516            .filter(|t| t.visible)
517            .map(units)

525    /// The time range to draw this frame, easing toward the window.
526    fn shown_range(&mut self, now: f64, dt: f64) -> (f64, f64) {
527        let (from, to) = self.window.range(now);
528        let (mut a, mut b) = match self.shown {
529            Some(s) if !std::mem::take(&mut self.snap) => s,
530            _ => (from, to),
531        };
532        let span = to - from;
533        window::ease(&mut a, from, dt, span);
534        window::ease(&mut b, to, dt, span);
535        self.shown = Some((a, b));
536        (a, b)
537    }

539    /// The value range to draw this frame: fitted to the visible data
540    /// (envelopes included) or fixed, easing toward it.
541    fn shown_values(&mut self, data: &[Vec<Pt>], from: f64, to: f64, dt: f64) -> (f64, f64) {
542        let log = self.y.log;
543        let target = if self.y.autoscale {
544            let values = data.iter().flat_map(|pts| {
545                let start = pts.partition_point(|p| p.t < from).saturating_sub(1);
546                pts[start..]
547                    .iter()
548                    .take_while(|p| p.t <= to)
549                    .flat_map(|p| [p.lo, p.hi, p.v])
550            });
551            render::autoscale(log, values).or(self.y.shown)
552        } else {
553            Some((self.y.lo, self.y.hi))
554        };
555        let (lo, hi) = target.unwrap_or(if log { (1.0, 10.0) } else { (0.0, 1.0) });
556        let (mut a, mut b) = self.y.shown.unwrap_or((lo, hi));
557        let scale = hi - lo;
558        window::ease(&mut a, lo, dt, scale);
559        window::ease(&mut b, hi, dt, scale);
560        if a.partial_cmp(&b) != Some(std::cmp::Ordering::Less) {
561            (a, b) = (lo, hi);
562        }
563        self.y.shown = Some((a, b));
564        (a, b)
565    }

567    /// Take in finished fetches and searches.
568    fn take_results(&mut self, now: f64) {
569        for done in self.fetcher.take() {
570            let Some(t) = self.traces.iter_mut().find(|t| t.id == done.trace) else {
571                continue;
572            };
573            if done.generation != t.fetch.generation {
574                continue;
575            }
576            t.fetch.in_flight = false;
577            if done.result.is_ok() {
578                t.fetch.failures = 0;
579            }
580            // The window really fetched: a raw-only trace's is shorter.
581            if let Some(r) = &mut t.fetch.requested {
582                r.from = r.from.max(done.from);
583            }
584            if let Some(info) = &done.info {
585                t.fetch.raw_only = Some(!info.binnable());
586            }
587            match done.result {
588                Ok(series) => {
589                    let series = series.unwrap_or_default();
590                    let scalar = series.kind == ValueKind::Scalar;
591                    if series.samples.is_empty() {
592                        // Nothing to tell the type by.
593                    } else if !scalar && done.op != Operator::Raw {
594                        // A waveform or text: ask again, raw.
595                        t.fetch.raw_only = Some(true);
596                        t.fetch.requested = None;
597                    } else if t.fetch.raw_only.is_none() {
598                        t.fetch.raw_only = Some(!scalar);
599                    }
600                    let meta = if series.meta.name.is_empty() {
601                        t.history
602                            .as_ref()
603                            .map(|h| h.meta.clone())
604                            .unwrap_or_default()
605                    } else {
606                        series.meta
607                    };
608                    t.history = Some(History {
609                        to: done.to,
610                        binned: series.binned,
611                        kind: series.kind,
612                        meta,
613                        samples: series.samples,
614                    });
615                    t.fetch.error = None;
616                }
617                Err(e) if e.is_not_archived() => {
618                    t.fetch.not_archived = true;
619                    t.fetch.error = None;
620                    t.history = None;
621                }
622                Err(e) => {
623                    log::warn!("{}: {e}", t.name);
624                    t.fetch.error = Some(e.to_string());
625                    t.fetch.failures += 1;
626                }
627            }
628        }
629        bar::take_found(self, now);
630    }

632    /// Start the fetches the visible window needs, debounced while the
633    /// user pans and zooms.
634    fn schedule_fetches(&mut self, now: f64, width_px: f32) {
635        self.pending = false;
636        let Some(client) = self.archiver.clone() else {
637            return;
638        };
639        let (from, to) = self.window.range(now);
640        let settled = now - self.changed_at >= DEBOUNCE;
641        let live = self.window.live;
642        for t in &mut self.traces {
643            let Some(name) = t.archive_name() else {
644                continue;
645            };
646            if t.fetch.not_archived {
647                continue;
648            }
649            let raw = t.fetch.raw_only.unwrap_or(false);
650            let from = fetch::needed_from(from, to, raw);
651            let need = match t.fetch.requested {
652                None => true,
653                Some(r) => {
654                    let binned =
655                        r.op != Operator::Raw && t.history.as_ref().is_none_or(|h| h.binned);
656                    let covered = fetch::covers((r.from, r.to), binned, from, to);
657                    // After an error: again once the backoff has passed, or
658                    // after the base delay for a different window.
659                    let wait = if covered {
660                        fetch::backoff(RETRY, t.fetch.failures)
661                    } else {
662                        RETRY
663                    };
664                    if t.fetch.error.is_some() {
665                        !t.fetch.in_flight && now - r.at > wait
666                    } else {
667                        !covered || (live && !t.fetch.in_flight && now - r.at > LIVE_REFRESH)
668                    }
669                }
670            };
671            if !need {
672                continue;
673            }
674            if t.fetch.requested.is_some() && !settled {
675                self.pending = true;
676                continue;
677            }
678            let (a, b) = fetch::padded(from, to);
679            let op = fetch::operator(width_px, raw);
680            let raw_from = fetch::padded(fetch::needed_from(from, to, true), to).0;
681            t.fetch.generation += 1;
682            t.fetch.in_flight = true;
683            t.fetch.requested = Some(series::Requested {
684                from: a,
685                to: b,
686                op,
687                at: now,
688            });
689            let job = fetch::Job {
690                trace: t.id,
691                generation: t.fetch.generation,
692                pv: name,
693                from: a,
694                to: b,
695                op,
696                raw_from,
697                probe: t.fetch.raw_only.is_none(),
698            };
699            self.fetcher
700                .fetch(&self.pvs, &client, job, self.repaint.clone());
701        }
702    }

704    /// Pan, zoom, box zoom and reset on the plot.
705    fn interact(&mut self, ui: &egui::Ui, resp: &egui::Response, f: &render::Frame, now: f64) {
706        let data = f.data;
707        let span = self.window.span;
708        let per_px = span / data.width().max(1.0) as f64;
709        if resp.clicked() || resp.drag_started() {
710            self.active = true;
711        } else if ui.input(|i| i.pointer.any_pressed()) && !resp.contains_pointer() {
712            self.active = false;
713        }
714
715        if resp.drag_started() {
716            let origin = ui
717                .input(|i| i.pointer.press_origin())
718                .unwrap_or(data.center());
719            let shift = ui.input(|i| i.modifiers.shift);
720            self.glide = 0.0;
721            self.drag = Some(
722                if shift || resp.dragged_by(egui::PointerButton::Secondary) {
723                    Drag::Box(origin)
724                } else if f.value_axis().contains(origin) {
725                    Drag::Axis
726                } else {
727                    Drag::Pan
728                },
729            );
730        }
731        if resp.dragged() {
732            let d = resp.drag_delta();
733            match self.drag {
734                Some(Drag::Pan) => {
735                    self.window.pan(-d.x as f64 * per_px, now);
736                    if !self.y.autoscale && d.y != 0.0 {
737                        let (lo, hi) = self.y.shown.unwrap_or((self.y.lo, self.y.hi));
738                        let frac = -(d.y / data.height().max(1.0)) as f64;
739                        (self.y.lo, self.y.hi) = window::pan_values(lo, hi, frac, self.y.log);
740                        self.y.shown = Some((self.y.lo, self.y.hi));
741                    }
742                    self.snap = true;
743                    self.changed_at = now;
744                }
745                Some(Drag::Axis) => {
746                    let factor = (d.y as f64 * 0.01).exp();
747                    self.zoom_values(factor, None, f);
748                    self.y.shown = Some((self.y.lo, self.y.hi));
749                }
750                _ => {}
751            }
752        }
753        if resp.drag_stopped() {
754            match self.drag.take() {
755                Some(Drag::Pan) => {
756                    let v = ui.input(|i| i.pointer.velocity()).x as f64;
757                    if v.abs() > 150.0 {
758                        self.glide = -v * per_px;
759                    }
760                }
761                Some(Drag::Box(origin)) => {
762                    if let Some(p) = ui.input(|i| i.pointer.latest_pos()) {
763                        let r = Rect::from_two_pos(origin, p).intersect(data);
764                        if r.width() > 6.0 {
765                            self.window.set(f.time_at(r.left()), f.time_at(r.right()));
766                        }
767                        if r.height() > 6.0 {
768                            self.y.autoscale = false;
769                            (self.y.lo, self.y.hi) = (f.value_at(r.bottom()), f.value_at(r.top()));
770                        }
771                    }
772                }
773                _ => {}
774            }
775        }
776
777        if resp.contains_pointer() {
778            let (scroll, zoom, mods, pointer) = ui.input(|i| {
779                (
780                    i.smooth_scroll_delta,
781                    i.zoom_delta(),
782                    i.modifiers,
783                    i.pointer.hover_pos(),
784                )
785            });
786            let anchor = pointer.map_or(1.0, |p| ((p.x - data.left()) / data.width()) as f64);
787            let at_value = pointer.map(|p| f.value_at(p.y));
788            if mods.shift && (scroll.x != 0.0 || scroll.y != 0.0) {
789                // Shift+wheel arrives as horizontal scrolling.
790                let amount = (scroll.x + scroll.y) as f64;
791                self.zoom_values((-amount / 300.0).exp(), at_value, f);
792            } else if zoom != 1.0 {
793                if mods.ctrl || mods.command {
794                    self.zoom_values(1.0 / zoom as f64, at_value, f);
795                } else {
796                    self.window.zoom(1.0 / zoom as f64, anchor, now);
797                    self.changed_at = now;
798                }
799            } else {
800                if scroll.y != 0.0 {
801                    self.window
802                        .zoom((-scroll.y as f64 / 300.0).exp(), anchor, now);
803                    self.changed_at = now;
804                }
805                if scroll.x != 0.0 {
806                    self.window.pan(-scroll.x as f64 * per_px, now);
807                    self.changed_at = now;
808                    self.snap = true;
809                }
810            }
811            if scroll != Vec2::ZERO {
812                // Not for a scroll area around the plot too.
813                ui.ctx().input_mut(|i| i.smooth_scroll_delta = Vec2::ZERO);
814            }
815        }
816        if resp.double_clicked() {
817            self.reset();
818        }
819    }

821    /// Zoom values by `factor` about `at` (the middle when `None`), leaving
822    /// autoscale.
823    fn zoom_values(&mut self, factor: f64, at: Option<f64>, f: &render::Frame) {
824        let (lo, hi) = self.y.shown.unwrap_or((f.ys.lo, f.ys.hi));
825        let mid = if self.y.log && lo > 0.0 {
826            (lo * hi).sqrt()
827        } else {
828            (lo + hi) / 2.0
829        };
830        let (l, h) = window::zoom_values(lo, hi, factor, at.unwrap_or(mid), self.y.log);
831        if l.is_finite() && h.is_finite() && l < h {
832            self.y.autoscale = false;
833            (self.y.lo, self.y.hi) = (l, h);
834        }
835    }

862    /// The keyboard shortcuts, when the pointer is over the plot or it was
863    /// clicked last, and no text field has the keyboard.
864    fn keyboard(&mut self, ui: &egui::Ui, whole: Rect, now: f64) {
865        let typing = ui.ctx().memory(|m| m.focused().is_some());
866        if typing || !(self.active || ui.rect_contains_pointer(whole)) {
867            return;
868        }
869        use egui::Key;
870        let pressed = |k: Key| ui.input(|i| i.key_pressed(k) && !i.modifiers.ctrl);
871        let span = self.window.span;
872        if pressed(Key::Slash) {
873            self.search.focus = true;
874        }
875        if pressed(Key::ArrowLeft) {
876            self.window.pan(-0.1 * span, now);
877            self.changed_at = now;
878        }
879        if pressed(Key::ArrowRight) {
880            self.window.pan(0.1 * span, now);
881            self.changed_at = now;
882        }
883        if pressed(Key::Plus) || pressed(Key::Equals) {
884            self.window.zoom(0.8, 0.5, now);
885            self.changed_at = now;
886        }
887        if pressed(Key::Minus) {
888            self.window.zoom(1.25, 0.5, now);
889            self.changed_at = now;
890        }
891        if pressed(Key::L) {
892            self.toggle_live(now);
893        }
894        if pressed(Key::A) {
895            self.y.autoscale = true;
896        }
897        if pressed(Key::Home) {
898            self.reset();
899        }
900    }

903/// The units of trace `t`: the live PV's, else the archive's.
904fn units(t: &Trace) -> String {
905    let live = t.pv.state().units;
906    if !live.is_empty() {
907        return live;
908    }
909    t.history
910        .as_ref()
911        .and_then(|h| h.meta.egu.clone())
912        .unwrap_or_default()
913}

915/// The precision of trace `t`: the live PV's, else the archive's.
916fn precision(t: &Trace, state: &pv::PvState) -> Option<usize> {
917    state
918        .precision
919        .filter(|p| *p >= 0)
920        .map(|p| p as usize)
921        .or_else(|| {
922            t.history
923                .as_ref()
924                .and_then(|h| h.meta.prec)
925                .map(|p| p as usize)
926        })
927}

929/// A value for the readout and chips: at `prec` decimals, else compactly.
930fn format_value(v: f64, prec: Option<usize>, units: &str) -> String {
931    if !v.is_finite() {
932        return "—".into();
933    }
934    let text = match prec {
935        Some(p) => {
936            let s = format!("{v:.*}", p.min(12));
937            // -0.04 at one decimal reads "0.0", not "-0.0".
938            match s.strip_prefix('-') {
939                Some(rest) if rest.chars().all(|c| c == '0' || c == '.') => rest.to_string(),
940                _ => s,
941            }
942        }
943        None => {
944            let a = v.abs();
945            if a != 0.0 && !(1e-3..1e6).contains(&a) {
946                format!("{v:.3e}")
947            } else {
948                let s = format!("{v:.4}");
949                let s = s.trim_end_matches('0').trim_end_matches('.');
950                if s == "-0" { "0".into() } else { s.to_string() }
951            }
952        }
953    };
954    if units.is_empty() {
955        text
956    } else {
957        format!("{text} {units}")
958    }
959}
