64/// An archived + live plot of a set of PVs on one shared time axis.
65///
66/// Owns its traces, background fetches and live subscriptions; dropping it
67/// unsubscribes and abandons in-flight fetches. Its only shared state is
68/// the counter [`Plot::new`] salts ids with, so any number may exist at
69/// once.
70pub struct Plot {
71    pvs: PvManager,
72    archiver: Option<ArchiverClient>,
73    /// Salt for every egui id this plot uses.
74    id: egui::Id,

77    window: Window,
78    /// The span a reset (double-click, Home) returns to.
79    home_span: f64,
80    /// The range drawn, easing toward `window`'s.
81    shown: Option<(f64, f64)>,
82    y: YState,
83    /// The ranges drawn on the value axes after the first, by units,
84    /// easing toward their targets (always autoscaled).
85    axis_shown: Vec<(String, (f64, f64))>,
86    /// Each trace's value axis this frame, in trace order.
87    axis_of: Vec<usize>,
88    fetcher: Fetcher,
89    /// When (seconds since 1970) the user last panned or zoomed, for the
90    /// fetch debounce.
91    changed_at: f64,
92    /// A fetch is waiting for the debounce.
93    pending: bool,
94    /// Where finished requests ask for a repaint: set every frame.
95    repaint: Option<fetch::Repaint>,
96    /// When (egui time) the plot last started following now, for the
97    /// Live pill's pulse.
98    live_since: Option<f64>,
99    /// The Live pill's dot pulses this frame.
100    pulse: bool,
101    /// egui's time at the last frame, for animation steps.
102    last_frame: Option<f64>,
103    /// Draw the next frame without easing (direct manipulation).
104    snap: bool,
105    /// Kinetic panning after a drag, in seconds of time per second.

110    /// Keyboard shortcuts apply: the plot was clicked last.

112    search: bar::SearchState,

229    fn add(&mut self, name: &str) -> Option<&mut Trace> {
230        let name = name.trim();
231        let key = pv::PvName::parse(name).key();
232        if name.is_empty() || self.traces.iter().any(|t| t.pv.name().key() == key) {
233            return None;
234        }
235        // The first palette colour no other trace has.
236        let slot = (0..)
237            .find(|s| !self.traces.iter().any(|t| t.slot == *s))
238            .unwrap_or(0);
239        let mut trace = Trace::new(self.next_trace, name, self.pvs.subscribe(name));
240        trace.slot = slot;
241        self.next_trace += 1;
242        self.traces.push(trace);
243        self.traces.last_mut()
244    }

369    fn show(&mut self, ui: &mut egui::Ui, menu: Option<&mut dyn FnMut(&mut egui::Ui)>) {
370        let ctx = ui.ctx().clone();
371        self.repaint = Some(fetch::Repaint {
372            ctx: ctx.clone(),
373            viewport: ctx.viewport_id(),
374        });
375        let now = now();
376        let time = ui.input(|i| i.time);
377        let dt = self.last_frame.map_or(0.0, |t| (time - t).clamp(0.0, 0.1));
378        self.last_frame = Some(time);
379        self.take_results(now);
380        self.pulse = self.pulsing(ui, time);
381        let whole = ui.available_rect_before_wrap();
382        self.keyboard(ui, whole, now);
383
384        bar::top_bar(self, ui, menu, now);
385        ui.add_space(6.0);
386
387        // The legend's height first, so the plot fills the rest; never
388        // smaller than MIN_SIZE (an auto-sized window offers little).
389        let width = ui.available_width().max(MIN_SIZE.x);
390        let mut chips = legend::layout(self, ui, width);
391        let gap = if chips.height > 0.0 { 6.0 } else { 0.0 };
392        let plot_h = (ui.available_height() - chips.height - gap).max(MIN_SIZE.y);
393        let (rect, resp) =
394            ui.allocate_exact_size(Vec2::new(width, plot_h), Sense::click_and_drag());
395        ui.add_space(gap);
396        legend::interact(self, ui, &mut chips);
397
398        // Kinetic panning after a drag.
399        if self.glide != 0.0 && self.drag.is_none() {
400            self.window.pan(self.glide * dt, now);
401            self.glide *= (-5.0 * dt).exp();
402            self.changed_at = now;
403            self.snap = true;
404            if self.glide.abs() < self.window.span * 0.01 || self.window.live {
405                self.glide = 0.0;
406            }
407        }
408
409        let (from, to) = self.shown_range(now, dt);
410        let dark = ui.visuals().dark_mode;
411        let data: Vec<Vec<Pt>> = self
412            .traces
413            .iter()
414            .map(|t| {
415                if t.visible {
416                    t.points(from, to, now)
417                } else {
418                    Vec::new()
419                }
420            })
421            .collect();
422        let axes = self.value_axes(&data, from, to, dt, dark);
423
424        let font = egui::FontId::proportional(11.5);
425        let f = render::frame(ui.painter(), rect, (from, to), &axes, self.y.log, &font);
426        self.interact(ui, &resp, &f, now);
427        let ppp = ui.ctx().pixels_per_point();
428        self.schedule_fetches(now, f.data.width() * ppp);
429
430        // Paint.
431        let lines: Vec<render::Line> = self
432            .traces
433            .iter()
434            .zip(&data)
435            .zip(&self.axis_of)
436            .filter(|((t, _), _)| t.visible)
437            .map(|((t, pts), &axis)| {
438                let color = self.color(t, dark);
439                let (color, width) = match self.highlight {
440                    Some(h) if h == t.id => (color, 2.5),
441                    Some(_) => (color.gamma_multiply(0.25), 1.5),
442                    None => (color, 1.5),
443                };
444                render::Line {
445                    pts,
446                    color,
447                    width,
448                    axis,
449                }
450            })
451            .collect();
452        let painter = ui.painter_at(rect);
453        // The crosshair, first so its time tag can hide the label under it.
454        let hair = (!self.traces.is_empty() && !matches!(self.drag, Some(Drag::Pan | Drag::Axis)))
455            .then(|| hover::hairline(self, ui, &f, &data))
456            .flatten();
457        if self.traces.is_empty() {
458            painter.rect_filled(rect, 10.0, ui.visuals().extreme_bg_color);
459        } else {
460            let cover = hair.map(|(p, t)| hover::time_tag_rect(&painter, &f, p.x, t));
461            render::paint(&painter, &f, &lines, ui.visuals(), cover);
462            render::now_marker(&painter, &f, now, ui.visuals(), &font);
463        }
464        if let Some(Drag::Box(origin)) = self.drag
465            && let Some(p) = ui.input(|i| i.pointer.latest_pos())
466        {
467            let r = Rect::from_two_pos(origin, p).intersect(f.data);
468            let accent = ui.visuals().selection.bg_fill;
469            painter.rect(
470                r,
471                2.0,
472                accent.gamma_multiply(0.15),
473                egui::Stroke::new(1.0, accent),
474                egui::StrokeKind::Inside,
475            );
476        }
477        if self.traces.is_empty() {
478            hover::empty_state(&painter, &f, ui.visuals());
479        } else if let Some(hair) = hair {
480            hover::crosshair(self, ui, &f, &data, dark, hair);
481        }
482        hover::status(self, ui, &f);
483        legend::paint(self, ui, &chips, &data, dark);
484
485        // Keep moving: easing, gliding, and time passing while live.
486        let target = self.window.range(now);
487        let settled = self.shown.is_some_and(|(a, b)| {
488            let tol = self.window.span * 1e-3;
489            (a - target.0).abs() <= tol && (b - target.1).abs() <= tol
490        });
491        if self.glide != 0.0 || !settled || self.pending {
492            ctx.request_repaint();
493        } else if self.window.live {
494            let per_pixel = self.window.span / f.data.width().max(1.0) as f64;
495            ctx.request_repaint_after(std::time::Duration::from_secs_f64(
496                per_pixel.clamp(1.0 / 15.0, 1.0),
497            ));
498            if self.pulse {
499                ctx.request_repaint_after(std::time::Duration::from_millis(66));
500            }
501        }
502    }

504    /// Whether the Live pill's dot pulses this frame: for a few seconds
505    /// after following now starts, and while the pointer is over the plot.
506    /// Otherwise the dot is still, and the plot repaints only at the
507    /// scroll pace (no 15 fps animation in an idle window).
508    fn pulsing(&mut self, ui: &egui::Ui, time: f64) -> bool {
509        if !self.window.live {
510            self.live_since = None;
511            return false;
512        }
513        let since = *self.live_since.get_or_insert(time);
514        time - since < PULSE_SECS || ui.ui_contains_pointer()
515    }

526    /// The time range to draw this frame, easing toward the window.
527    fn shown_range(&mut self, now: f64, dt: f64) -> (f64, f64) {
528        let (from, to) = self.window.range(now);
529        let (mut a, mut b) = match self.shown {
530            Some(s) if !std::mem::take(&mut self.snap) => s,
531            _ => (from, to),
532        };
533        let span = to - from;
534        window::ease(&mut a, from, dt, span);
535        window::ease(&mut b, to, dt, span);
536        self.shown = Some((a, b));
537        (a, b)
538    }

540    /// The value axes to draw this frame, one per distinct unit among the
541    /// visible traces with data (at most [`MAX_AXES`]), and `axis_of`, each trace's.
542    /// The first takes the fixed range, log scale and value zooming; the
543    /// others autoscale to their own traces.
544    fn value_axes(
545        &mut self,
546        data: &[Vec<Pt>],
547        from: f64,
548        to: f64,
549        dt: f64,
550        dark: bool,
551    ) -> Vec<render::AxisSpec> {
552        // Only traces with something to show claim an axis.
553        let trace_units: Vec<Option<String>> = self
554            .traces
555            .iter()
556            .zip(data)
557            .map(|(t, pts)| (t.visible && pts.iter().any(|p| p.v.is_finite())).then(|| units(t)))
558            .collect();
559        let (names, axis_of) = group_units(&trace_units);
560        self.axis_of = axis_of;
561        let several = names.len() > 1;
562        self.axis_shown.retain(|(u, _)| names[1..].contains(u));
563        let mut specs = Vec::new();
564        for (a, name) in names.iter().enumerate() {
565            let members: Vec<usize> = (0..self.traces.len())
566                .filter(|&i| trace_units[i].is_some() && self.axis_of[i] == a)
567                .collect();
568            let pts = members.iter().map(|&i| &data[i]);
569            let whole =
570                !members.is_empty() && members.iter().all(|&i| whole_valued(&self.traces[i]));
571            let (lo, hi) = if a == 0 {
572                self.shown_values(pts, (from, to), whole, dt)
573            } else {
574                let fit = fit_values(self.y.log, whole, pts, from, to);
575                let i = match self.axis_shown.iter().position(|(u, _)| u == name) {
576                    Some(i) => i,
577                    None => {
578                        let first = fit.unwrap_or((0.0, 1.0));
579                        self.axis_shown.push((name.clone(), first));
580                        self.axis_shown.len() - 1
581                    }
582                };
583                let shown = &mut self.axis_shown[i].1;
584                *shown = ease_values(*shown, fit.unwrap_or(*shown), dt);
585                *shown
586            };
587            let title = axis_title(members.iter().filter_map(|&i| trace_units[i].as_deref()));
588            let color = match members.as_slice() {
589                [one] if several => Some(self.color(&self.traces[*one], dark)),
590                _ => None,
591            };
592            specs.push(render::AxisSpec {
593                lo,
594                hi,
595                title,
596                color,
597                whole,
598            });
599        }
600        specs
601    }

603    /// The primary axis' value range to draw this frame: fitted to its
604    /// traces' visible data (envelopes included) or fixed, easing toward
605    /// it.
606    fn shown_values<'a>(
607        &mut self,
608        data: impl Iterator<Item = &'a Vec<Pt>>,
609        (from, to): (f64, f64),
610        whole: bool,
611        dt: f64,
612    ) -> (f64, f64) {
613        let log = self.y.log;
614        let target = if self.y.autoscale {
615            fit_values(log, whole, data, from, to).or(self.y.shown)
616        } else {
617            Some((self.y.lo, self.y.hi))
618        };

634    /// Take in finished fetches and searches.
635    fn take_results(&mut self, now: f64) {
636        for done in self.fetcher.take() {
637            let Some(t) = self.traces.iter_mut().find(|t| t.id == done.trace) else {
638                continue;
639            };
640            if done.generation != t.fetch.generation {
641                continue;
642            }
643            t.fetch.in_flight = false;
644            if done.result.is_ok() {
645                t.fetch.failures = 0;
646            }
647            // The window really fetched: a raw-only trace's is shorter.
648            if let Some(r) = &mut t.fetch.requested
649                && !done.append
650            {
651                r.from = r.from.max(done.from);
652            }
653            if let Some(info) = &done.info {
654                t.fetch.raw_only = Some(!info.binnable());
655            }
656            match done.result {
657                Ok(series) if done.append => {
658                    let series = series.unwrap_or_default();
659                    let keep_from = t.fetch.requested.map_or(f64::NEG_INFINITY, |r| r.from);
660                    match &mut t.history {
661                        Some(h) if !h.binned && !series.binned => {
662                            h.append(series.samples, keep_from, done.to);
663                            t.fetch.error = None;
664                            // Grown past what `optimized_N` returns raw:
665                            // fetch it whole again, binned.
666                            if fetch::outgrown(t.fetch.requested.map(|r| r.op), h.samples.len()) {
667                                t.fetch.requested = None;
668                            }
669                        }
670                        // Not what the history holds: fetch it all again.
671                        _ => t.fetch.requested = None,
672                    }
673                }
674                Ok(series) => {
675                    let series = series.unwrap_or_default();
676                    let scalar = series.kind == ValueKind::Scalar;
677                    if series.samples.is_empty() {
678                        // Nothing to tell the type by.
679                    } else if !scalar && done.op != Operator::Raw {
680                        // A waveform or text: ask again, raw.
681                        t.fetch.raw_only = Some(true);
682                        t.fetch.requested = None;
683                    } else if t.fetch.raw_only.is_none() {
684                        t.fetch.raw_only = Some(!scalar);
685                    }
686                    let meta = if series.meta.name.is_empty() {
687                        t.history
688                            .as_ref()
689                            .map(|h| h.meta.clone())
690                            .unwrap_or_default()
691                    } else {
692                        series.meta
693                    };
694                    t.history = Some(History {
695                        to: done.to,
696                        binned: series.binned,
697                        kind: series.kind,
698                        meta,
699                        samples: series.samples,
700                    });
701                    t.fetch.error = None;
702                }
703                Err(e) if e.is_not_archived() => {
704                    t.fetch.not_archived = true;
705                    t.fetch.error = None;
706                    t.history = None;
707                }
708                Err(e) => {
709                    log::warn!("{}: {e}", t.name);
710                    t.fetch.error = Some(e.to_string());
711                    t.fetch.failures += 1;
712                }
713            }
714        }
715        bar::take_found(self, now);
716    }

718    /// Start the fetches the visible window needs, debounced while the
719    /// user pans and zooms.
720    fn schedule_fetches(&mut self, now: f64, width_px: f32) {
721        self.pending = false;
722        let Some(client) = self.archiver.clone() else {
723            return;
724        };
725        let (from, to) = self.window.range(now);
726        let settled = now - self.changed_at >= DEBOUNCE;
727        let live = self.window.live;
728        for t in &mut self.traces {
729            let Some(name) = t.archive_name() else {
730                continue;
731            };
732            // One request per trace at a time: when it is done, the next
733            // frame asks for wherever the window is then.
734            if t.fetch.not_archived || t.fetch.in_flight {
735                continue;
736            }
737            let raw = t.fetch.raw_only.unwrap_or(false);
738            let from = fetch::needed_from(from, to, raw);
739            // A live refresh of raw history: only from its last sample on.
740            let mut tail = None;
741            let need = match t.fetch.requested {
742                None => true,
743                Some(r) => {
744                    let binned =
745                        r.op != Operator::Raw && t.history.as_ref().is_none_or(|h| h.binned);
746                    let covered = fetch::covers((r.from, r.to), binned, from, to);
747                    // After an error: again once the backoff has passed, or
748                    // after the base delay for a different window.
749                    let wait = if covered {
750                        fetch::backoff(RETRY, t.fetch.failures)
751                    } else {
752                        RETRY
753                    };
754                    if t.fetch.error.is_some() {
755                        now - r.at > wait
756                    } else if !covered {
757                        true
758                    } else if live
759                        && now - r.at
760                            > fetch::refresh_after(LIVE_REFRESH, (r.from, r.to), r.op, binned)
761                    {
762                        tail = t
763                            .history
764                            .as_ref()
765                            .filter(|h| !h.binned)
766                            .and_then(|h| h.samples.last())
767                            .map(|s| s.t)
768                            .filter(|t| *t > r.from)
769                            .map(|t| (t, r));
770                        true
771                    } else {
772                        false
773                    }
774                }
775            };
776            if !need {
777                continue;
778            }
779            if t.fetch.requested.is_some() && !settled {
780                self.pending = true;
781                continue;
782            }
783            let (a, b) = fetch::padded(from, to);
784            let op = fetch::operator(width_px, raw);
785            let raw_from = fetch::padded(fetch::needed_from(from, to, true), to).0;
786            t.fetch.generation += 1;
787            t.fetch.in_flight = true;
788            let job = match tail {
789                // The history keeps what is still in the padded window.
790                Some((last, r)) => {
791                    t.fetch.requested = Some(series::Requested {
792                        from: r.from.max(a).min(last),
793                        to: b,
794                        op: r.op,
795                        at: now,
796                    });
797                    fetch::Job {
798                        trace: t.id,
799                        generation: t.fetch.generation,
800                        pv: name,
801                        from: last,
802                        to: b,
803                        op: Operator::Raw,
804                        raw_from: last,
805                        probe: false,
806                        append: true,
807                    }
808                }
809                None => {
810                    t.fetch.requested = Some(series::Requested {
811                        from: a,
812                        to: b,
813                        op,
814                        at: now,
815                    });
816                    fetch::Job {
817                        trace: t.id,
818                        generation: t.fetch.generation,
819                        pv: name,
820                        from: a,
821                        to: b,
822                        op,
823                        raw_from,
824                        probe: t.fetch.raw_only.is_none(),
825                        append: false,
826                    }
827                }
828            };
829            self.fetcher
830                .fetch(&self.pvs, &client, job, self.repaint.clone());
831        }
832    }

836        let data = f.data;
837        let span = self.window.span;
838        let per_px = span / data.width().max(1.0) as f64;
839        if resp.clicked() || resp.drag_started() {
840            self.active = true;
841        } else if ui.input(|i| i.pointer.any_pressed()) && !resp.contains_pointer() {
842            self.active = false;
843        }
844
845        if resp.drag_started() {
846            let origin = ui
847                .input(|i| i.pointer.press_origin())
848                .unwrap_or(data.center());
849            let shift = ui.input(|i| i.modifiers.shift);
850            self.glide = 0.0;
851            self.drag = Some(
852                if shift || resp.dragged_by(egui::PointerButton::Secondary) {
853                    Drag::Box(origin)
854                } else if f.value_axis().contains(origin) {
855                    Drag::Axis
856                } else {
857                    Drag::Pan

860        }
861        if resp.dragged() {
862            let d = resp.drag_delta();
863            match self.drag {
864                Some(Drag::Pan) => {
865                    self.window.pan(-d.x as f64 * per_px, now);
866                    if !self.y.autoscale && d.y != 0.0 {
867                        let (lo, hi) = self.y.shown.unwrap_or((self.y.lo, self.y.hi));
868                        let frac = -(d.y / data.height().max(1.0)) as f64;

870                        self.y.shown = Some((self.y.lo, self.y.hi));
871                    }
872                    self.snap = true;
873                    self.changed_at = now;
874                }
875                Some(Drag::Axis) => {
876                    let factor = (d.y as f64 * 0.01).exp();
877                    self.zoom_values(factor, None, f);
878                    self.y.shown = Some((self.y.lo, self.y.hi));
879                }
880                _ => {}
881            }
882        }
883        if resp.drag_stopped() {
884            match self.drag.take() {
885                Some(Drag::Pan) => {
886                    let v = ui.input(|i| i.pointer.velocity()).x as f64;
887                    if v.abs() > 150.0 {
888                        self.glide = -v * per_px;
889                    }
890                }
891                Some(Drag::Box(origin)) => {

894                        if r.width() > 6.0 {
895                            self.window.set(f.time_at(r.left()), f.time_at(r.right()));
896                        }
897                        if r.height() > 6.0 {
898                            self.y.autoscale = false;
899                            (self.y.lo, self.y.hi) = (f.value_at(r.bottom()), f.value_at(r.top()));
900                        }
901                    }
902                }
903                _ => {}
904            }
905        }
906
907        if resp.contains_pointer() {
908            let (scroll, zoom, mods, pointer) = ui.input(|i| {
909                (
910                    i.smooth_scroll_delta,
911                    i.zoom_delta(),
912                    i.modifiers,
913                    i.pointer.hover_pos(),
914                )
915            });
916            let anchor = pointer.map_or(1.0, |p| ((p.x - data.left()) / data.width()) as f64);
917            let at_value = pointer.map(|p| f.value_at(p.y));
918            if mods.shift && (scroll.x != 0.0 || scroll.y != 0.0) {
919                // Shift+wheel arrives as horizontal scrolling.
920                let amount = (scroll.x + scroll.y) as f64;
921                self.zoom_values((-amount / 300.0).exp(), at_value, f);
922            } else if zoom != 1.0 {
923                if mods.ctrl || mods.command {
924                    self.zoom_values(1.0 / zoom as f64, at_value, f);
925                } else {
926                    self.window.zoom(1.0 / zoom as f64, anchor, now);
927                    self.changed_at = now;
928                }
929            } else {
930                if scroll.y != 0.0 {
931                    self.window
932                        .zoom((-scroll.y as f64 / 300.0).exp(), anchor, now);
933                    self.changed_at = now;
934                }
935                if scroll.x != 0.0 {
936                    self.window.pan(-scroll.x as f64 * per_px, now);
937                    self.changed_at = now;
938                    self.snap = true;
939                }
940            }
941            if scroll != Vec2::ZERO {
942                // Not for a scroll area around the plot too.
943                ui.ctx().input_mut(|i| i.smooth_scroll_delta = Vec2::ZERO);
944            }
945        }
946        if resp.double_clicked() {
947            self.reset();
948        }
949    }

1033/// Group traces by units (`None`: hidden) into value axes, in order of
1034/// first appearance: the axes' units, and each trace's axis (0 when
1035/// hidden). Units beyond [`MAX_AXES`] share the last axis. With no visible
1036/// traces there is one axis, without units.
1037fn group_units(units: &[Option<String>]) -> (Vec<String>, Vec<usize>) {
1038    let mut names: Vec<String> = Vec::new();
1039    let of = units
1040        .iter()
1041        .map(|u| {
1042            let Some(u) = u else { return 0 };
1043            match names.iter().position(|n| n == u) {
1044                Some(i) => i,
1045                None if names.len() < MAX_AXES => {
1046                    names.push(u.clone());
1047                    names.len() - 1
1048                }
1049                None => MAX_AXES - 1,
1050            }
1051        })
1052        .collect();
1053    if names.is_empty() {
1054        names.push(String::new());
1055    }
1056    (names, of)
1057}

1082/// The autoscaled range for `data` between `from` and `to` (at least two
1083/// wide, for `whole` numbers on a linear scale).
1084fn fit_values<'a>(
1085    log: bool,
1086    whole: bool,
1087    data: impl Iterator<Item = &'a Vec<Pt>>,
1088    from: f64,
1089    to: f64,
1090) -> Option<(f64, f64)> {
1091    let values = data.flat_map(|pts| {
1092        let start = pts.partition_point(|p| p.t < from).saturating_sub(1);
1093        pts[start..]
1094            .iter()
1095            .take_while(|p| p.t <= to)
1096            .flat_map(|p| [p.lo, p.hi, p.v])
1097    });
1098    let fit = render::autoscale(log, values);
1099    if whole && !log {
1100        fit.map(|(lo, hi)| axis::whole_range(lo, hi))
1101    } else {
1102        fit
1103    }
1104}

1106/// A value range `shown` one frame (`dt`) closer to `target`.
1107fn ease_values(shown: (f64, f64), target: (f64, f64), dt: f64) -> (f64, f64) {
1108    let (lo, hi) = target;
1109    let (mut a, mut b) = shown;
1110    let scale = hi - lo;
1111    window::ease(&mut a, lo, dt, scale);
1112    window::ease(&mut b, hi, dt, scale);
1113    if a.partial_cmp(&b) != Some(std::cmp::Ordering::Less) {
1114        (a, b) = (lo, hi);
1115    }
1116    (a, b)
1117}

1145/// A value for the readout and chips: at `prec` decimals (in exponent
1146/// form from a million up), else compactly.
1147fn format_value(v: f64, prec: Option<usize>, units: &str) -> String {
1148    if !v.is_finite() {
1149        return "—".into();
1150    }
1151    let text = match prec {
1152        // Large values as the axis labels them: 1.469e9, not 1468514835.908.
1153        Some(p) if v.abs() >= 1e6 => format!("{v:.*e}", p.min(12)),
1154        Some(p) => {
1155            let s = format!("{v:.*}", p.min(12));
1156            // -0.04 at one decimal reads "0.0", not "-0.0".
1157            match s.strip_prefix('-') {
1158                Some(rest) if rest.chars().all(|c| c == '0' || c == '.') => rest.to_string(),
1159                _ => s,
1160            }
1161        }
1162        None => {
1163            let a = v.abs();
1164            if a != 0.0 && !(1e-3..1e6).contains(&a) {
1165                format!("{v:.3e}")
1166            } else {
1167                let s = format!("{v:.4}");
1168                let s = s.trim_end_matches('0').trim_end_matches('.');
1169                if s == "-0" { "0".into() } else { s.to_string() }
1170            }
1171        }
1172    };
1173    if units.is_empty() {
1174        text
1175    } else {
1176        format!("{text} {units}")
1177    }
1178}

1246    /// A plot in a headless egui, fed events frame by frame.
1247    struct Harness {
1248        ctx: egui::Context,
1249        plot: Plot,
1250        t: f64,
1251    }
1252
1253    impl Harness {
1254        fn new(plot: Plot) -> Self {
1255            let mut h = Self {
1256                ctx: egui::Context::default(),
1257                plot,
1258                t: 0.0,
1259            };
1260            h.frame(vec![]);
1261            h
1262        }
1263
1264        fn frame(&mut self, events: Vec<egui::Event>) {
1265            let input = egui::RawInput {
1266                screen_rect: Some(Rect::from_min_size(Pos2::ZERO, Vec2::new(1000.0, 600.0))),
1267                time: Some(self.t),
1268                events,
1269                ..Default::default()
1270            };
1271            self.t += 0.05;
1272            let Self { ctx, plot, .. } = self;
1273            let mut out = ctx.run_ui(input, |ui| plot.ui(ui));
1274            out.textures_delta.clear();
1275        }
1276
1277        fn key(&mut self, key: egui::Key) {
1278            let ev = |pressed| egui::Event::Key {
1279                key,
1280                physical_key: None,
1281                pressed,
1282                repeat: false,
1283                modifiers: egui::Modifiers::NONE,
1284            };
1285            self.frame(vec![ev(true)]);
1286            self.frame(vec![ev(false)]);
1287        }
1288
1289        fn button(&mut self, at: Pos2, pressed: bool) {
1290            self.frame(vec![egui::Event::PointerButton {
1291                pos: at,
1292                button: egui::PointerButton::Primary,
1293                pressed,
1294                modifiers: egui::Modifiers::NONE,
1295            }]);
1296        }
1297
1298        /// Frames until `done`, for up to 10 s of real time.
1299        fn until(&mut self, mut done: impl FnMut(&Plot) -> bool) {
1300            let start = std::time::Instant::now();
1301            while !done(&self.plot) {
1302                assert!(start.elapsed().as_secs() < 10, "timed out");
1303                std::thread::sleep(std::time::Duration::from_millis(20));
1304                self.frame(vec![]);
1305            }
1306        }
1307    }
