72/// An archived + live plot of a set of PVs on one shared time axis.
73///
74/// Owns its traces, background fetches and live subscriptions; dropping it
75/// unsubscribes and abandons in-flight fetches. Its only shared state is
76/// the counter [`Plot::new`] salts ids with, so any number may exist at
77/// once.
78///
79/// Array (waveform) PVs are not plotted: their chip shows the latest array
80/// (`[936] 0, 0, 0, …`) and says "not plotted", and their history is not
81/// fetched. (Earlier builds plotted element 0 over time; a host such as
82/// bobview that wants a waveform's elements should show them itself.)

85    archiver: Option<ArchiverClient>,
86    /// Salt for every egui id this plot uses.

90    window: Window,

92    home_span: f64,
93    /// The range drawn, easing toward `window`'s.
94    shown: Option<(f64, f64)>,
95    y: YState,
96    /// The ranges drawn on the value axes after the first, by units,
97    /// easing toward their targets (always autoscaled).
98    axis_shown: Vec<(String, (f64, f64))>,
99    /// Each trace's value axis this frame, in trace order.
100    axis_of: Vec<usize>,
101    fetcher: Fetcher,
102    /// When (seconds since 1970) the user last panned or zoomed, for the
103    /// fetch debounce.

247    /// Change (or remove) the archiver: refetches every trace's history,
248    /// and checks that the new one can be reached.
249    pub fn set_archiver(&mut self, archiver: Option<ArchiverClient>) {
250        if let Some(a) = &archiver {
251            remember(&mut self.recent, a.base());
252        }
253        self.archiver = archiver;
254        self.monitor.reset(self.archiver.is_some());
255        // Searches in flight were for the old archiver: drop their answers
256        // and search again.
257        self.search.restart();
258        for t in &mut self.traces {
259            t.history = None;
260            // The generation carries on, so a late answer from the old
261            // archiver is still recognised as stale.
262            t.fetch = series::FetchState {
263                generation: t.fetch.generation + 1,
264                ..Default::default()
265            };
266        }
267    }

322    fn add(&mut self, name: &str) -> Option<&mut Trace> {
323        let name = name.trim();
324        let key = pv::PvName::parse(name).key();
325        if name.is_empty() || self.traces.iter().any(|t| t.pv.name().key() == key) {
326            return None;
327        }
328        // The first palette colour no other trace has.
329        let slot = (0..)
330            .find(|s| !self.traces.iter().any(|t| t.slot == *s))
331            .unwrap_or(0);
332        let mut trace = Trace::new(self.next_trace, name, self.pvs.subscribe(name));
333        trace.slot = slot;
334        self.next_trace += 1;
335        self.traces.push(trace);
336        self.traces.last_mut()
337    }

398    /// Show `view`, replacing the traces and range. With archiver
399    /// switching on ([`Plot::set_archiver_switching`]), a view that names
400    /// an archiver switches to it; otherwise, or for a view without one,
401    /// the archiver stays as it is.
402    pub fn set_view(&mut self, view: &View) {
403        if let Some(url) = view.archiver.as_ref().filter(|_| self.switch_archivers) {
404            match ArchiverClient::new(url) {
405                Ok(a) if self.archiver.as_ref() != Some(&a) => self.set_archiver(Some(a)),
406                Ok(_) => {}
407                Err(e) => log::warn!("view archiver {url}: {e}"),
408            }
409        }
410        self.traces.clear();
411        for tv in &view.pvs {
412            if let Some(t) = self.add(&tv.name) {
413                t.color = tv.color.clone();
414                t.visible = tv.visible;
415            }
416        }
417        self.set_range(&view.range);
418        self.y = YState {
419            autoscale: view.y.autoscale || view.y.min.is_none() || view.y.max.is_none(),
420            lo: view.y.min.unwrap_or(0.0),
421            hi: view.y.max.unwrap_or(1.0),
422            log: view.y.log,
423            shown: None,
424        };
425        self.fit_log_range();
426        self.snap = true;
427    }

430    pub fn set_range(&mut self, range: &view::Range) {
431        self.window = match range {
432            view::Range::Relative { secs } => Window::live(*secs),

435                    (Some(a), Some(b)) => Window::fixed(a, b),
436                    _ => {

439                    }
440                }
441            }
442        };
443        self.home_span = self.window.span;
444        self.snap = true;

494    fn show(&mut self, ui: &mut egui::Ui, menu: Option<&mut dyn FnMut(&mut egui::Ui)>) {
495        let ctx = ui.ctx().clone();
496        self.repaint = Some(fetch::Repaint {
497            ctx: ctx.clone(),
498            viewport: ctx.viewport_id(),
499        });
500        let now = now();
501        let time = ui.input(|i| i.time);
502        let dt = self.last_frame.map_or(0.0, |t| (time - t).clamp(0.0, 0.1));
503        self.last_frame = Some(time);
504        self.take_results(now);
505        self.pulse = self.pulsing(ui, time);
506        let whole = ui.available_rect_before_wrap();
507        self.keyboard(ui, whole, now);
508
509        bar::top_bar(self, ui, menu, now);
510        ui.add_space(6.0);
511
512        // The legend's height first, so the plot fills the rest; never
513        // smaller than MIN_SIZE (an auto-sized window offers little).
514        let width = ui.available_width().max(MIN_SIZE.x);
515        let mut chips = legend::layout(self, ui, width);
516        let gap = if chips.height > 0.0 { 6.0 } else { 0.0 };
517        let plot_h = (ui.available_height() - chips.height - gap).max(MIN_SIZE.y);
518        let (rect, resp) =
519            ui.allocate_exact_size(Vec2::new(width, plot_h), Sense::click_and_drag());
520        ui.add_space(gap);
521        legend::interact(self, ui, &mut chips);
522        #[cfg(test)]
523        {
524            self.chips_shown = chips.shown();
525        }
526
527        // Kinetic panning after a drag.
528        if self.glide != 0.0 && self.drag.is_none() {
529            self.window.pan(self.glide * dt, now);
530            self.glide *= (-5.0 * dt).exp();
531            self.changed_at = now;
532            self.snap = true;
533            if self.glide.abs() < self.window.span * 0.01 || self.window.live {
534                self.glide = 0.0;
535            }
536        }
537
538        let (from, to) = self.shown_range(now, dt);
539        let dark = ui.visuals().dark_mode;
540        let data: Vec<Vec<Pt>> = self
541            .traces
542            .iter()
543            .map(|t| {
544                if t.visible {
545                    t.points(from, to, now)
546                } else {
547                    Vec::new()
548                }
549            })
550            .collect();
551        let axes = self.value_axes(&data, from, to, dt, dark);
552
553        let font = egui::FontId::proportional(11.5);
554        let f = render::frame(ui.painter(), rect, (from, to), &axes, self.y.log, &font);
555        self.interact(ui, &resp, &f, now);
556        let ppp = ui.ctx().pixels_per_point();
557        self.schedule_fetches(now, f.data.width() * ppp);
558        self.schedule_probe(now);
559
560        // Paint. Bands are lighter the more traces have one; a picked-out
561        // trace keeps its band, the others lose theirs.
562        let binned: Vec<bool> = data.iter().map(|pts| pts.iter().any(Pt::is_bin)).collect();
563        let bands = (self.traces.iter().zip(&binned))
564            .filter(|(t, b)| t.visible && **b)
565            .count();
566        let lines: Vec<render::Line> = self
567            .paint_order()
568            .into_iter()
569            .map(|i| {
570                let t = &self.traces[i];
571                let color = self.color(t, dark);
572                let (color, width, band) = match self.highlight {
573                    Some(h) if h == t.id => (color, 2.5, render::band_alpha(1)),
574                    Some(_) => (color.gamma_multiply(0.25), 1.5, 0.0),
575                    None => (color, 1.5, render::band_alpha(bands)),
576                };
577                let band = if binned[i] { band } else { 0.0 };
578                render::Line {
579                    pts: &data[i],
580                    color,
581                    width,
582                    axis: self.axis_of.get(i).copied().unwrap_or(0),
583                    band,
584                }
585            })
586            .collect();
587        let painter = ui.painter_at(rect);
588        // The crosshair, first so its time tag can hide the label under it.
589        let hair = (!self.traces.is_empty() && !matches!(self.drag, Some(Drag::Pan | Drag::Axis)))
590            .then(|| hover::hairline(self, ui, &f, &data))
591            .flatten();
592        if self.traces.is_empty() {
593            painter.rect_filled(rect, 10.0, ui.visuals().extreme_bg_color);
594        } else {
595            let cover = hair.map(|(p, t)| hover::time_tag_rect(&painter, &f, p.x, t));
596            render::paint(&painter, &f, &lines, ui.visuals(), cover);
597            render::now_marker(&painter, &f, now, ui.visuals(), &font);
598        }
599        if let Some(Drag::Box(origin)) = self.drag
600            && let Some(p) = ui.input(|i| i.pointer.latest_pos())
601        {
602            let r = Rect::from_two_pos(origin, p).intersect(f.data);
603            let accent = ui.visuals().selection.bg_fill;
604            painter.rect(
605                r,
606                2.0,
607                accent.gamma_multiply(0.15),
608                egui::Stroke::new(1.0, accent),
609                egui::StrokeKind::Inside,
610            );
611        }
612        if self.traces.is_empty() {
613            hover::empty_state(&painter, &f, ui.visuals());
614        } else if let Some(hair) = hair {
615            hover::crosshair(self, ui, &f, &data, dark, hair);
616        }
617        hover::status(self, ui, &f);
618        legend::paint(self, ui, &chips, &data, dark);
619
620        // Keep moving: easing, gliding, and time passing while live.
621        let target = self.window.range(now);
622        let settled = self.shown.is_some_and(|(a, b)| {
623            let tol = self.window.span * 1e-3;
624            (a - target.0).abs() <= tol && (b - target.1).abs() <= tol
625        });
626        if self.glide != 0.0 || !settled || self.pending {
627            ctx.request_repaint();
628        } else if self.window.live {
629            let per_pixel = self.window.span / f.data.width().max(1.0) as f64;
630            ctx.request_repaint_after(std::time::Duration::from_secs_f64(
631                per_pixel.clamp(1.0 / 15.0, 1.0),
632            ));
633            if self.pulse {
634                ctx.request_repaint_after(std::time::Duration::from_millis(66));
635            }
636        }
637    }

686    /// The value axes to draw this frame, one per distinct unit among the
687    /// visible traces with data (at most [`MAX_AXES`]), and `axis_of`, each trace's.
688    /// The first takes the fixed range, log scale and value zooming; the
689    /// others autoscale to their own traces.
690    fn value_axes(
691        &mut self,
692        data: &[Vec<Pt>],
693        from: f64,
694        to: f64,
695        dt: f64,
696        dark: bool,
697    ) -> Vec<render::AxisSpec> {
698        // Only traces with something to show claim an axis.
699        let trace_units: Vec<Option<String>> = self
700            .traces
701            .iter()
702            .zip(data)
703            .map(|(t, pts)| (t.visible && pts.iter().any(|p| p.v.is_finite())).then(|| units(t)))
704            .collect();
705        let (names, axis_of) = group_units(&trace_units);
706        self.axis_of = axis_of;
707        let several = names.len() > 1;
708        self.axis_shown.retain(|(u, _)| names[1..].contains(u));
709        let mut specs = Vec::new();
710        for (a, name) in names.iter().enumerate() {
711            let members: Vec<usize> = (0..self.traces.len())
712                .filter(|&i| trace_units[i].is_some() && self.axis_of[i] == a)
713                .collect();
714            let pts = members.iter().map(|&i| &data[i]);
715            let whole =
716                !members.is_empty() && members.iter().all(|&i| whole_valued(&self.traces[i]));
717            let (lo, hi) = if a == 0 {
718                self.shown_values(pts, (from, to), whole, dt)
719            } else {
720                let fit = fit_values(self.y.log, whole, pts, from, to);
721                let i = match self.axis_shown.iter().position(|(u, _)| u == name) {
722                    Some(i) => i,
723                    None => {
724                        let first = fit.unwrap_or((0.0, 1.0));
725                        self.axis_shown.push((name.clone(), first));
726                        self.axis_shown.len() - 1
727                    }
728                };
729                let shown = &mut self.axis_shown[i].1;
730                *shown = ease_values(*shown, fit.unwrap_or(*shown), dt);
731                *shown
732            };
733            let title = axis_title(members.iter().filter_map(|&i| trace_units[i].as_deref()));
734            let color = match members.as_slice() {
735                [one] if several => Some(self.color(&self.traces[*one], dark)),
736                _ => None,
737            };
738            specs.push(render::AxisSpec {
739                lo,
740                hi,
741                title,
742                color,
743                whole,
744            });
745        }
746        specs
747    }

780    /// Take in finished fetches, searches and archiver probes, and what
781    /// they say of the archiver.
782    fn take_results(&mut self, now: f64) {
783        let mut back = false;
784        if let Some((epoch, result)) = self.fetcher.take_probed() {
785            back |= self.monitor.probed(epoch, result, now);
786        }
787        for done in self.fetcher.take() {
788            let Some(t) = self.traces.iter_mut().find(|t| t.id == done.trace) else {
789                continue;
790            };
791            if done.generation != t.fetch.generation {
792                continue;
793            }
794            t.fetch.in_flight = false;
795            match &done.result {
796                Ok(_) => {
797                    t.fetch.failures = 0;
798                    back |= self.monitor.ok(now);
799                }
800                Err(e) if status::archiver_fault(e, false) => self.monitor.suspect(e, now),
801                // Not archived, or something about this PV: the archiver
802                // answered, but that says nothing of the others.
803                Err(_) => {}
804            }
805            // The window really fetched: a raw-only trace's is shorter.
806            if let Some(r) = &mut t.fetch.requested
807                && !done.append
808            {
809                r.from = r.from.max(done.from);
810            }
811            if let Some(info) = &done.info {
812                t.fetch.raw_only = Some(!info.binnable());
813            }
814            match done.result {
815                Ok(series) if done.append => {
816                    let series = series.unwrap_or_default();
817                    let keep_from = t.fetch.requested.map_or(f64::NEG_INFINITY, |r| r.from);
818                    match &mut t.history {
819                        Some(h) if !h.binned && !series.binned => {
820                            h.append(series.samples, keep_from, done.to);
821                            t.fetch.error = None;
822                            // Grown past what `optimized_N` returns raw:
823                            // fetch it whole again, binned.
824                            if fetch::outgrown(t.fetch.requested.map(|r| r.op), h.samples.len()) {
825                                t.fetch.requested = None;
826                            }
827                        }
828                        // Not what the history holds: fetch it all again.
829                        _ => t.fetch.requested = None,
830                    }
831                }
832                Ok(series) => {
833                    let series = series.unwrap_or_default();
834                    let scalar = series.kind == ValueKind::Scalar;
835                    if series.samples.is_empty() {
836                        // Nothing to tell the type by.
837                    } else if !scalar && done.op != Operator::Raw {
838                        // Text: ask again, raw. A waveform is not plotted,
839                        // so what came is enough.
840                        t.fetch.raw_only = Some(true);
841                        if series.kind != ValueKind::Waveform {
842                            t.fetch.requested = None;
843                        }
844                    } else if t.fetch.raw_only.is_none() {
845                        t.fetch.raw_only = Some(!scalar);
846                    }
847                    let meta = if series.meta.name.is_empty() {
848                        t.history
849                            .as_ref()
850                            .map(|h| h.meta.clone())
851                            .unwrap_or_default()
852                    } else {
853                        series.meta
854                    };
855                    // An array's samples are not plotted: knowing it is
856                    // archived is enough, so they are not kept.
857                    let mut samples = series.samples;
858                    if series.kind == ValueKind::Waveform {
859                        samples = Vec::new();
860                    }
861                    t.history = Some(History {
862                        to: done.to,
863                        binned: series.binned,
864                        kind: series.kind,
865                        meta,
866                        samples,
867                    });
868                    t.fetch.error = None;
869                }
870                Err(e) if e.is_not_archived() => {
871                    t.fetch.not_archived = true;
872                    t.fetch.error = None;
873                    t.history = None;
874                }
875                Err(e) => {
876                    log::warn!("{}: {e}", t.name);
877                    t.fetch.archivers_fault = status::archiver_fault(&e, false);
878                    t.fetch.error = Some(e.to_string());
879                    t.fetch.failures += 1;
880                }
881            }
882        }
883        back |= bar::take_found(self, now);
884        if back {
885            // The archiver is back: ask again now for what failed because
886            // it was down (not for what fails for reasons of its own,
887            // which keeps backing off).
888            for t in &mut self.traces {
889                if t.fetch.error.is_some() && t.fetch.archivers_fault && !t.fetch.in_flight {
890                    t.fetch.failures = 0;
891                    t.fetch.requested = None;
892                }
893            }
894        }
895    }

897    /// Check the archiver when one was just set, and every
898    /// `status::REPROBE` seconds while it cannot be reached.
899    fn schedule_probe(&mut self, now: f64) {
900        let Some(client) = &self.archiver else {
901            return;
902        };
903        if self.monitor.should_probe(now) {
904            self.monitor.sent(now);
905            self.fetcher
906                .probe(&self.pvs, client, self.monitor.epoch, self.repaint.clone());
907        }
908        if self.monitor.status.state == ArchiverState::Unreachable {
909            // Wake for the next probe even in an idle window.
910            let next = self.monitor.probed_at + self.monitor.reprobe - now;
911            if let Some(r) = &self.repaint {
912                r.ctx.request_repaint_after_for(
913                    std::time::Duration::from_secs_f64(next.clamp(0.05, status::REPROBE)),
914                    r.viewport,
915                );
916            }
917        }
918    }

920    /// Start the fetches the visible window needs, debounced while the
921    /// user pans and zooms.
922    fn schedule_fetches(&mut self, now: f64, width_px: f32) {
923        self.pending = false;
924        let Some(client) = self.archiver.clone() else {
925            return;
926        };
927        let (from, to) = self.window.range(now);
928        let settled = now - self.changed_at >= DEBOUNCE;
929        let live = self.window.live;
930        for t in &mut self.traces {
931            let Some(name) = t.archive_name() else {
932                continue;
933            };
934            // One request per trace at a time: when it is done, the next
935            // frame asks for wherever the window is then.
936            if t.fetch.not_archived || t.fetch.in_flight {
937                continue;
938            }
939            // An array is not plotted: once it is known to be archived,
940            // there is nothing more to fetch.
941            if t.history
942                .as_ref()
943                .is_some_and(|h| h.kind == ValueKind::Waveform)
944            {
945                continue;
946            }
947            let raw = t.fetch.raw_only.unwrap_or(false);
948            let from = fetch::needed_from(from, to, raw);
949            // A live refresh of raw history: only from its last sample on.
950            let mut tail = None;
951            let need = match t.fetch.requested {
952                None => true,
953                Some(r) => {
954                    let binned =
955                        r.op != Operator::Raw && t.history.as_ref().is_none_or(|h| h.binned);
956                    let covered = fetch::covers((r.from, r.to), binned, from, to);
957                    // After an error: again once the backoff has passed, or
958                    // after the base delay for a different window.
959                    let wait = if covered {
960                        fetch::backoff(RETRY, t.fetch.failures)
961                    } else {
962                        RETRY
963                    };
964                    if t.fetch.error.is_some() {
965                        now - r.at > wait
966                    } else if !covered {
967                        true
968                    } else if live
969                        && now - r.at
970                            > fetch::refresh_after(LIVE_REFRESH, (r.from, r.to), r.op, binned)
971                    {
972                        tail = t
973                            .history
974                            .as_ref()
975                            .filter(|h| !h.binned)
976                            .and_then(|h| h.samples.last())
977                            .map(|s| s.t)
978                            .filter(|t| *t > r.from)
979                            .map(|t| (t, r));
980                        true
981                    } else {
982                        false
983                    }
984                }
985            };
986            if !need {
987                continue;
988            }
989            if t.fetch.requested.is_some() && !settled {
990                self.pending = true;
991                continue;
992            }
993            let (a, b) = fetch::padded(from, to);
994            let op = fetch::operator(width_px, raw);
995            let raw_from = fetch::padded(fetch::needed_from(from, to, true), to).0;
996            t.fetch.generation += 1;
997            t.fetch.in_flight = true;
998            let job = match tail {
999                // The history keeps what is still in the padded window.
1000                Some((last, r)) => {
1001                    t.fetch.requested = Some(series::Requested {
1002                        from: r.from.max(a).min(last),
1003                        to: b,
1004                        op: r.op,
1005                        at: now,
1006                    });
1007                    fetch::Job {
1008                        trace: t.id,
1009                        generation: t.fetch.generation,
1010                        pv: name,
1011                        from: last,
1012                        to: b,
1013                        op: Operator::Raw,
1014                        raw_from: last,
1015                        probe: false,
1016                        append: true,
1017                    }
1018                }
1019                None => {
1020                    t.fetch.requested = Some(series::Requested {
1021                        from: a,
1022                        to: b,
1023                        op,
1024                        at: now,
1025                    });
1026                    fetch::Job {
1027                        trace: t.id,
1028                        generation: t.fetch.generation,
1029                        pv: name,
1030                        from: a,
1031                        to: b,
1032                        op,
1033                        raw_from,
1034                        probe: t.fetch.raw_only.is_none(),
1035                        append: false,
1036                    }
1037                }
1038            };
1039            self.fetcher
1040                .fetch(&self.pvs, &client, job, self.repaint.clone());
1041        }
1042    }

1044    /// Pan, zoom, box zoom and reset on the plot.
1045    fn interact(&mut self, ui: &egui::Ui, resp: &egui::Response, f: &render::Frame, now: f64) {
1046        let data = f.data;
1047        let span = self.window.span;
1048        let per_px = span / data.width().max(1.0) as f64;
1049        if resp.clicked() || resp.drag_started() {
1050            self.active = true;
1051        } else if ui.input(|i| i.pointer.any_pressed()) && !resp.contains_pointer() {
1052            self.active = false;
1053        }
1054
1055        if resp.drag_started() {
1056            let origin = ui
1057                .input(|i| i.pointer.press_origin())
1058                .unwrap_or(data.center());
1059            let shift = ui.input(|i| i.modifiers.shift);
1060            self.glide = 0.0;
1061            self.drag = Some(
1062                if shift || resp.dragged_by(egui::PointerButton::Secondary) {
1063                    Drag::Box(origin)
1064                } else if f.value_axis().contains(origin) {
1065                    Drag::Axis
1066                } else {
1067                    Drag::Pan
1068                },
1069            );
1070        }
1071        if resp.dragged() {
1072            let d = resp.drag_delta();
1073            match self.drag {
1074                Some(Drag::Pan) => {
1075                    self.window.pan(-d.x as f64 * per_px, now);
1076                    if !self.y.autoscale && d.y != 0.0 {
1077                        let (lo, hi) = self.y.shown.unwrap_or((self.y.lo, self.y.hi));
1078                        let frac = -(d.y / data.height().max(1.0)) as f64;
1079                        (self.y.lo, self.y.hi) = window::pan_values(lo, hi, frac, self.y.log);
1080                        self.y.shown = Some((self.y.lo, self.y.hi));
1081                    }
1082                    self.snap = true;
1083                    self.changed_at = now;
1084                }
1085                Some(Drag::Axis) => {
1086                    let factor = (d.y as f64 * 0.01).exp();
1087                    self.zoom_values(factor, None, f);
1088                    self.y.shown = Some((self.y.lo, self.y.hi));
1089                }
1090                _ => {}
1091            }
1092        }
1093        if resp.drag_stopped() {
1094            match self.drag.take() {
1095                Some(Drag::Pan) => {
1096                    let v = ui.input(|i| i.pointer.velocity()).x as f64;
1097                    if v.abs() > 150.0 {
1098                        self.glide = -v * per_px;
1099                    }
1100                }
1101                Some(Drag::Box(origin)) => {
1102                    if let Some(p) = ui.input(|i| i.pointer.latest_pos()) {
1103                        let r = Rect::from_two_pos(origin, p).intersect(data);
1104                        if r.width() > 6.0 {
1105                            self.window.set(f.time_at(r.left()), f.time_at(r.right()));
1106                        }
1107                        if r.height() > 6.0 {
1108                            self.y.autoscale = false;
1109                            (self.y.lo, self.y.hi) = (f.value_at(r.bottom()), f.value_at(r.top()));
1110                        }
1111                    }
1112                }
1113                _ => {}
1114            }
1115        }
1116
1117        if resp.contains_pointer() {
1118            let (scroll, zoom, mods, pointer) = ui.input(|i| {
1119                (
1120                    i.smooth_scroll_delta,
1121                    i.zoom_delta(),
1122                    i.modifiers,
1123                    i.pointer.hover_pos(),
1124                )
1125            });
1126            let anchor = pointer.map_or(1.0, |p| ((p.x - data.left()) / data.width()) as f64);
1127            let at_value = pointer.map(|p| f.value_at(p.y));
1128            if mods.shift && (scroll.x != 0.0 || scroll.y != 0.0) {
1129                // Shift+wheel arrives as horizontal scrolling.
1130                let amount = (scroll.x + scroll.y) as f64;
1131                self.zoom_values((-amount / 300.0).exp(), at_value, f);
1132            } else if zoom != 1.0 {
1133                if mods.ctrl || mods.command {
1134                    self.zoom_values(1.0 / zoom as f64, at_value, f);
1135                } else {
1136                    self.window.zoom(1.0 / zoom as f64, anchor, now);
1137                    self.changed_at = now;
1138                }
1139            } else {
1140                if scroll.y != 0.0 {
1141                    self.window
1142                        .zoom((-scroll.y as f64 / 300.0).exp(), anchor, now);
1143                    self.changed_at = now;
1144                }
1145                if scroll.x != 0.0 {
1146                    self.window.pan(-scroll.x as f64 * per_px, now);
1147                    self.changed_at = now;
1148                    self.snap = true;
1149                }
1150            }
1151            if scroll != Vec2::ZERO {
1152                // Not for a scroll area around the plot too.
1153                ui.ctx().input_mut(|i| i.smooth_scroll_delta = Vec2::ZERO);
1154            }
1155        }
1156        if resp.double_clicked() {
1157            self.reset();
1158        }
1159    }

1161    /// Zoom values by `factor` about `at` (the middle when `None`), leaving
1162    /// autoscale.
1163    fn zoom_values(&mut self, factor: f64, at: Option<f64>, f: &render::Frame) {
1164        let (lo, hi) = self.y.shown.unwrap_or((f.ys.lo, f.ys.hi));
1165        let mid = if self.y.log && lo > 0.0 {
1166            (lo * hi).sqrt()
1167        } else {
1168            (lo + hi) / 2.0
1169        };
1170        let (l, h) = window::zoom_values(lo, hi, factor, at.unwrap_or(mid), self.y.log);
1171        if l.is_finite() && h.is_finite() && l < h {
1172            self.y.autoscale = false;
1173            (self.y.lo, self.y.hi) = (l, h);
1174        }
1175    }

1250/// Group traces by units (`None`: hidden) into value axes, in order of
1251/// first appearance: the axes' units, and each trace's axis (0 when
1252/// hidden). Units beyond [`MAX_AXES`] share the last axis. With no visible
1253/// traces there is one axis, without units.
1254fn group_units(units: &[Option<String>]) -> (Vec<String>, Vec<usize>) {
1255    let mut names: Vec<String> = Vec::new();
1256    let of = units
1257        .iter()
1258        .map(|u| {
1259            let Some(u) = u else { return 0 };
1260            match names.iter().position(|n| n == u) {
1261                Some(i) => i,
1262                None if names.len() < MAX_AXES => {
1263                    names.push(u.clone());
1264                    names.len() - 1
1265                }
1266                None => MAX_AXES - 1,
1267            }
1268        })
1269        .collect();
1270    if names.is_empty() {
1271        names.push(String::new());
1272    }
1273    (names, of)
1274}

1299/// The autoscaled range for `data` between `from` and `to` (at least two
1300/// wide, for `whole` numbers on a linear scale).
1301fn fit_values<'a>(
1302    log: bool,
1303    whole: bool,
1304    data: impl Iterator<Item = &'a Vec<Pt>>,
1305    from: f64,
1306    to: f64,
1307) -> Option<(f64, f64)> {
1308    let values = data.flat_map(|pts| {
1309        let start = pts.partition_point(|p| p.t < from).saturating_sub(1);
1310        pts[start..]
1311            .iter()
1312            .take_while(|p| p.t <= to)
1313            .flat_map(|p| [p.lo, p.hi, p.v])
1314    });
1315    let fit = render::autoscale(log, values);
1316    if whole && !log {
1317        fit.map(|(lo, hi)| axis::whole_range(lo, hi))
1318    } else {
1319        fit
1320    }
1321}

1323/// A value range `shown` one frame (`dt`) closer to `target`.
1324fn ease_values(shown: (f64, f64), target: (f64, f64), dt: f64) -> (f64, f64) {
1325    let (lo, hi) = target;
1326    let (mut a, mut b) = shown;
1327    let scale = hi - lo;
1328    window::ease(&mut a, lo, dt, scale);
1329    window::ease(&mut b, hi, dt, scale);
1330    if a.partial_cmp(&b) != Some(std::cmp::Ordering::Less) {
1331        (a, b) = (lo, hi);
1332    }
1333    (a, b)
1334}

1362/// A value for the readout and chips: at `prec` decimals (in exponent
1363/// form from a million up), else compactly.
1364fn format_value(v: f64, prec: Option<usize>, units: &str) -> String {
1365    if !v.is_finite() {
1366        return "—".into();
1367    }
1368    let text = match prec {
1369        // Large values as the axis labels them: 1.469e9, not 1468514835.908.
1370        Some(p) if v.abs() >= 1e6 => format!("{v:.*e}", p.min(12)),
1371        Some(p) => {
1372            let s = format!("{v:.*}", p.min(12));
1373            // -0.04 at one decimal reads "0.0", not "-0.0".
1374            match s.strip_prefix('-') {
1375                Some(rest) if rest.chars().all(|c| c == '0' || c == '.') => rest.to_string(),
1376                _ => s,
1377            }
1378        }
1379        None => {
1380            let a = v.abs();
1381            if a != 0.0 && !(1e-3..1e6).contains(&a) {
1382                format!("{v:.3e}")
1383            } else {
1384                let s = format!("{v:.4}");
1385                let s = s.trim_end_matches('0').trim_end_matches('.');
1386                if s == "-0" { "0".into() } else { s.to_string() }
1387            }
1388        }
1389    };
1390    if units.is_empty() {
1391        text
1392    } else {
1393        format!("{text} {units}")
1394    }
1395}
