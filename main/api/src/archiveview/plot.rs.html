183#[derive(Clone, Debug)]
184struct ValueAxis {
185    id: u32,
186    /// The name the user gave it; else it names itself.
187    name: Option<String>,
188    autoscale: bool,
189    /// The fixed range when not autoscaling.
190    lo: f64,
191    hi: f64,
192    log: bool,
193    /// The range drawn, easing toward the target.
194    shown: Option<(f64, f64)>,
195    /// It has had a trace, so it goes when its last one leaves. An axis
196    /// from a view that no trace is on yet waits for a trace to settle:
197    /// the one in `reserved`, else any.
198    used: bool,
199    reserved: Option<u64>,

203    fn new(id: u32) -> Self {
204        Self::from_view(id, &YAxis::default())
205    }

207    fn from_view(id: u32, y: &YAxis) -> Self {
208        Self {
209            id,
210            name: y.name.clone(),
211            autoscale: y.autoscale || y.min.is_none() || y.max.is_none(),
212            lo: y.min.unwrap_or(0.0),
213            hi: y.max.unwrap_or(1.0),
214            log: y.log,
215            shown: None,
216            used: false,
217            reserved: None,
218        }
219    }
220
221    fn view(&self) -> YAxis {
222        YAxis {
223            autoscale: self.autoscale,
224            min: (!self.autoscale).then_some(self.lo),
225            max: (!self.autoscale).then_some(self.hi),
226            log: self.log,
227            name: self.name.clone(),
228        }
229    }

231    /// On a log scale a fixed range must be above zero: back to
232    /// autoscaling if not.
233    fn fit_log_range(&mut self) {
234        if self.log && !self.autoscale && self.lo <= 0.0 {
235            self.autoscale = true;
236            self.shown = None;
237        }
238    }
239
240    /// Zoom by `factor` about `at` (the middle when `None`), from the
241    /// range `shown` (else its own), leaving autoscale.
242    fn zoom(&mut self, factor: f64, at: Option<f64>, shown: (f64, f64)) {
243        let (lo, hi) = self.shown.unwrap_or(shown);
244        let mid = if self.log && lo > 0.0 {
245            (lo * hi).sqrt()
246        } else {
247            (lo + hi) / 2.0
248        };
249        let (l, h) = window::zoom_values(lo, hi, factor, at.unwrap_or(mid), self.log);
250        if l.is_finite() && h.is_finite() && l < h {
251            self.autoscale = false;
252            (self.lo, self.hi) = (l, h);
253        }
254    }
255}

341    /// Change (or remove) the archiver: refetches every trace's history,
342    /// and checks that the new one can be reached.
343    pub fn set_archiver(&mut self, archiver: Option<ArchiverClient>) {
344        if let Some(a) = &archiver {
345            remember(&mut self.recent, a.base());
346        }
347        self.archiver = archiver;
348        self.monitor.reset(self.archiver.is_some());
349        // Searches in flight were for the old archiver: drop their answers
350        // and search again.
351        self.search.restart();
352        for t in &mut self.traces {
353            t.history = None;
354            // The generation carries on, so a late answer from the old
355            // archiver is still recognised as stale.
356            t.fetch = series::FetchState {
357                generation: t.fetch.generation + 1,
358                ..Default::default()
359            };
360        }
361    }

416    fn add(&mut self, name: &str) -> Option<&mut Trace> {
417        let name = name.trim();
418        let key = pv::PvName::parse(name).key();
419        if name.is_empty() || self.traces.iter().any(|t| t.pv.name().key() == key) {
420            return None;
421        }
422        // The first palette colour no other trace has.
423        let slot = (0..)
424            .find(|s| !self.traces.iter().any(|t| t.slot == *s))
425            .unwrap_or(0);
426        let mut trace = Trace::new(self.next_trace, name, self.pvs.subscribe(name));
427        trace.slot = slot;
428        self.next_trace += 1;
429        self.traces.push(trace);
430        self.traces.last_mut()
431    }

458    /// The current view (traces, range, value axes, archiver), to save.
459    pub fn view(&self) -> View {
460        let range = if self.window.live {
461            view::Range::Relative {
462                secs: self.window.span,
463            }
464        } else {
465            let (from, to) = self.window.range(now());
466            view::Range::Absolute {
467                start: view::format_time(from),
468                end: view::format_time(to),
469            }
470        };
471        // Only axes with traces, or kept for a PV yet to settle (pruned
472        // once that PV is gone): no stale fixed range comes back for
473        // whichever PV would take it.
474        let kept: Vec<&ValueAxis> = (self.axes.iter())
475            .filter(|a| self.has_traces(a) || a.reserved.is_some())
476            .collect();
477        let mut axes = kept.iter().map(|a| a.view());
478        View {
479            version: view::VERSION,
480            pvs: self
481                .traces
482                .iter()
483                .map(|t| TraceView {
484                    name: t.name.clone(),
485                    color: t.color.clone(),
486                    visible: t.visible,
487                    axis: t.axis.and_then(|id| kept.iter().position(|a| a.id == id)),
488                })
489                .collect(),
490            range,
491            y: axes.next().unwrap_or_default(),
492            axes: axes.collect(),
493            archiver: self.archiver.as_ref().map(|a| a.base().to_string()),
494        }
495    }

497    /// Show `view`, replacing the traces, range and value axes. Traces
498    /// the view puts on an axis stay there; others join one when their
499    /// data arrives. With archiver switching on
500    /// ([`Plot::set_archiver_switching`]), a view that names an archiver
501    /// switches to it; otherwise, or for a view without one, the archiver
502    /// stays as it is.
503    pub fn set_view(&mut self, view: &View) {
504        if let Some(url) = view.archiver.as_ref().filter(|_| self.switch_archivers) {
505            match ArchiverClient::new(url) {
506                Ok(a) if self.archiver.as_ref() != Some(&a) => self.set_archiver(Some(a)),
507                Ok(_) => {}
508                Err(e) => log::warn!("view archiver {url}: {e}"),
509            }
510        }
511        self.traces.clear();
512        self.axes = view
513            .value_axes()
514            .map(|y| {
515                let mut a = ValueAxis::from_view(self.next_axis, y);
516                self.next_axis += 1;
517                a.fit_log_range();
518                a
519            })
520            .collect();
521        self.axis_highlight = None;
522        self.renaming = None;
523        // A file from before there were several axes: its one axis is for
524        // its first visible PV, and its log scale for every axis.
525        let old = view.axes.is_empty() && view.pvs.iter().all(|t| t.axis.is_none());
526        self.default_log = old && view.y.log;
527        for tv in &view.pvs {
528            let axis = tv.axis.and_then(|i| self.axes.get_mut(i)).map(|a| {
529                a.used = true;
530                a.id
531            });
532            if let Some(t) = self.add(&tv.name) {
533                t.color = tv.color.clone();
534                t.visible = tv.visible;
535                t.axis = axis;
536            }
537        }
538        if old
539            && let Some(first) = self.traces.iter().find(|t| t.visible)
540            && let Some(y) = self.axes.first_mut()
541        {
542            y.reserved = Some(first.id);
543        }
544        self.set_range(&view.range);
545        self.snap = true;
546    }

548    /// Show `range`.
549    pub fn set_range(&mut self, range: &view::Range) {
550        self.window = match range {
551            view::Range::Relative { secs } => Window::live(*secs),
552            view::Range::Absolute { start, end } => {
553                match (view::parse_time(start), view::parse_time(end)) {
554                    (Some(a), Some(b)) => Window::fixed(a, b),
555                    _ => {
556                        log::warn!("bad view range {start}..{end}: showing the last hour");
557                        Window::live(3600.0)
558                    }
559                }
560            }
561        };
562        self.home_span = self.window.span;
563        self.snap = true;
564    }

586    /// True while any archive fetch or search is in flight, or one is due,
587    /// or a trace with data waits to settle on an axis (snapshots wait for
588    /// this).
589    pub fn is_loading(&self) -> bool {
590        self.fetcher.pending()
591            || self
592                .traces
593                .iter()
594                .any(|t| t.visible && !t.is_array() && t.axis.is_none() && t.data_since.is_some())
595            || self.pending
596            || (self.archiver.is_some() && self.monitor.due)
597            || bar::search_due(self)
598            || (self.archiver.is_some()
599                && self.traces.iter().any(|t| {
600                    t.archive_name().is_some()
601                        && !t.fetch.not_archived
602                        && t.fetch.requested.is_none()
603                }))
604    }

772            if self.pulse {

775        }

870    /// Settle new traces on axes, once each, in the legend's order, so
871    /// the outcome does not depend on which archive answer comes first.
872    ///
873    /// A trace is ready when it has data on screen and its archived
874    /// history has come (live data alone: after [`LIVE_WAIT`], so one
875    /// first sample does not decide). It waits while an earlier trace may
876    /// still get data: until that one settles, or turns out to have none,
877    /// or [`SETTLE_WAIT`] passes. Then the axes left with no trace go.
878    fn settle_axes(&mut self, data: &[Vec<Pt>], now: f64) {
879        // Settled traces shown since: their stats, for scoring.
880        for i in 0..self.traces.len() {
881            if self.traces[i].axis.is_some() && self.traces[i].stats.is_none() {
882                self.traces[i].stats = self.trace_stats(i, data);
883            }
884        }
885        let mut blocked = false;
886        for i in 0..self.traces.len() {
887            let t = &self.traces[i];
888            // Arrays are not plotted: they take no axis, nor hold others up.
889            if t.axis.is_some() || !t.visible || t.is_array() {
890                continue;
891            }
892            let seen = *self.traces[i].seen_at.get_or_insert(now);
893            let late = now - seen >= SETTLE_WAIT;
894            let stats = self.trace_stats(i, data);
895            if stats.is_some() {
896                self.traces[i].data_since.get_or_insert(now);
897            }
898            let t = &self.traces[i];
899            let pending = self.awaits_history(t);
900            let archived = t.history.as_ref().is_some_and(|h| !h.samples.is_empty());
901            let live = !pending && t.data_since.is_some_and(|s| now - s >= LIVE_WAIT);
902            // No units from the archive: its live units may yet come, and
903            // should decide (within the wait).
904            let units_due = units(t).is_empty() && t.archive_name().is_some() && !t.pv.connected();
905            if stats.is_none() || !(late || ((archived || live) && !units_due)) {
906                // It may yet get data, or more of it: those after it wait.
907                blocked |= !late && (pending || stats.is_some());
908                if late && stats.is_none() {
909                    // Nothing came in time: an axis kept for it (from an
910                    // old view) is not kept any longer.
911                    let id = self.traces[i].id;
912                    for a in &mut self.axes {
913                        if a.reserved == Some(id) {
914                            a.reserved = None;
915                        }
916                    }
917                }
918                continue;
919            }
920            if blocked {
921                continue;
922            }
923            let stats = stats.expect("ready");
924            let (known, unknown) = self.axis_stats();
925            let id = match axes::choose(&stats, &known, unknown) {
926                Choice::Join(id) => id,
927                Choice::New => self.new_axis(&stats, self.traces[i].id),
928            };
929            self.traces[i].stats = Some(stats);
930            self.put_on(i, id);
931        }
932        self.prune_axes();
933    }

965    /// its own with `None`, by hand. Alone on its axis already, it stays
966    /// there (keeping the axis' range, scale and name).
967    fn move_trace(&mut self, trace: u64, to: Option<u32>) {
968        let Some(i) = self.traces.iter().position(|t| t.id == trace) else {
969            return;
970        };
971        let on = self.traces[i].axis;
972        let id = match to {
973            Some(id) if self.axes.iter().any(|a| a.id == id) => id,
974            Some(_) => return,
975            None if on.is_some() && self.traces.iter().filter(|t| t.axis == on).count() == 1 => {
976                return;
977            }
978            None => {
979                // Fresh, not one waiting for a view's trace.
980                let a = ValueAxis::new(self.next_axis);
981                self.next_axis += 1;
982                let id = a.id;
983                self.axes.push(a);
984                id
985            }
986        };
987        self.put_on(i, id);
988        self.prune_axes();
989    }

991    /// Drop the axes with no trace: those whose last trace has left, and
992    /// one from a view kept for a trace that is gone, is an array, or
993    /// got no data. Keep the others in the legend's order (of each one's
994    /// first trace), so the first chip's axis is always the first.
995    fn prune_axes(&mut self) {
996        let traces = &self.traces;
997        self.axes.retain(|a| {
998            traces.iter().any(|t| t.axis == Some(a.id))
999                || (!a.used
1000                    && a.reserved
1001                        .is_some_and(|r| traces.iter().any(|t| t.id == r && !t.is_array())))
1002        });
1003        self.axes.sort_by_key(|a| {
1004            traces
1005                .iter()
1006                .position(|t| t.axis == Some(a.id))
1007                .unwrap_or(usize::MAX)
1008        });
1009        if let Some(h) = self.axis_highlight
1010            && !self.axes.iter().any(|a| a.id == h)
1011        {
1012            self.axis_highlight = None;
1013        }
1014    }

1136    /// Take in finished fetches, searches and archiver probes, and what
1137    /// they say of the archiver.
1138    fn take_results(&mut self, now: f64) {
1139        let mut back = false;
1140        if let Some((epoch, result)) = self.fetcher.take_probed() {
1141            back |= self.monitor.probed(epoch, result, now);
1142        }
1143        for done in self.fetcher.take() {
1144            let Some(t) = self.traces.iter_mut().find(|t| t.id == done.trace) else {
1145                continue;
1146            };
1147            if done.generation != t.fetch.generation {
1148                continue;
1149            }
1150            t.fetch.in_flight = false;
1151            match &done.result {
1152                Ok(_) => {
1153                    t.fetch.failures = 0;
1154                    back |= self.monitor.ok(now);
1155                }
1156                Err(e) if status::archiver_fault(e, false) => self.monitor.suspect(e, now),
1157                // Not archived, or something about this PV: the archiver
1158                // answered, but that says nothing of the others.
1159                Err(_) => {}
1160            }
1161            // The window really fetched: a raw-only trace's is shorter.
1162            if let Some(r) = &mut t.fetch.requested
1163                && !done.append
1164            {
1165                r.from = r.from.max(done.from);
1166            }
1167            if let Some(info) = &done.info {
1168                t.fetch.raw_only = Some(!info.binnable());
1169            }
1170            match done.result {
1171                Ok(series) if done.append => {
1172                    let series = series.unwrap_or_default();
1173                    let keep_from = t.fetch.requested.map_or(f64::NEG_INFINITY, |r| r.from);
1174                    match &mut t.history {
1175                        Some(h) if !h.binned && !series.binned => {
1176                            h.append(series.samples, keep_from, done.to);
1177                            t.fetch.error = None;
1178                            // Grown past what `optimized_N` returns raw:
1179                            // fetch it whole again, binned.
1180                            if fetch::outgrown(t.fetch.requested.map(|r| r.op), h.samples.len()) {
1181                                t.fetch.requested = None;
1182                            }
1183                        }
1184                        // Not what the history holds: fetch it all again.
1185                        _ => t.fetch.requested = None,
1186                    }
1187                }
1188                Ok(series) => {
1189                    let series = series.unwrap_or_default();
1190                    let scalar = series.kind == ValueKind::Scalar;
1191                    if series.samples.is_empty() {
1192                        // Nothing to tell the type by.
1193                    } else if !scalar && done.op != Operator::Raw {
1194                        // Text: ask again, raw. A waveform is not plotted,
1195                        // so what came is enough.
1196                        t.fetch.raw_only = Some(true);
1197                        if series.kind != ValueKind::Waveform {
1198                            t.fetch.requested = None;
1199                        }
1200                    } else if t.fetch.raw_only.is_none() {
1201                        t.fetch.raw_only = Some(!scalar);
1202                    }
1203                    let meta = if series.meta.name.is_empty() {
1204                        t.history
1205                            .as_ref()
1206                            .map(|h| h.meta.clone())
1207                            .unwrap_or_default()
1208                    } else {
1209                        series.meta
1210                    };
1211                    // An array's samples are not plotted: knowing it is
1212                    // archived is enough, so they are not kept.
1213                    let mut samples = series.samples;
1214                    if series.kind == ValueKind::Waveform {
1215                        samples = Vec::new();
1216                    }
1217                    t.history = Some(History {
1218                        to: done.to,
1219                        binned: series.binned,
1220                        kind: series.kind,
1221                        meta,
1222                        samples,
1223                    });
1224                    t.fetch.error = None;
1225                }
1226                Err(e) if e.is_not_archived() => {
1227                    t.fetch.not_archived = true;
1228                    t.fetch.error = None;
1229                    t.history = None;
1230                }
1231                Err(e) => {
1232                    log::warn!("{}: {e}", t.name);
1233                    t.fetch.archivers_fault = status::archiver_fault(&e, false);
1234                    t.fetch.error = Some(e.to_string());
1235                    t.fetch.failures += 1;
1236                }
1237            }
1238        }
1239        back |= bar::take_found(self, now);
1240        if back {
1241            // The archiver is back: ask again now for what failed because
1242            // it was down (not for what fails for reasons of its own,
1243            // which keeps backing off).
1244            for t in &mut self.traces {
1245                if t.fetch.error.is_some() && t.fetch.archivers_fault && !t.fetch.in_flight {
1246                    t.fetch.failures = 0;
1247                    t.fetch.requested = None;
1248                }
1249            }
1250        }
1251    }

1253    /// Check the archiver when one was just set, and every
1254    /// `status::REPROBE` seconds while it cannot be reached.
1255    fn schedule_probe(&mut self, now: f64) {
1256        let Some(client) = &self.archiver else {
1257            return;
1258        };
1259        if self.monitor.should_probe(now) {
1260            self.monitor.sent(now);
1261            self.fetcher
1262                .probe(&self.pvs, client, self.monitor.epoch, self.repaint.clone());
1263        }
1264        if self.monitor.status.state == ArchiverState::Unreachable {
1265            // Wake for the next probe even in an idle window.
1266            let next = self.monitor.probed_at + self.monitor.reprobe - now;
1267            if let Some(r) = &self.repaint {
1268                r.ctx.request_repaint_after_for(
1269                    std::time::Duration::from_secs_f64(next.clamp(0.05, status::REPROBE)),
1270                    r.viewport,
1271                );
1272            }
1273        }
1274    }

1276    /// Start the fetches the visible window needs, debounced while the
1277    /// user pans and zooms.
1278    fn schedule_fetches(&mut self, now: f64, width_px: f32) {
1279        self.pending = false;
1280        let Some(client) = self.archiver.clone() else {
1281            return;
1282        };
1283        let (from, to) = self.window.range(now);
1284        let settled = now - self.changed_at >= DEBOUNCE;
1285        let live = self.window.live;
1286        for t in &mut self.traces {
1287            let Some(name) = t.archive_name() else {
1288                continue;
1289            };
1290            // One request per trace at a time: when it is done, the next
1291            // frame asks for wherever the window is then.
1292            if t.fetch.not_archived || t.fetch.in_flight {
1293                continue;
1294            }
1295            // An array is not plotted: once it is known to be archived,
1296            // there is nothing more to fetch.
1297            if t.history
1298                .as_ref()
1299                .is_some_and(|h| h.kind == ValueKind::Waveform)
1300            {
1301                continue;
1302            }
1303            let raw = t.fetch.raw_only.unwrap_or(false);
1304            let from = fetch::needed_from(from, to, raw);
1305            // A live refresh of raw history: only from its last sample on.
1306            let mut tail = None;
1307            let need = match t.fetch.requested {
1308                None => true,
1309                Some(r) => {
1310                    let binned =
1311                        r.op != Operator::Raw && t.history.as_ref().is_none_or(|h| h.binned);
1312                    let covered = fetch::covers((r.from, r.to), binned, from, to);
1313                    // After an error: again once the backoff has passed, or
1314                    // after the base delay for a different window.
1315                    let wait = if covered {
1316                        fetch::backoff(RETRY, t.fetch.failures)
1317                    } else {
1318                        RETRY
1319                    };
1320                    if t.fetch.error.is_some() {
1321                        now - r.at > wait
1322                    } else if !covered {
1323                        true
1324                    } else if live
1325                        && now - r.at
1326                            > fetch::refresh_after(LIVE_REFRESH, (r.from, r.to), r.op, binned)
1327                    {
1328                        tail = t
1329                            .history
1330                            .as_ref()
1331                            .filter(|h| !h.binned)
1332                            .and_then(|h| h.samples.last())
1333                            .map(|s| s.t)
1334                            .filter(|t| *t > r.from)
1335                            .map(|t| (t, r));
1336                        true
1337                    } else {
1338                        false
1339                    }
1340                }
1341            };
1342            if !need {
1343                continue;
1344            }
1345            if t.fetch.requested.is_some() && !settled {
1346                self.pending = true;
1347                continue;
1348            }
1349            let (a, b) = fetch::padded(from, to);
1350            let op = fetch::operator(width_px, raw);
1351            let raw_from = fetch::padded(fetch::needed_from(from, to, true), to).0;
1352            t.fetch.generation += 1;
1353            t.fetch.in_flight = true;
1354            let job = match tail {
1355                // The history keeps what is still in the padded window.
1356                Some((last, r)) => {
1357                    t.fetch.requested = Some(series::Requested {
1358                        from: r.from.max(a).min(last),
1359                        to: b,
1360                        op: r.op,
1361                        at: now,
1362                    });
1363                    fetch::Job {
1364                        trace: t.id,
1365                        generation: t.fetch.generation,
1366                        pv: name,
1367                        from: last,
1368                        to: b,
1369                        op: Operator::Raw,
1370                        raw_from: last,
1371                        probe: false,
1372                        append: true,
1373                    }
1374                }
1375                None => {
1376                    t.fetch.requested = Some(series::Requested {
1377                        from: a,
1378                        to: b,
1379                        op,
1380                        at: now,
1381                    });
1382                    fetch::Job {
1383                        trace: t.id,
1384                        generation: t.fetch.generation,
1385                        pv: name,
1386                        from: a,
1387                        to: b,
1388                        op,
1389                        raw_from,
1390                        probe: t.fetch.raw_only.is_none(),
1391                        append: false,
1392                    }
1393                }
1394            };
1395            self.fetcher
1396                .fetch(&self.pvs, &client, job, self.repaint.clone());
1397        }
1398    }

1481                }

1505                    self.changed_at = now;
1506                }
1507            } else {
1508                if scroll.y != 0.0 {
1509                    self.window
1510                        .zoom((-scroll.y as f64 / 300.0).exp(), anchor, now);

1526        }

1666/// Axis `a`'s value range to draw this frame: fitted to `data` (its
1667/// traces' visible points, envelopes included) or fixed, easing toward
1668/// it.
1669fn shown_values<'a>(
1670    a: &mut ValueAxis,
1671    data: impl Iterator<Item = &'a Vec<Pt>>,
1672    (from, to): (f64, f64),
1673    whole: bool,
1674    dt: f64,
1675) -> (f64, f64) {
1676    let target = if a.autoscale {
1677        fit_values(a.log, whole, data, from, to).or(a.shown)
1678    } else {
1679        Some((a.lo, a.hi))
1680    };
1681    let target = target.unwrap_or(if a.log { (1.0, 10.0) } else { (0.0, 1.0) });
1682    let shown = ease_values(a.shown.unwrap_or(target), target, dt);
1683    a.shown = Some(shown);
1684    shown
1685}

1694/// Whether trace `t` holds whole numbers: an enum (live with state
1695/// labels, or archived scalar data fetched raw, as only enums are).

1698        || (t.fetch.raw_only == Some(true)

1704/// The autoscaled range for `data` between `from` and `to` (at least two
1705/// wide, for `whole` numbers on a linear scale).
1706fn fit_values<'a>(
1707    log: bool,
1708    whole: bool,
1709    data: impl Iterator<Item = &'a Vec<Pt>>,
1710    from: f64,
1711    to: f64,
1712) -> Option<(f64, f64)> {
1713    let values = data.flat_map(|pts| {
1714        let start = pts.partition_point(|p| p.t < from).saturating_sub(1);
1715        pts[start..]
1716            .iter()
1717            .take_while(|p| p.t <= to)
1718            .flat_map(|p| [p.lo, p.hi, p.v])
1719    });
1720    let fit = render::autoscale(log, values);
1721    if whole && !log {
1722        fit.map(|(lo, hi)| axis::whole_range(lo, hi))
1723    } else {
1724        fit
1725    }
1726}

1728/// A value range `shown` one frame (`dt`) closer to `target`.
1729fn ease_values(shown: (f64, f64), target: (f64, f64), dt: f64) -> (f64, f64) {
1730    let (lo, hi) = target;
1731    let (mut a, mut b) = shown;
1732    let scale = hi - lo;
1733    window::ease(&mut a, lo, dt, scale);
1734    window::ease(&mut b, hi, dt, scale);
1735    if a.partial_cmp(&b) != Some(std::cmp::Ordering::Less) {
1736        (a, b) = (lo, hi);
1737    }
1738    (a, b)
1739}

1767/// A value for the readout and chips: at `prec` decimals (in exponent
1768/// form from a million up), else compactly.
1769fn format_value(v: f64, prec: Option<usize>, units: &str) -> String {
1770    if !v.is_finite() {
1771        return "—".into();
1772    }
1773    let text = match prec {
1774        // Large values as the axis labels them: 1.469e9, not 1468514835.908.
1775        Some(p) if v.abs() >= 1e6 => format!("{v:.*e}", p.min(12)),
1776        Some(p) => {
1777            let s = format!("{v:.*}", p.min(12));
1778            // -0.04 at one decimal reads "0.0", not "-0.0".
1779            match s.strip_prefix('-') {
1780                Some(rest) if rest.chars().all(|c| c == '0' || c == '.') => rest.to_string(),
1781                _ => s,
1782            }
1783        }
1784        None => {
1785            let a = v.abs();
1786            if a != 0.0 && !(1e-3..1e6).contains(&a) {
1787                format!("{v:.3e}")
1788            } else {
1789                let s = format!("{v:.4}");
1790                let s = s.trim_end_matches('0').trim_end_matches('.');
1791                if s == "-0" { "0".into() } else { s.to_string() }
1792            }
1793        }
1794    };
1795    if units.is_empty() {
1796        text
1797    } else {
1798        format!("{text} {units}")
1799    }

1948    /// A plot in a headless egui, fed events frame by frame.
1949    struct Harness {
1950        ctx: egui::Context,
1951        plot: Plot,
1952        t: f64,
1953        /// The modifier keys held.
1954        mods: egui::Modifiers,
1955    }
1956
1957    impl Harness {
1958        fn new(plot: Plot) -> Self {
1959            let mut h = Self {
1960                ctx: egui::Context::default(),
1961                plot,
1962                t: 0.0,
1963                mods: egui::Modifiers::NONE,
1964            };
1965            h.frame(vec![]);
1966            h
1967        }
1968
1969        fn frame(&mut self, mut events: Vec<egui::Event>) {
1970            events.insert(0, egui::Event::ModifiersChanged(self.mods));
1971            let input = egui::RawInput {
1972                screen_rect: Some(Rect::from_min_size(Pos2::ZERO, Vec2::new(1000.0, 600.0))),
1973                time: Some(self.t),
1974                events,
1975                ..Default::default()
1976            };
1977            self.t += 0.05;
1978            let Self { ctx, plot, .. } = self;
1979            let mut out = ctx.run_ui(input, |ui| plot.ui(ui));
1980            out.textures_delta.clear();
1981        }
1982
1983        fn key(&mut self, key: egui::Key) {
1984            let ev = |pressed| egui::Event::Key {
1985                key,
1986                physical_key: None,
1987                pressed,
1988                repeat: false,
1989                modifiers: egui::Modifiers::NONE,
1990            };
1991            self.frame(vec![ev(true)]);
1992            self.frame(vec![ev(false)]);
1993        }
1994
1995        fn button(&mut self, at: Pos2, pressed: bool) {
1996            self.frame(vec![egui::Event::PointerButton {
1997                pos: at,
1998                button: egui::PointerButton::Primary,
1999                pressed,
2000                modifiers: egui::Modifiers::NONE,
2001            }]);
2002        }
2003
2004        /// Frames until `done`, for up to 10 s of real time.
2005        fn until(&mut self, mut done: impl FnMut(&Plot) -> bool) {
2006            let start = std::time::Instant::now();
2007            while !done(&self.plot) {
2008                assert!(start.elapsed().as_secs() < 10, "timed out");
2009                std::thread::sleep(std::time::Duration::from_millis(20));
2010                self.frame(vec![]);
2011            }
2012        }
2013    }
