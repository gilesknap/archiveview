200/// A value axis and the traces on it (those whose `axis` is its id).
201#[derive(Clone, Debug)]
202struct ValueAxis {
203    id: u32,
204    /// The name the user gave it; else it names itself.
205    name: Option<String>,
206    autoscale: bool,
207    /// The fixed range when not autoscaling.
208    lo: f64,
209    hi: f64,
210    log: bool,
211    /// The range drawn, easing toward the target.
212    shown: Option<(f64, f64)>,
213    /// It has had a trace, so it goes when its last one leaves. An axis
214    /// from a view that no trace is on yet waits for a trace to settle:
215    /// the one in `reserved`, else any.
216    used: bool,
217    reserved: Option<u64>,
218}

225    fn from_view(id: u32, y: &YAxis) -> Self {
226        Self {
227            id,
228            name: y.name.clone(),
229            autoscale: y.autoscale || y.min.is_none() || y.max.is_none(),
230            lo: y.min.unwrap_or(0.0),
231            hi: y.max.unwrap_or(1.0),
232            log: y.log,
233            shown: None,
234            used: false,
235            reserved: None,
236        }

239    fn view(&self) -> YAxis {
240        YAxis {
241            autoscale: self.autoscale,
242            min: (!self.autoscale).then_some(self.lo),
243            max: (!self.autoscale).then_some(self.hi),
244            log: self.log,
245            name: self.name.clone(),
246        }
247    }
248
249    /// On a log scale a fixed range must be above zero: back to
250    /// autoscaling if not.
251    fn fit_log_range(&mut self) {
252        if self.log && !self.autoscale && self.lo <= 0.0 {
253            self.autoscale = true;
254            self.shown = None;
255        }
256    }
257
258    /// Zoom by `factor` about `at` (the middle when `None`), from the
259    /// range `shown` (else its own), leaving autoscale.
260    fn zoom(&mut self, factor: f64, at: Option<f64>, shown: (f64, f64)) {
261        let (lo, hi) = self.shown.unwrap_or(shown);
262        let mid = if self.log && lo > 0.0 {
263            (lo * hi).sqrt()
264        } else {
265            (lo + hi) / 2.0
266        };
267        let (l, h) = window::zoom_values(lo, hi, factor, at.unwrap_or(mid), self.log);
268        if l.is_finite() && h.is_finite() && l < h {
269            self.autoscale = false;
270            (self.lo, self.hi) = (l, h);
271        }
272    }
273}

368    /// Change (or remove) the archiver: refetches every trace's history,
369    /// and checks that the new one can be reached.
370    pub fn set_archiver(&mut self, archiver: Option<ArchiverClient>) {
371        if let Some(a) = &archiver {
372            remember(&mut self.recent, a.base());
373        }
374        self.archiver = archiver;
375        self.monitor.reset(self.archiver.is_some());
376        // Searches in flight were for the old archiver: drop their answers
377        // and search again.
378        self.search.restart();
379        self.forget_parked_histories();
380        for t in &mut self.traces {
381            t.history = None;
382            // The generation carries on, so a late answer from the old
383            // archiver is still recognised as stale.
384            t.fetch = series::FetchState {
385                generation: t.fetch.generation + 1,
386                ..Default::default()
387            };
388        }
389    }

436    /// Add traces (skipping names already plotted, compared by
437    /// `pv::PvName::key`): subscribe live and fetch the visible history.
438    /// One undo step ("add X"), however they were picked.
439    pub fn add_pvs(&mut self, names: &[String]) {
440        // What came before is a step of its own.
441        self.commit(false);
442        for name in names {
443            self.add(name);
444        }

447    fn add(&mut self, name: &str) -> Option<&mut Trace> {
448        let name = name.trim();
449        let key = pv::PvName::parse(name).key();
450        if name.is_empty() || self.traces.iter().any(|t| t.pv.name().key() == key) {
451            return None;
452        }
453        // The first palette colour no other trace has.
454        let slot = (0..)
455            .find(|s| !self.traces.iter().any(|t| t.slot == *s))
456            .unwrap_or(0);
457        let mut trace = Trace::new(self.next_trace, name, self.pvs.subscribe(name));
458        trace.slot = slot;
459        self.next_trace += 1;
460        self.traces.push(trace);
461        self.traces.last_mut()
462    }

464    /// Add the PVs named in pasted `text` (the primary selection after a
465    /// middle-click, or the clipboard): see [`pv_names`]. More than
466    /// [`MAX_PASTE`] adds none (a screenful selected by accident). Says what it did,
467    /// as [`Plot::take_pasted`] does after.
468    pub fn paste_pvs(&mut self, text: &str) -> Pasted {
469        let mut pasted = Pasted::default();
470        let names = pv_names(text);
471        if names.len() > MAX_PASTE {
472            pasted.too_many = names.len();
473            return self.pasted(pasted);
474        }
475        let plotted: Vec<String> = self.traces.iter().map(|t| t.pv.name().key()).collect();
476        let mut adding = Vec::new();
477        for name in names {
478            let key = pv::PvName::parse(&name).key();
479            if plotted.contains(&key) {
480                pasted.present.push(name);
481            } else if !adding.contains(&key) {
482                // Once, however it is spelt (`A:B`, `ca://A:B`).
483                adding.push(key);
484                pasted.added.push(name);
485            }
486        }
487        // As the search box adds: one call, one step to undo.
488        self.add_pvs(&pasted.added);
489        if pasted.added.is_empty() && pasted.present.is_empty() {
490            pasted.text = text.to_string();
491        }
492        self.pasted(pasted)
493    }

572    /// The current view (traces, range, value axes, archiver), to save.
573    pub fn view(&self) -> View {
574        let range = if self.window.live {
575            view::Range::Relative {
576                secs: self.window.span,
577            }
578        } else {
579            let (from, to) = self.window.range(now());
580            view::Range::Absolute {
581                start: view::format_time(from),
582                end: view::format_time(to),
583            }
584        };
585        // Only axes with traces, or kept for a PV yet to settle (pruned
586        // once that PV is gone): no stale fixed range comes back for
587        // whichever PV would take it.
588        let kept: Vec<&ValueAxis> = (self.axes.iter())
589            .filter(|a| self.has_traces(a) || a.reserved.is_some())
590            .collect();
591        let mut axes = kept.iter().map(|a| a.view());
592        View {
593            version: view::VERSION,
594            pvs: self
595                .traces
596                .iter()
597                .map(|t| TraceView {
598                    name: t.name.clone(),
599                    color: t.color.clone(),
600                    visible: t.visible,
601                    axis: t.axis.and_then(|id| kept.iter().position(|a| a.id == id)),
602                })
603                .collect(),
604            range,
605            y: axes.next().unwrap_or_default(),
606            axes: axes.collect(),
607            archiver: self.archiver.as_ref().map(|a| a.base().to_string()),
608        }
609    }

809        let gap = if chips.height > 0.0 { 6.0 } else { 0.0 };
810        let plot_h = (ui.available_height() - chips.height - gap).max(MIN_SIZE.y);

960            if self.pulse {
961                self.repaint_after(&ctx, Duration::from_millis(66));
962            }
963        }
964    }

1002    /// The time range to draw this frame, easing toward the window.
1003    fn shown_range(&mut self, now: f64, dt: f64) -> (f64, f64) {
1004        let (from, to) = self.window.range(now);
1005        let (mut a, mut b) = match self.shown {
1006            // With a minimum repaint interval, time passing while live
1007            // moves what is drawn as it moves the window, rather than
1008            // easing after it: easing each step would spend the frames the
1009            // host asked to save. Only from a live frame: going live eases.
1010            Some((a, b)) if !std::mem::take(&mut self.snap) => {
1011                let passed = match self.shown_live_at {
1012                    Some(at) if self.window.live && !self.min_repaint.is_zero() => now - at,
1013                    _ => 0.0,
1014                };
1015                (a + passed, b + passed)
1016            }
1017            _ => (from, to),
1018        };
1019        self.shown_live_at = self.window.live.then_some(now);
1020        let span = to - from;
1021        window::ease(&mut a, from, dt, span);
1022        window::ease(&mut b, to, dt, span);
1023        self.shown = Some((a, b));
1024        (a, b)
1025    }

1072    /// Settle new traces on axes, once each, in the legend's order, so
1073    /// the outcome does not depend on which archive answer comes first.
1074    ///
1075    /// A trace is ready when it has data on screen and its archived
1076    /// history has come (live data alone: after [`LIVE_WAIT`], so one
1077    /// first sample does not decide). It waits while an earlier trace may
1078    /// still get data: until that one settles, or turns out to have none,
1079    /// or [`SETTLE_WAIT`] passes. Then the axes left with no trace go.
1080    fn settle_axes(&mut self, data: &[Vec<Pt>], now: f64) {
1081        // Settled traces shown since: their stats, for scoring.
1082        for i in 0..self.traces.len() {
1083            if self.traces[i].axis.is_some() && self.traces[i].stats.is_none() {
1084                self.traces[i].stats = self.trace_stats(i, data);
1085            }
1086        }
1087        let mut blocked = false;
1088        for i in 0..self.traces.len() {
1089            let t = &self.traces[i];
1090            // Arrays are not plotted: they take no axis, nor hold others up.
1091            if t.axis.is_some() || !t.visible || t.is_array() {
1092                continue;
1093            }
1094            let seen = *self.traces[i].seen_at.get_or_insert(now);
1095            let late = now - seen >= SETTLE_WAIT;
1096            let stats = self.trace_stats(i, data);
1097            if stats.is_some() {
1098                self.traces[i].data_since.get_or_insert(now);
1099            }
1100            let t = &self.traces[i];
1101            let pending = self.awaits_history(t);
1102            let archived = t.history.as_ref().is_some_and(|h| !h.samples.is_empty());
1103            let live = !pending && t.data_since.is_some_and(|s| now - s >= LIVE_WAIT);
1104            // No units from the archive: its live units may yet come, and
1105            // should decide (within the wait).
1106            let units_due = units(t).is_empty() && t.archive_name().is_some() && !t.pv.connected();
1107            if stats.is_none() || !(late || ((archived || live) && !units_due)) {
1108                // It may yet get data, or more of it: those after it wait.
1109                blocked |= !late && (pending || stats.is_some());
1110                if late && stats.is_none() {
1111                    // Nothing came in time: an axis kept for it (from an
1112                    // old view) is not kept any longer.
1113                    let id = self.traces[i].id;
1114                    for a in &mut self.axes {
1115                        if a.reserved == Some(id) {
1116                            a.reserved = None;
1117                        }
1118                    }
1119                }
1120                continue;
1121            }
1122            if blocked {
1123                continue;
1124            }
1125            let stats = stats.expect("ready");
1126            let (known, unknown) = self.axis_stats();
1127            let id = match axes::choose(&stats, &known, unknown) {
1128                Choice::Join(id) => id,
1129                Choice::New => self.new_axis(&stats, self.traces[i].id),
1130            };
1131            self.traces[i].stats = Some(stats);
1132            self.put_on(i, id);
1133        }
1134        self.prune_axes();
1135    }

1168    /// there (keeping the axis' range, scale and name).
1169    fn move_trace(&mut self, trace: u64, to: Option<u32>) {
1170        let Some(i) = self.traces.iter().position(|t| t.id == trace) else {
1171            return;
1172        };

1174        let id = match to {
1175            Some(id) if self.axes.iter().any(|a| a.id == id) => id,
1176            Some(_) => return,
1177            None if on.is_some() && self.traces.iter().filter(|t| t.axis == on).count() == 1 => {
1178                return;
1179            }
1180            None => {
1181                // Fresh, not one waiting for a view's trace.
1182                let a = ValueAxis::new(self.next_axis);
1183                self.next_axis += 1;
1184                let id = a.id;
1185                self.axes.push(a);
1186                id
1187            }
1188        };
1189        self.put_on(i, id);
1190        self.prune_axes();

1338    /// Take in finished fetches, searches and archiver probes, and what
1339    /// they say of the archiver.
1340    fn take_results(&mut self, now: f64) {
1341        let mut back = false;
1342        if let Some((epoch, result)) = self.fetcher.take_probed() {
1343            back |= self.monitor.probed(epoch, result, now);
1344        }
1345        for done in self.fetcher.take() {
1346            let Some(t) = self.traces.iter_mut().find(|t| t.id == done.trace) else {
1347                continue;
1348            };
1349            if done.generation != t.fetch.generation {
1350                continue;
1351            }
1352            t.fetch.in_flight = false;
1353            match &done.result {
1354                Ok(_) => {
1355                    t.fetch.failures = 0;
1356                    back |= self.monitor.ok(now);
1357                }
1358                Err(e) if status::archiver_fault(e, false) => self.monitor.suspect(e, now),
1359                // Not archived, or something about this PV: the archiver
1360                // answered, but that says nothing of the others.
1361                Err(_) => {}
1362            }
1363            // The window really fetched: a raw-only trace's is shorter.
1364            if let Some(r) = &mut t.fetch.requested
1365                && !done.append
1366            {
1367                r.from = r.from.max(done.from);
1368            }
1369            if let Some(info) = &done.info {
1370                t.fetch.raw_only = Some(!info.binnable());
1371            }
1372            match done.result {
1373                Ok(series) if done.append => {
1374                    let series = series.unwrap_or_default();
1375                    let keep_from = t.fetch.requested.map_or(f64::NEG_INFINITY, |r| r.from);
1376                    match &mut t.history {
1377                        Some(h) if !h.binned && !series.binned => {
1378                            h.append(series.samples, keep_from, done.to);
1379                            t.fetch.error = None;
1380                            // Grown past what `optimized_N` returns raw:
1381                            // fetch it whole again, binned.
1382                            if fetch::outgrown(t.fetch.requested.map(|r| r.op), h.samples.len()) {
1383                                t.fetch.requested = None;
1384                            }
1385                        }
1386                        // Not what the history holds: fetch it all again.
1387                        _ => t.fetch.requested = None,
1388                    }
1389                }
1390                Ok(series) => {
1391                    let series = series.unwrap_or_default();
1392                    let scalar = series.kind == ValueKind::Scalar;
1393                    if series.samples.is_empty() {
1394                        // Nothing to tell the type by.
1395                    } else if !scalar && done.op != Operator::Raw {
1396                        // Text: ask again, raw. A waveform is not plotted,
1397                        // so what came is enough.
1398                        t.fetch.raw_only = Some(true);
1399                        if series.kind != ValueKind::Waveform {
1400                            t.fetch.requested = None;
1401                        }
1402                    } else if t.fetch.raw_only.is_none() {
1403                        t.fetch.raw_only = Some(!scalar);
1404                    }
1405                    let meta = if series.meta.name.is_empty() {
1406                        t.history
1407                            .as_ref()
1408                            .map(|h| h.meta.clone())
1409                            .unwrap_or_default()
1410                    } else {
1411                        series.meta
1412                    };
1413                    // An array's samples are not plotted: knowing it is
1414                    // archived is enough, so they are not kept.
1415                    let mut samples = series.samples;
1416                    if series.kind == ValueKind::Waveform {
1417                        samples = Vec::new();
1418                    }
1419                    t.history = Some(History {
1420                        to: done.to,
1421                        binned: series.binned,
1422                        kind: series.kind,
1423                        meta,
1424                        samples,
1425                    });
1426                    t.fetch.error = None;
1427                }
1428                Err(e) if e.is_not_archived() => {
1429                    t.fetch.not_archived = true;
1430                    t.fetch.error = None;
1431                    t.history = None;
1432                }
1433                Err(e) => {
1434                    log::warn!("{}: {e}", t.name);
1435                    t.fetch.archivers_fault = status::archiver_fault(&e, false);
1436                    t.fetch.error = Some(e.to_string());
1437                    t.fetch.failures += 1;
1438                }
1439            }
1440        }
1441        back |= bar::take_found(self, now);
1442        if back {
1443            // The archiver is back: ask again now for what failed because
1444            // it was down (not for what fails for reasons of its own,
1445            // which keeps backing off).
1446            for t in &mut self.traces {
1447                if t.fetch.error.is_some() && t.fetch.archivers_fault && !t.fetch.in_flight {
1448                    t.fetch.failures = 0;
1449                    t.fetch.requested = None;
1450                }
1451            }
1452        }
1453    }

1455    /// Check the archiver when one was just set, and every
1456    /// `status::REPROBE` seconds while it cannot be reached.
1457    fn schedule_probe(&mut self, now: f64) {
1458        let Some(client) = &self.archiver else {
1459            return;
1460        };
1461        if self.monitor.should_probe(now) {
1462            self.monitor.sent(now);
1463            self.fetcher
1464                .probe(&self.pvs, client, self.monitor.epoch, self.repaint.clone());
1465        }
1466        if self.monitor.status.state == ArchiverState::Unreachable {
1467            // Wake for the next probe even in an idle window.
1468            let next = self.monitor.probed_at + self.monitor.reprobe - now;
1469            if let Some(r) = &self.repaint {
1470                let after = Duration::from_secs_f64(next.clamp(0.05, status::REPROBE));
1471                r.ctx
1472                    .request_repaint_after_for(after.max(self.min_repaint), r.viewport);
1473            }
1474        }
1475    }

1477    /// Start the fetches the visible window needs, debounced while the
1478    /// user pans and zooms.
1479    fn schedule_fetches(&mut self, now: f64, width_px: f32) {
1480        self.pending = false;
1481        let Some(client) = self.archiver.clone() else {
1482            return;
1483        };
1484        let (from, to) = self.window.range(now);
1485        let settled = now - self.changed_at >= DEBOUNCE;
1486        let live = self.window.live;
1487        for t in &mut self.traces {
1488            let Some(name) = t.archive_name() else {
1489                continue;
1490            };
1491            // One request per trace at a time: when it is done, the next
1492            // frame asks for wherever the window is then.
1493            if t.fetch.not_archived || t.fetch.in_flight {
1494                continue;
1495            }
1496            // An array is not plotted: once it is known to be archived,
1497            // there is nothing more to fetch.
1498            if t.history
1499                .as_ref()
1500                .is_some_and(|h| h.kind == ValueKind::Waveform)
1501            {
1502                continue;
1503            }
1504            let raw = t.fetch.raw_only.unwrap_or(false);
1505            let from = fetch::needed_from(from, to, raw);
1506            // A live refresh of raw history: only from its last sample on.
1507            let mut tail = None;
1508            let need = match t.fetch.requested {
1509                None => true,
1510                Some(r) => {
1511                    let binned =
1512                        r.op != Operator::Raw && t.history.as_ref().is_none_or(|h| h.binned);
1513                    let covered = fetch::covers((r.from, r.to), binned, from, to);
1514                    // After an error: again once the backoff has passed, or
1515                    // after the base delay for a different window.
1516                    let wait = if covered {
1517                        fetch::backoff(RETRY, t.fetch.failures)
1518                    } else {
1519                        RETRY
1520                    };
1521                    if t.fetch.error.is_some() {
1522                        now - r.at > wait
1523                    } else if !covered {
1524                        true
1525                    } else if live
1526                        && now - r.at
1527                            > fetch::refresh_after(LIVE_REFRESH, (r.from, r.to), r.op, binned)
1528                    {
1529                        tail = t
1530                            .history
1531                            .as_ref()
1532                            .filter(|h| !h.binned)
1533                            .and_then(|h| h.samples.last())
1534                            .map(|s| s.t)
1535                            .filter(|t| *t > r.from)
1536                            .map(|t| (t, r));
1537                        true
1538                    } else {
1539                        false
1540                    }
1541                }
1542            };
1543            if !need {
1544                continue;
1545            }
1546            if t.fetch.requested.is_some() && !settled {
1547                self.pending = true;
1548                continue;
1549            }
1550            let (a, b) = fetch::padded(from, to);
1551            let op = fetch::operator(width_px, raw);
1552            let raw_from = fetch::padded(fetch::needed_from(from, to, true), to).0;
1553            t.fetch.generation += 1;
1554            t.fetch.in_flight = true;
1555            let job = match tail {
1556                // The history keeps what is still in the padded window.
1557                Some((last, r)) => {
1558                    t.fetch.requested = Some(series::Requested {
1559                        from: r.from.max(a).min(last),
1560                        to: b,
1561                        op: r.op,
1562                        at: now,
1563                    });
1564                    fetch::Job {
1565                        trace: t.id,
1566                        generation: t.fetch.generation,
1567                        pv: name,
1568                        from: last,
1569                        to: b,
1570                        op: Operator::Raw,
1571                        raw_from: last,
1572                        probe: false,
1573                        append: true,
1574                    }
1575                }
1576                None => {
1577                    t.fetch.requested = Some(series::Requested {
1578                        from: a,
1579                        to: b,
1580                        op,
1581                        at: now,
1582                    });
1583                    fetch::Job {
1584                        trace: t.id,
1585                        generation: t.fetch.generation,
1586                        pv: name,
1587                        from: a,
1588                        to: b,
1589                        op,
1590                        raw_from,
1591                        probe: t.fetch.raw_only.is_none(),
1592                        append: false,
1593                    }
1594                }
1595            };
1596            self.fetcher
1597                .fetch(&self.pvs, &client, job, self.repaint.clone());
1598        }
1599    }

1932/// Axis `a`'s value range to draw this frame: fitted to `data` (its
1933/// traces' visible points, envelopes included) or fixed, easing toward
1934/// it.
1935fn shown_values<'a>(
1936    a: &mut ValueAxis,
1937    data: impl Iterator<Item = &'a Vec<Pt>>,
1938    (from, to): (f64, f64),
1939    whole: bool,
1940    dt: f64,
1941) -> (f64, f64) {
1942    let target = if a.autoscale {
1943        fit_values(a.log, whole, data, from, to).or(a.shown)
1944    } else {
1945        Some((a.lo, a.hi))
1946    };
1947    let target = target.unwrap_or(if a.log { (1.0, 10.0) } else { (0.0, 1.0) });
1948    let shown = ease_values(a.shown.unwrap_or(target), target, dt);
1949    a.shown = Some(shown);
1950    shown
1951}

1953/// What picking window `w` is called, as an undo step: "show the last

1987/// The autoscaled range for `data` between `from` and `to` (at least two
1988/// wide, for `whole` numbers on a linear scale).
1989fn fit_values<'a>(
1990    log: bool,
1991    whole: bool,
1992    data: impl Iterator<Item = &'a Vec<Pt>>,
1993    from: f64,
1994    to: f64,
1995) -> Option<(f64, f64)> {
1996    let values = data.flat_map(|pts| {
1997        let start = pts.partition_point(|p| p.t < from).saturating_sub(1);
1998        pts[start..]
1999            .iter()
2000            .take_while(|p| p.t <= to)
2001            .flat_map(|p| [p.lo, p.hi, p.v])
2002    });
2003    let fit = render::autoscale(log, values);
2004    if whole && !log {
2005        fit.map(|(lo, hi)| axis::whole_range(lo, hi))
2006    } else {
2007        fit
2008    }
2009}

2011/// A value range `shown` one frame (`dt`) closer to `target`.
2012fn ease_values(shown: (f64, f64), target: (f64, f64), dt: f64) -> (f64, f64) {
2013    let (lo, hi) = target;
2014    let (mut a, mut b) = shown;
2015    let scale = hi - lo;
2016    window::ease(&mut a, lo, dt, scale);
2017    window::ease(&mut b, hi, dt, scale);
2018    if a.partial_cmp(&b) != Some(std::cmp::Ordering::Less) {
2019        (a, b) = (lo, hi);
2020    }
2021    (a, b)
2022}

2068/// A value for the readout and chips: at `prec` decimals (in exponent
2069/// form from a million up), else compactly.
2070fn format_value(v: f64, prec: Option<usize>, units: &str) -> String {
2071    if !v.is_finite() {
2072        return "—".into();
2073    }
2074    let text = match prec {
2075        // Large values as the axis labels them: 1.469e9, not 1468514835.908.
2076        Some(p) if v.abs() >= 1e6 => format!("{v:.*e}", p.min(12)),
2077        Some(p) => {
2078            let s = format!("{v:.*}", p.min(12));
2079            // -0.04 at one decimal reads "0.0", not "-0.0".
2080            match s.strip_prefix('-') {
2081                Some(rest) if rest.chars().all(|c| c == '0' || c == '.') => rest.to_string(),
2082                _ => s,
2083            }
2084        }
2085        None => {
2086            let a = v.abs();
2087            if a != 0.0 && !(1e-3..1e6).contains(&a) {
2088                format!("{v:.3e}")
2089            } else {
2090                let s = format!("{v:.4}");
2091                let s = s.trim_end_matches('0').trim_end_matches('.');
2092                if s == "-0" { "0".into() } else { s.to_string() }
2093            }
2094        }
2095    };
2096    if units.is_empty() {
2097        text
2098    } else {
2099        format!("{text} {units}")
2100    }

3905    impl Harness {
3906        /// The wheel turned over the middle of the plot, then frames for
3907        /// the smooth scrolling to finish.
3908        fn wheel(&mut self, dy: f32) {
3909            self.frame(vec![egui::Event::PointerMoved(MIDDLE)]);
3910            self.frame(vec![egui::Event::MouseWheel {
3911                unit: egui::MouseWheelUnit::Point,
3912                delta: Vec2::new(0.0, dy),
3913                phase: egui::TouchPhase::Move,
3914                modifiers: self.mods,
3915            }]);
3916            for _ in 0..20 {
3917                self.frame(vec![]);
3918            }
3919        }
3920
3921        /// `key` pressed with `mods` held.
3922        fn key_with(&mut self, key: egui::Key, mods: egui::Modifiers) {
3923            let held = std::mem::replace(&mut self.mods, mods);
3924            let ev = |pressed| egui::Event::Key {
3925                key,
3926                physical_key: None,
3927                pressed,
3928                repeat: false,
3929                modifiers: mods,
3930            };
3931            self.frame(vec![ev(true)]);
3932            self.frame(vec![ev(false)]);
3933            self.mods = held;
3934        }
3935
3936        /// A click of mouse button `button` over the plot.
3937        fn click_with(&mut self, button: egui::PointerButton) {
3938            for pressed in [true, false] {
3939                self.frame(vec![egui::Event::PointerButton {
3940                    pos: MIDDLE,
3941                    button,
3942                    pressed,
3943                    modifiers: egui::Modifiers::NONE,
3944                }]);
3945            }
3946        }
3947
3948        /// A drag across the plot from `from` to `to` (x), pausing for
3949        /// `pause` half way.
3950        fn drag(&mut self, from: f32, to: f32, pause: std::time::Duration) {
3951            let at = |x: f32| Pos2::new(x, 300.0);
3952            self.frame(vec![egui::Event::PointerMoved(at(from))]);
3953            self.button(at(from), true);
3954            let mid = (from + to) / 2.0;
3955            for x in [from + 10.0, mid] {
3956                self.frame(vec![egui::Event::PointerMoved(at(x))]);
3957            }
3958            std::thread::sleep(pause);
3959            self.frame(vec![egui::Event::PointerMoved(at(to))]);
3960            self.button(at(to), false);
3961            self.frame(vec![]);
3962        }
