1//! `--snapshot OUT.png`: run the app headless and rasterise one frame in
2//! software, so no GPU or display is needed (adapted from bobview's
3//! `snapshot.rs`). Used by CI and the docs, and to check the UI.

14/// Longest a snapshot waits, after `wait`, for fetches to finish; after
15/// that it renders what it has.
16const DEADLINE: Duration = Duration::from_secs(60);
17
18/// Snapshot settings.
19#[derive(Clone, Debug)]
20pub struct Options {
21    /// Image size in pixels.
22    pub size: [u32; 2],
23    /// Pixels per point.
24    pub scale_factor: f32,
25    /// Seconds to run before capturing, at least; then also until no fetch
26    /// is in flight (giving up after 60 s).
27    pub wait: f64,
28    /// Render dark (else light, as headless has no OS theme).
29    pub dark: bool,
30    /// Rest the pointer here (pixels) before capturing, to show the
31    /// crosshair readout.
32    pub hover: Option<[f32; 2]>,
33    /// Type this into the search box before capturing, to show its
34    /// completions.
35    pub search: Option<String>,
36}
37
38impl Default for Options {
39    fn default() -> Self {
40        Self {
41            size: [1280, 800],
42            scale_factor: 1.0,
43            wait: 2.0,
44            dark: false,
45            hover: None,
46            search: None,
47        }
48    }
49}

51/// Render the app headless and write the PNG to `out`.
52pub fn run(app: AppOptions, opts: &Options, out: &Path) -> Result<(), String> {
53    let ctx = egui::Context::default();
54    let app_opts = AppOptions {
55        theme: Some(if opts.dark { Theme::Dark } else { Theme::Light }),
56        ..app
57    };
58    let mut app = ArchiveApp::new(&ctx, app_opts);
59    let mut textures: HashMap<TextureId, ColorImage> = HashMap::new();
60    let begin = Instant::now();
61    let native = opts.scale_factor;
62    let px = Vec2::new(opts.size[0] as f32, opts.size[1] as f32);
63    let wait = Duration::from_secs_f64(opts.wait);
64    // Frames after loading: a few to settle animations, with the pointer
65    // resting where asked.
66    let mut settle = 0;
67    // Typing into the search box: `/`, then the text.
68    let key = |pressed| egui::Event::Key {
69        key: egui::Key::Slash,
70        physical_key: None,
71        pressed,
72        repeat: false,
73        modifiers: egui::Modifiers::NONE,
74    };
75    let mut typing: std::collections::VecDeque<Vec<egui::Event>> = match &opts.search {
76        Some(text) => [
77            vec![key(true)],
78            vec![key(false)],
79            vec![],
80            vec![egui::Event::Text(text.clone())],
81        ]
82        .into_iter()
83        .collect(),
84        None => Default::default(),
85    };
86
87    let output = loop {
88        let ready = begin.elapsed() >= wait && !app.is_loading();
89        let ppp = native * ctx.zoom_factor();
90        let screen = Rect::from_min_size(Pos2::ZERO, px / ppp);
91        let mut events = Vec::new();
92        if ready {
93            // The pointer rests where asked, or on the plot to type (the
94            // keys need it there).
95            let at = opts
96                .hover
97                .or(opts.search.as_ref().map(|_| [px.x / 2.0, px.y / 2.0]));
98            match at {
99                // Once typed, the pointer leaves unless asked to rest, so a
100                // search snapshot has no stray crosshair.
101                Some(_) if opts.hover.is_none() && typing.is_empty() => {
102                    events.push(egui::Event::PointerGone);
103                }
104                Some([x, y]) => events.push(egui::Event::PointerMoved(Pos2::new(x, y) / ppp)),
105                None => {}
106            }
107            events.extend(typing.pop_front().unwrap_or_default());
108        }
109        let mut input = egui::RawInput {
110            screen_rect: Some(screen),
111            time: Some(begin.elapsed().as_secs_f64()),
112            events,
113            ..Default::default()
114        };
115        let info = input.viewports.entry(egui::ViewportId::ROOT).or_default();
116        info.native_pixels_per_point = Some(native);
117        info.inner_rect = Some(screen);
118        let mut output = ctx.run_ui(input, |ui| app.draw(ui));
119        for (id, deltas) in &output.textures_delta.set {
120            for delta in deltas {
121                apply_texture(&mut textures, *id, delta);
122            }
123        }
124        output.textures_delta.clear();
125        if begin.elapsed() > wait + DEADLINE {
126            log::warn!("snapshot taken after {DEADLINE:?} with archive fetches still running");
127            break output;
128        }
129        if ready && typing.is_empty() {
130            settle += 1;
131            // Long enough for 150 ms easing to finish.
132            if settle > 12 {
133                break output;
134            }
135        }
136        std::thread::sleep(Duration::from_millis(if ready { 30 } else { 50 }));
137    };
138
139    let ppp = output.pixels_per_point;
140    let prims = ctx.tessellate(output.shapes, ppp);
141    let mut canvas = Canvas::new(px.x as usize, px.y as usize);
142    for prim in prims {
143        if let epaint::Primitive::Mesh(mesh) = &prim.primitive {
144            canvas.draw_mesh(mesh, prim.clip_rect, ppp, textures.get(&mesh.texture_id));
145        }
146    }
147    canvas.save(out)
148}

150fn apply_texture(
151    textures: &mut HashMap<TextureId, ColorImage>,
152    id: TextureId,
153    delta: &epaint::ImageDelta,
154) {
155    let egui::ImageData::Color(image) = &delta.image;
156    match delta.pos {
157        None => {
158            textures.insert(id, (**image).clone());
159        }
160        Some([x0, y0]) => {
161            if let Some(tex) = textures.get_mut(&id) {
162                let [w, h] = image.size;
163                for y in 0..h {
164                    for x in 0..w {
165                        let (tx, ty) = (x0 + x, y0 + y);
166                        if tx < tex.size[0] && ty < tex.size[1] {
167                            tex.pixels[ty * tex.size[0] + tx] = image.pixels[y * w + x];
168                        }
169                    }
170                }
171            }
172        }
173    }
174}

176/// Premultiplied RGBA float framebuffer.
177struct Canvas {
178    w: usize,
179    h: usize,
180    px: Vec<[f32; 4]>,
181}
182
183impl Canvas {
184    fn new(w: usize, h: usize) -> Self {
185        Self {
186            w,
187            h,
188            px: vec![[0.0, 0.0, 0.0, 1.0]; w * h],
189        }
190    }
191
192    /// Draw a mesh in points at `ppp` pixels per point.
193    fn draw_mesh(&mut self, mesh: &epaint::Mesh, clip: Rect, ppp: f32, tex: Option<&ColorImage>) {
194        let clip = Rect::from_min_max(clip.min * ppp, clip.max * ppp);
195        let clip = clip.intersect(Rect::from_min_size(
196            Pos2::ZERO,
197            Vec2::new(self.w as f32, self.h as f32),
198        ));
199        if clip.is_negative() {
200            return;
201        }
202        for tri in mesh.indices.as_chunks::<3>().0 {
203            let v = tri.map(|i| {
204                let mut v = mesh.vertices[i as usize];
205                v.pos = (v.pos.to_vec2() * ppp).to_pos2();
206                v
207            });
208            self.triangle([&v[0], &v[1], &v[2]], clip, tex);
209        }
210    }
211
212    fn triangle(&mut self, v: [&epaint::Vertex; 3], clip: Rect, tex: Option<&ColorImage>) {
213        let (p0, p1, p2) = (v[0].pos, v[1].pos, v[2].pos);
214        let area = (p1.x - p0.x) * (p2.y - p0.y) - (p2.x - p0.x) * (p1.y - p0.y);
215        if area.abs() < 1e-6 || !area.is_finite() {
216            return;
217        }
218        let min_x = p0.x.min(p1.x).min(p2.x).max(clip.min.x).floor() as i64;
219        let max_x = p0.x.max(p1.x).max(p2.x).min(clip.max.x).ceil() as i64;
220        let min_y = p0.y.min(p1.y).min(p2.y).max(clip.min.y).floor() as i64;
221        let max_y = p0.y.max(p1.y).max(p2.y).min(clip.max.y).ceil() as i64;
222        let col = |c: egui::Color32| {
223            [c.r() as f32, c.g() as f32, c.b() as f32, c.a() as f32].map(|x| x / 255.0)
224        };
225        let cs = [col(v[0].color), col(v[1].color), col(v[2].color)];
226        for y in min_y.max(0)..max_y.min(self.h as i64) {
227            for x in min_x.max(0)..max_x.min(self.w as i64) {
228                let p = Pos2::new(x as f32 + 0.5, y as f32 + 0.5);
229                let w0 = ((p1.x - p.x) * (p2.y - p.y) - (p2.x - p.x) * (p1.y - p.y)) / area;
230                let w1 = ((p2.x - p.x) * (p0.y - p.y) - (p0.x - p.x) * (p2.y - p.y)) / area;
231                let w2 = 1.0 - w0 - w1;
232                if w0 < -1e-4 || w1 < -1e-4 || w2 < -1e-4 {
233                    continue;
234                }
235                let mut c = [0.0f32; 4];
236                for (i, ch) in c.iter_mut().enumerate() {
237                    *ch = cs[0][i] * w0 + cs[1][i] * w1 + cs[2][i] * w2;
238                }
239                if let Some(t) = tex {
240                    let u = v[0].uv.x * w0 + v[1].uv.x * w1 + v[2].uv.x * w2;
241                    let vv = v[0].uv.y * w0 + v[1].uv.y * w1 + v[2].uv.y * w2;
242                    let s = sample(t, u, vv);
243                    for (i, ch) in c.iter_mut().enumerate() {
244                        *ch *= s[i];
245                    }
246                }
247                let d = &mut self.px[y as usize * self.w + x as usize];
248                for i in 0..4 {
249                    d[i] = c[i] + d[i] * (1.0 - c[3]);
250                }
251            }
252        }
253    }
254
255    fn save(&self, path: &Path) -> Result<(), String> {
256        let file = std::fs::File::create(path).map_err(|e| format!("{}: {e}", path.display()))?;
257        let mut enc =
258            png::Encoder::new(std::io::BufWriter::new(file), self.w as u32, self.h as u32);
259        enc.set_color(png::ColorType::Rgb);
260        enc.set_depth(png::BitDepth::Eight);
261        let mut writer = enc.write_header().map_err(|e| e.to_string())?;
262        let data: Vec<u8> = self
263            .px
264            .iter()
265            .flat_map(|p| [p[0], p[1], p[2]].map(|c| (c.clamp(0.0, 1.0) * 255.0).round() as u8))
266            .collect();
267        writer.write_image_data(&data).map_err(|e| e.to_string())
268    }
269}

271/// Bilinear sample of a premultiplied texture, as 0..1 floats.
272fn sample(t: &ColorImage, u: f32, v: f32) -> [f32; 4] {
273    let [w, h] = t.size;
274    let x = (u * w as f32 - 0.5).clamp(0.0, (w - 1) as f32);
275    let y = (v * h as f32 - 0.5).clamp(0.0, (h - 1) as f32);
276    let (x0, y0) = (x.floor() as usize, y.floor() as usize);
277    let (x1, y1) = ((x0 + 1).min(w - 1), (y0 + 1).min(h - 1));
278    let (fx, fy) = (x - x0 as f32, y - y0 as f32);
279    let px = |x: usize, y: usize| {
280        let c = t.pixels[y * w + x];
281        [c.r(), c.g(), c.b(), c.a()].map(|v| v as f32 / 255.0)
282    };
283    let (a, b, c, d) = (px(x0, y0), px(x1, y0), px(x0, y1), px(x1, y1));
284    let mut out = [0.0; 4];
285    for i in 0..4 {
286        out[i] =
287            (a[i] * (1.0 - fx) + b[i] * fx) * (1.0 - fy) + (c[i] * (1.0 - fx) + d[i] * fx) * fy;
288    }
289    out
290}

296    fn png(path: &Path) -> (u32, u32, Vec<u8>) {
297        let decoder =
298            png::Decoder::new(std::io::BufReader::new(std::fs::File::open(path).unwrap()));
299        let mut reader = decoder.read_info().unwrap();
300        let mut buf = vec![0; reader.output_buffer_size().unwrap()];
301        let info = reader.next_frame(&mut buf).unwrap();
302        buf.truncate(info.buffer_size());
303        (info.width, info.height, buf)
304    }
305
306    /// How many pixels are near `rgb` (within a small tolerance).
307    fn count_near(data: &[u8], rgb: [u8; 3]) -> usize {
308        data.chunks(3)
309            .filter(|p| (0..3).all(|i| (p[i] as i32 - rgb[i] as i32).abs() <= 12))
310            .count()
311    }

314    fn renders_against_the_mock_archiver() {
315        let mock = mockarchiver::MockArchiver::start_local().unwrap();
316        let dir = std::env::temp_dir().join(format!("archiveview-snap-{}", std::process::id()));
317        std::fs::create_dir_all(&dir).unwrap();
318        let out = dir.join("mock.png");
319        let app = AppOptions {
320            archiver: Some(archiver::ArchiverClient::new(&mock.url()).unwrap()),
321            pvs: vec!["MOCK:SINE".into(), "MOCK:NAN".into()],
322            ..Default::default()
323        };
324        let opts = Options {
325            size: [800, 500],
326            wait: 0.2,
327            ..Default::default()
328        };
329        run(app, &opts, &out).unwrap();
330        let (w, h, data) = png(&out);
331        assert_eq!((w, h), (800, 500));
332        // The first two palette colours were drawn (the traces).
333        assert!(count_near(&data, [31, 119, 180]) > 200, "a blue trace");
334        assert!(count_near(&data, [214, 39, 40]) > 200, "a red trace");
335
336        // The empty state renders too, dark.
337        let empty = dir.join("empty.png");
338        let opts = Options {
339            size: [640, 400],
340            wait: 0.0,
341            dark: true,
342            ..Default::default()
343        };
344        run(AppOptions::default(), &opts, &empty).unwrap();
345        assert!(std::fs::metadata(&empty).unwrap().len() > 1000);
346        std::fs::remove_dir_all(&dir).ok();
347    }
