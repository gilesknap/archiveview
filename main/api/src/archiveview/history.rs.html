1//! Recent searches: the search box's terms that were useful, newest first.
2//!
3//! A term is remembered when a PV is added from its results (or the typed
4//! text itself is added), so typos and half-typed prefixes are not. The
5//! list is plain data: [`crate::Plot`] keeps one in memory, and the
6//! standalone app loads and saves it as `history.json` (see
7//! `config::load_history`). A host embedding the plot may persist it the
8//! same way, or not at all.
9
10use serde::{Deserialize, Serialize};
11
12/// The most searches kept.
13pub const MAX_SEARCHES: usize = 50;
14
15/// A remembered search: the text as typed, and whether it was a regex.
16#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
17pub struct RecentSearch {
18    pub text: String,
19    #[serde(default, skip_serializing_if = "is_false")]
20    pub regex: bool,
21}
22
23fn is_false(b: &bool) -> bool {
24    !*b
25}
26
27impl RecentSearch {
28    /// Whether it is the same search as `other`. Plain text is searched
29    /// ignoring case, so `bl16i` and `BL16I` are one search; a glob (`*`
30    /// or `?`) or a regex is matched with case, so its case counts.
31    fn same(&self, other: &RecentSearch) -> bool {
32        self.regex == other.regex
33            && (self.text == other.text
34                || (!self.regex
35                    && !self.text.contains(['*', '?'])
36                    && self.text.to_lowercase() == other.text.to_lowercase()))
37    }
38}
39
40/// The recent searches, most recent first, without duplicates, at most
41/// [`MAX_SEARCHES`]. Also the shape of `history.json`; however it is
42/// built or read, it is cleaned up the same way ([`SearchHistory::new`]).
43#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
44#[serde(from = "HistoryFile")]
45pub struct SearchHistory {
46    searches: Vec<RecentSearch>,
47}
48
49/// `history.json` as read, before it is cleaned up.
50#[derive(Deserialize)]
51struct HistoryFile {
52    #[serde(default)]
53    searches: Vec<RecentSearch>,
54}
55
56impl From<HistoryFile> for SearchHistory {
57    fn from(file: HistoryFile) -> Self {
58        Self::new(file.searches)
59    }
60}
61
62impl SearchHistory {
63    /// A history of `searches`, most recent first; blank and repeated
64    /// entries are dropped, and any past [`MAX_SEARCHES`].
65    pub fn new(searches: Vec<RecentSearch>) -> Self {
66        let mut h = Self::default();
67        for s in searches.into_iter().rev() {
68            h.record(&s.text, s.regex);
69        }
70        h
71    }
72
73    /// The searches, most recent first.
74    pub fn searches(&self) -> &[RecentSearch] {
75        &self.searches
76    }
77
78    pub fn is_empty(&self) -> bool {
79        self.searches.is_empty()
80    }
81
82    /// Remember `text` (trimmed) as the most recent search, moving it to
83    /// the top if it is already there (in the newest spelling, for plain
84    /// text in another case). Returns whether the list changed.
85    pub fn record(&mut self, text: &str, regex: bool) -> bool {
86        let text = text.trim();
87        if text.is_empty() {
88            return false;
89        }
90        let entry = RecentSearch {
91            text: text.to_string(),
92            regex,
93        };
94        if self.searches.first() == Some(&entry) {
95            return false;
96        }
97        self.searches.retain(|s| !s.same(&entry));
98        self.searches.insert(0, entry);
99        self.searches.truncate(MAX_SEARCHES);
100        true
101    }
102
103    /// Forget one search. Returns whether it was there.
104    pub fn forget(&mut self, search: &RecentSearch) -> bool {
105        let before = self.searches.len();
106        self.searches.retain(|s| s != search);
107        self.searches.len() != before
108    }
109
110    pub fn clear(&mut self) {
111        self.searches.clear();
112    }
113
114    /// The searches offered while `text` is typed: those starting with it,
115    /// then those containing it (ignoring case), each most recent first.
116    /// Not `text` itself, nor anything with the box empty.
117    pub fn matching(&self, text: &str) -> Vec<&RecentSearch> {
118        let text = text.trim().to_lowercase();
119        if text.is_empty() {
120            return Vec::new();
121        }
122        let (mut starts, mut contains) = (Vec::new(), Vec::new());
123        for s in &self.searches {
124            let lower = s.text.to_lowercase();
125            if lower == text {
126                continue;
127            } else if lower.starts_with(&text) {
128                starts.push(s);
129            } else if lower.contains(&text) {
130                contains.push(s);
131            }
132        }
133        starts.extend(contains);
134        starts
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    fn texts(h: &SearchHistory) -> Vec<&str> {
143        h.searches().iter().map(|s| s.text.as_str()).collect()
144    }
145
146    #[test]
147    fn newest_first_without_duplicates() {
148        let mut h = SearchHistory::default();
149        assert!(h.record("BL16I", false));
150        assert!(h.record(" SR-DI ", false));
151        assert!(!h.record("SR-DI", false), "already the newest");
152        assert!(!h.record("  ", false));
153        assert_eq!(texts(&h), ["SR-DI", "BL16I"]);
154        // A repeat moves to the top; the regex flag makes it another search.
155        assert!(h.record("BL16I", false));
156        assert_eq!(texts(&h), ["BL16I", "SR-DI"]);
157        assert!(h.record("BL16I", true));
158        assert_eq!(texts(&h), ["BL16I", "BL16I", "SR-DI"]);
159        assert!(h.searches()[0].regex);
160
161        assert!(h.forget(&RecentSearch {
162            text: "BL16I".into(),
163            regex: true
164        }));
165        assert_eq!(texts(&h), ["BL16I", "SR-DI"]);
166        assert!(!h.searches()[0].regex);
167        // Plain text ignores case, keeping the newest spelling; a glob's
168        // case counts.
169        assert!(h.record("bl16i", false));
170        assert_eq!(texts(&h), ["bl16i", "SR-DI"]);
171        h.record("BL*", false);
172        h.record("bl*", false);
173        assert_eq!(texts(&h), ["bl*", "BL*", "bl16i", "SR-DI"]);
174        h.clear();
175        assert!(h.is_empty());
176    }

178    #[test]
179    fn capped_at_the_most_recent() {
180        let mut h = SearchHistory::default();
181        for i in 0..MAX_SEARCHES + 10 {
182            h.record(&format!("PV{i}"), false);
183        }
184        assert_eq!(h.searches().len(), MAX_SEARCHES);
185        assert_eq!(h.searches()[0].text, format!("PV{}", MAX_SEARCHES + 9));
186        assert_eq!(h.searches().last().unwrap().text, "PV10");
187        // As loaded: cleaned up the same way, order kept.
188        let loaded = SearchHistory::new(
189            ["b", "a", "b", " ", "c"]
190                .iter()
191                .map(|t| RecentSearch {
192                    text: t.to_string(),
193                    regex: false,
194                })
195                .collect(),
196        );
197        assert_eq!(texts(&loaded), ["b", "a", "c"]);
198        // Deserialised by a host: cleaned up too.
199        let read: SearchHistory = serde_json::from_str(
200            r#"{"searches":[{"text":"a"},{"text":"A"},{"text":" "},{"text":"b","regex":true}]}"#,
201        )
202        .unwrap();
203        assert_eq!(texts(&read), ["a", "b"]);
204        assert_eq!(serde_json::from_str::<SearchHistory>("{}").unwrap(), none());
205    }

211    #[test]
212    fn matches_start_then_contain() {
213        let mut h = SearchHistory::default();
214        for t in ["xbl16i", "BL16I-EA-*", "SR-DI", "bl16i", "BL16I-MO"] {
215            h.record(t, false);
216        }
217        let m: Vec<&str> = h
218            .matching("BL16I")
219            .iter()
220            .map(|s| s.text.as_str())
221            .collect();
222        assert_eq!(m, ["BL16I-MO", "BL16I-EA-*", "xbl16i"]);
223        assert!(h.matching(" ").is_empty());
224    }
