8/// A font family registered by [`install_fonts`]; regular is egui's
9/// proportional.
10pub const BOLD: &str = "bold";
11const ITALIC: &str = "italic";
12const BOLD_ITALIC: &str = "bold-italic";
13
14/// Install Liberation Sans (the font Phoebus and bobview use) ahead of
15/// egui's built-in fonts, which stay as fallbacks for symbols. Only the
16/// standalone app calls this: an embedded plot uses its host's fonts.
17pub fn install_fonts(ctx: &egui::Context) {
18    let mut fonts = egui::FontDefinitions::default();
19    let mut add = |key: &str, data: &'static [u8]| {
20        fonts
21            .font_data
22            .insert(key.into(), Arc::new(egui::FontData::from_static(data)));
23        key.to_string()
24    };
25    let regular = add(
26        "liberation",
27        include_bytes!("../fonts/LiberationSans-Regular.ttf"),
28    );
29    let bold = add(
30        "liberation-bold",
31        include_bytes!("../fonts/LiberationSans-Bold.ttf"),
32    );
33    let italic = add(
34        "liberation-italic",
35        include_bytes!("../fonts/LiberationSans-Italic.ttf"),
36    );
37    let bold_italic = add(
38        "liberation-bold-italic",
39        include_bytes!("../fonts/LiberationSans-BoldItalic.ttf"),
40    );
41    let symbols = add(
42        "noto-math-symbols",
43        include_bytes!("../fonts/NotoSansMath-Symbols.ttf"),
44    );
45
46    let mut fallback = fonts.families[&FontFamily::Proportional].clone();
47    fallback.push(symbols.clone());
48    fonts
49        .families
50        .entry(FontFamily::Monospace)
51        .or_default()
52        .push(symbols);
53    let with = |first: String| {
54        std::iter::once(first)
55            .chain(fallback.iter().cloned())
56            .collect::<Vec<_>>()
57    };
58    fonts
59        .families
60        .insert(FontFamily::Name(BOLD.into()), with(bold));
61    fonts
62        .families
63        .insert(FontFamily::Name(ITALIC.into()), with(italic));
64    fonts
65        .families
66        .insert(FontFamily::Name(BOLD_ITALIC.into()), with(bold_italic));
67    fonts
68        .families
69        .insert(FontFamily::Proportional, with(regular));
70    ctx.set_fonts(fonts);
71}

73/// A bold font of `size`, when the app installed one; else egui's default
74/// (an embedding host may not have [`BOLD`]).
75pub fn bold(ctx: &egui::Context, size: f32) -> egui::FontId {
76    let family = FontFamily::Name(BOLD.into());
77    let has = ctx.fonts(|f| f.families().contains(&family));
78    egui::FontId::new(
79        size,
80        if has {
81            family
82        } else {
83            FontFamily::Proportional
84        },
85    )
86}

88/// Trace colours, in the order PVs are added: bobview's sequence (blue,
89/// red, green, orange, purple, teal, brown, pink, grey, olive) in the
90/// Tableau 10 shades, which read well on white and on dark grey.
91const PALETTE: &[[u8; 3]] = &[
92    [31, 119, 180],
93    [214, 39, 40],
94    [44, 160, 44],
95    [255, 127, 14],
96    [148, 103, 189],
97    [23, 190, 207],
98    [140, 86, 75],
99    [227, 119, 194],
100    [127, 127, 127],
101    [188, 189, 34],
102];

104/// The `i`th palette colour, for a light or dark background.
105pub fn palette(i: usize, dark: bool) -> Color32 {
106    let [r, g, b] = PALETTE[i % PALETTE.len()];
107    adapt(Color32::from_rgb(r, g, b), dark)
108}
109
110/// A colour as drawn on a dark background: moved 22% of the way to white
111/// so it keeps its contrast.
112pub fn adapt(c: Color32, dark: bool) -> Color32 {
113    if !dark {
114        return c;
115    }
116    let up = |v: u8| v + ((255 - v) as f32 * 0.22) as u8;
117    Color32::from_rgb(up(c.r()), up(c.g()), up(c.b()))
118}

120/// `#rrggbb` (or `rrggbb`) as a colour.
121pub fn parse_hex(text: &str) -> Option<Color32> {
122    let h = text.trim().trim_start_matches('#');
123    if h.len() != 6 || !h.is_ascii() {
124        return None;
125    }
126    let n = |i: usize| u8::from_str_radix(&h[i..i + 2], 16).ok();
127    Some(Color32::from_rgb(n(0)?, n(2)?, n(4)?))
128}

130/// The alarm colours of the Phoebus palette (MINOR, MAJOR, INVALID).
131pub fn severity(sev: pv::Severity) -> Option<Color32> {
132    match sev {
133        pv::Severity::NoAlarm => None,
134        pv::Severity::Minor => Some(Color32::from_rgb(255, 128, 0)),
135        pv::Severity::Major => Some(Color32::from_rgb(255, 0, 0)),
136        pv::Severity::Invalid => Some(Color32::from_rgb(255, 0, 255)),
137    }
138}

140/// The "live" accent: the Live pill's dot (only; the now marker is grey, as
141/// red is also a trace and an alarm colour).
142pub fn live(dark: bool) -> Color32 {
143    if dark {
144        Color32::from_rgb(255, 99, 99)
145    } else {
146        Color32::from_rgb(220, 38, 38)
147    }
148}

150/// The archiver pill's dot when the archiver answers.
151pub fn reachable(dark: bool) -> Color32 {
152    if dark {
153        Color32::from_rgb(74, 222, 128)
154    } else {
155        Color32::from_rgb(22, 163, 74)
156    }
157}

159/// The archiver pill's dot when the archiver fails: amber, so it is not
160/// taken for the Live pill's red dot beside it.
161pub fn unreachable(dark: bool) -> Color32 {
162    if dark {
163        Color32::from_rgb(251, 191, 36)
164    } else {
165        Color32::from_rgb(217, 119, 6)
166    }

174    fn colours_parse_and_cycle() {
175        assert_eq!(parse_hex("#d62728"), Some(Color32::from_rgb(214, 39, 40)));
176        assert_eq!(parse_hex("1f77b4"), Some(Color32::from_rgb(31, 119, 180)));
177        assert_eq!(parse_hex("#12345"), None);
178        assert_eq!(parse_hex("#gg0000"), None);
179        assert_eq!(palette(0, false), palette(10, false));
180        let (l, d) = (palette(0, false), palette(0, true));
181        assert!(
182            d.r() > l.r() && d.g() > l.g() && d.b() > l.b(),
183            "lighter on dark"
184        );
185    }

187    /// Every character outside ASCII in the code (not its comments): the
188    /// symbols the interface shows, with where each is used.
189    fn ui_glyphs() -> Vec<(char, String)> {
190        fn walk(dir: &std::path::Path, out: &mut Vec<std::path::PathBuf>) {
191            for e in std::fs::read_dir(dir).unwrap().flatten() {
192                let p = e.path();
193                if p.is_dir() {
194                    walk(&p, out);
195                } else if p.extension().is_some_and(|x| x == "rs") {
196                    out.push(p);
197                }
198            }
199        }
200        let mut files = Vec::new();
201        walk(
202            &std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("src"),
203            &mut files,
204        );
205        let mut glyphs: Vec<(char, String)> = Vec::new();
206        for f in files {
207            let text = std::fs::read_to_string(&f).unwrap();
208            for (n, line) in text.lines().enumerate() {
209                // The code before a comment (a URL's "//" is in a string,
210                // and has no symbols after it worth missing).
211                let code = line.split("//").next().unwrap_or("");
212                for c in code.chars().filter(|c| !c.is_ascii()) {
213                    if !glyphs.iter().any(|(g, _)| *g == c) {
214                        let at = f.file_name().unwrap().to_string_lossy();
215                        glyphs.push((c, format!("{at}:{}", n + 1)));
216                    }
217                }
218            }
219        }
220        glyphs
221    }

223    /// A glyph missing from the fonts is drawn as a box: every symbol the
224    /// interface uses must be in the fonts [`install_fonts`] installs (the
225    /// proportional ones: monospace shows only ".*").
226    /// (bobview, which embeds the plot, installs the same fallbacks; egui's
227    /// own fonts alone lack arrows such as → and ⏎.)
228    #[test]
229    fn every_symbol_shown_is_in_the_fonts() {
230        let glyphs = ui_glyphs();
231        assert!(
232            glyphs.iter().any(|(c, _)| *c == '…'),
233            "the scan finds symbols"
234        );
235        let ctx = egui::Context::default();
236        install_fonts(&ctx);
237        // Fonts are loaded on the first frame.
238        let mut out = ctx.run_ui(egui::RawInput::default(), |_| {});
239        out.textures_delta.clear();
240        for font in [egui::FontId::proportional(13.0), bold(&ctx, 13.0)] {
241            let missing: Vec<String> = ctx.fonts_mut(|f| {
242                (glyphs.iter())
243                    .filter(|(c, _)| !f.has_glyph(&font, *c))
244                    .map(|(c, at)| format!("{c} (U+{:04X}, {at})", *c as u32))
245                    .collect()
246            });
247            assert!(missing.is_empty(), "not in {font:?}: {missing:?}");
248        }
249    }
