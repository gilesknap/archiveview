1//! Settings and saved views in `$XDG_CONFIG_HOME/archiveview` (by default
2//! `~/.config/archiveview`), the same way bobview keeps `~/.config/bobview`:
3//!
4//! - `settings.json`: [`Settings`].
5//! - `views/<name>.json`: a [`View`]; `views/default.json` opens at startup
6//!   when no PVs or `--view` are given.
7//!
8//! Files are pretty JSON written atomically (hidden temp file, then rename).
9//! Only the standalone app uses this module; an embedding app keeps
10//! [`View`]s in its own config. (Adapted from bobview's `layout.rs`.)
11
12use std::path::{Path, PathBuf};

14use serde::{Deserialize, Serialize};

16use crate::view::View;

18/// The view restored at startup.
19pub const DEFAULT_VIEW: &str = "default";
20
21/// `$XDG_CONFIG_HOME/archiveview` when that is set and non-empty, else
22/// `$HOME/.config/archiveview`; `None` with neither.
23pub fn config_dir() -> Option<PathBuf> {
24    let base = std::env::var_os("XDG_CONFIG_HOME")
25        .filter(|d| !d.is_empty())
26        .map(PathBuf::from)
27        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".config")))?;
28    Some(base.join("archiveview"))
29}

31/// The colour scheme.
32#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
33#[serde(rename_all = "lowercase")]
34pub enum Theme {
35    /// Follow the OS (eframe's system theme).
36    #[default]
37    System,
38    Light,
39    Dark,
40}

42/// `settings.json`. Every field defaults, so old and partial files load.
43#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
44pub struct Settings {
45    /// Used when `--archiver` is not given.
46    #[serde(default, skip_serializing_if = "Option::is_none")]
47    pub archiver_url: Option<String>,
48    #[serde(default)]
49    pub theme: Theme,
50    /// Save the current view as `default` when the window closes, so the
51    /// next start reopens where this one left off (on unless turned off).
52    #[serde(default = "yes")]
53    pub save_default_on_exit: bool,
54}
55
56fn yes() -> bool {
57    true
58}
59
60impl Default for Settings {
61    fn default() -> Self {
62        Self {
63            archiver_url: None,
64            theme: Theme::default(),
65            save_default_on_exit: true,
66        }
67    }
68}
69
70impl Settings {
71    /// From `settings.json`, or the defaults when it is missing or bad
72    /// (logging a warning for a bad one).
73    pub fn load() -> Self {
74        config_dir().map_or_else(Self::default, |d| Self::load_from(&d))
75    }
76
77    /// Write `settings.json`, logging a warning on failure.
78    pub fn save(&self) {
79        if let Some(d) = config_dir() {
80            self.save_to(&d);
81        }
82    }
83
84    fn load_from(dir: &Path) -> Self {
85        let path = dir.join("settings.json");
86        let Ok(text) = std::fs::read_to_string(&path) else {
87            return Self::default();
88        };
89        serde_json::from_str(&text).unwrap_or_else(|e| {
90            log::warn!("{}: {e}: using the default settings", path.display());
91            Self::default()
92        })
93    }
94
95    fn save_to(&self, dir: &Path) {
96        let written = serde_json::to_string_pretty(self)
97            .map_err(std::io::Error::other)
98            .and_then(|text| write_atomic(&dir.join("settings.json"), &(text + "\n")));
99        if let Err(e) = written {
100            log::warn!("cannot save settings in {}: {e}", dir.display());
101        }
102    }
103}

105/// A name usable as a view file name: non-empty, no leading dot, no `/` or `\`.
106pub fn valid_name(name: &str) -> bool {
107    let name = name.trim();
108    !name.is_empty() && !name.starts_with('.') && !name.contains(['/', '\\'])
109}
110
111/// The saved views' names: `default` first (when saved), the rest sorted.
112pub fn list_views() -> Vec<String> {
113    config_dir().map_or_else(Vec::new, |d| Views(d).list())
114}
115
116/// Whether a view called `name` is saved.
117pub fn view_exists(name: &str) -> bool {
118    config_dir().is_some_and(|d| Views(d).exists(name))
119}
120
121/// Load a saved view; refuses a newer `version` than this program's.
122pub fn load_view(name: &str) -> Result<View, String> {
123    Views(config_dir().ok_or("no home directory")?).load(name)
124}
125
126pub fn save_view(name: &str, view: &View) -> Result<(), String> {
127    Views(config_dir().ok_or("no home directory")?).save(name, view)
128}
129
130pub fn delete_view(name: &str) -> Result<(), String> {
131    Views(config_dir().ok_or("no home directory")?).delete(name)
132}

134/// The views saved under a config directory.
135struct Views(PathBuf);
136
137impl Views {
138    fn file(&self, name: &str) -> Option<PathBuf> {
139        valid_name(name).then(|| self.0.join("views").join(format!("{}.json", name.trim())))
140    }
141
142    fn list(&self) -> Vec<String> {
143        let mut names: Vec<String> = std::fs::read_dir(self.0.join("views"))
144            .map(|rd| {
145                rd.filter_map(Result::ok)
146                    .filter_map(|e| {
147                        let p = e.path();
148                        (p.extension()? == "json")
149                            .then(|| p.file_stem()?.to_str().map(str::to_string))
150                            .flatten()
151                    })
152                    .filter(|n| !n.starts_with('.'))
153                    .collect()
154            })
155            .unwrap_or_default();
156        names.sort_by(|a, b| (a != DEFAULT_VIEW).cmp(&(b != DEFAULT_VIEW)).then(a.cmp(b)));
157        names
158    }
159
160    fn exists(&self, name: &str) -> bool {
161        self.file(name).is_some_and(|f| f.is_file())
162    }
163
164    fn load(&self, name: &str) -> Result<View, String> {
165        let path = self.file(name).ok_or_else(|| format!("no view {name:?}"))?;
166        let text =
167            std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
168        let view: View =
169            serde_json::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))?;
170        if view.version > crate::view::VERSION {
171            return Err(format!(
172                "{}: view version {} is newer than this archiveview's {}",
173                path.display(),
174                view.version,
175                crate::view::VERSION
176            ));
177        }
178        Ok(view)
179    }
180
181    fn save(&self, name: &str, view: &View) -> Result<(), String> {
182        let path = self
183            .file(name)
184            .ok_or_else(|| format!("not a view name: {name:?}"))?;
185        let write = || -> std::io::Result<()> {
186            let text = serde_json::to_string_pretty(view).map_err(std::io::Error::other)?;
187            write_atomic(&path, &(text + "\n"))
188        };
189        write().map_err(|e| format!("{}: {e}", path.display()))?;
190        log::info!("saved view {}", path.display());
191        Ok(())
192    }
193
194    fn delete(&self, name: &str) -> Result<(), String> {
195        let path = self.file(name).ok_or_else(|| format!("no view {name:?}"))?;
196        std::fs::remove_file(&path).map_err(|e| format!("{}: {e}", path.display()))
197    }
198}

200/// Write `text` to `path` through a hidden file beside it, so a crash or a
201/// full disk leaves the old file whole rather than half written.
202fn write_atomic(path: &Path, text: &str) -> std::io::Result<()> {
203    let dir = path.parent().unwrap_or(Path::new("."));
204    std::fs::create_dir_all(dir)?;
205    let name = path.file_name().unwrap_or_default().to_string_lossy();
206    // Hidden and not .json, so never listed as a view.
207    let tmp = dir.join(format!(".{name}.{}.tmp", std::process::id()));
208    std::fs::write(&tmp, text)
209        .and_then(|()| std::fs::rename(&tmp, path))
210        .inspect_err(|_| {
211            std::fs::remove_file(&tmp).ok();
212        })
213}

215#[cfg(test)]
216mod tests {
217    use super::*;
218    use crate::view::{Range, TraceView};

220    fn temp(name: &str) -> PathBuf {
221        let dir = std::env::temp_dir().join(format!("archiveview-{name}-{}", std::process::id()));
222        std::fs::remove_dir_all(&dir).ok();
223        dir
224    }
225
226    #[test]
227    fn settings_round_trip() {
228        let dir = temp("settings");
229        assert_eq!(Settings::load_from(&dir), Settings::default());
230        let s = Settings {
231            archiver_url: Some("http://archiver:17668".into()),
232            theme: Theme::Dark,
233            save_default_on_exit: false,
234        };
235        s.save_to(&dir);
236        assert_eq!(Settings::load_from(&dir), s);
237        let text = std::fs::read_to_string(dir.join("settings.json")).unwrap();
238        assert!(text.contains("\"theme\": \"dark\""), "{text}");
239        // Partial files load; bad ones give the defaults.
240        std::fs::write(dir.join("settings.json"), r#"{"theme":"light"}"#).unwrap();
241        assert_eq!(Settings::load_from(&dir).theme, Theme::Light);
242        assert!(
243            Settings::load_from(&dir).save_default_on_exit,
244            "on by default"
245        );
246        std::fs::write(dir.join("settings.json"), "{ bad").unwrap();
247        assert_eq!(Settings::load_from(&dir), Settings::default());
248        std::fs::remove_dir_all(&dir).ok();
249    }
250
251    #[test]
252    fn views_round_trip() {
253        let dir = temp("views");
254        let views = Views(dir.clone());
255        assert!(views.list().is_empty());
256        let view = View {
257            pvs: vec![TraceView {
258                name: "MOCK:SINE".into(),
259                color: None,
260                visible: true,
261            }],
262            range: Range::Relative { secs: 86_400.0 },
263            ..View::default()
264        };
265        views.save("beam", &view).unwrap();
266        views.save(DEFAULT_VIEW, &View::default()).unwrap();
267        views.save("alpha", &View::default()).unwrap();
268        assert_eq!(views.list(), ["default", "alpha", "beam"]);
269        assert!(views.exists("beam") && !views.exists("nope"));
270        assert_eq!(views.load("beam").unwrap(), view);
271        assert!(views.save("../x", &view).is_err());
272        assert!(views.save(".hidden", &view).is_err());
273        assert!(views.load("nope").is_err());
274        // No temp files left behind.
275        let files = std::fs::read_dir(dir.join("views")).unwrap().count();
276        assert_eq!(files, 3);
277
278        // A newer version is refused.
279        let newer = View {
280            version: crate::view::VERSION + 1,
281            ..View::default()
282        };
283        views.save("newer", &newer).unwrap();
284        assert!(views.load("newer").unwrap_err().contains("newer"));
285
286        views.delete("beam").unwrap();
287        assert_eq!(views.list(), ["default", "alpha", "newer"]);
288        assert!(views.delete("beam").is_err());
289        std::fs::remove_dir_all(&dir).ok();
290    }
