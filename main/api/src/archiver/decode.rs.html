1//! Decoding response bodies. Tolerant by design: servers differ in
2//! version and in how they serialise, so unknown keys are ignored, numbers
3//! may arrive as strings, and odd samples are skipped rather than failing
4//! the whole response.

12/// Parse JSON after replacing bare `NaN`/`Infinity` tokens.
13fn json(body: &str) -> Result<Value> {
14    serde_json::from_str(&sanitize_json(body)).map_err(|e| {
15        let head: String = body.trim_start().chars().take(80).collect();
16        Error::Parse(format!("{e} (body starts {head:?})"))
17    })
18}

20/// A value as text: strings as they are, numbers and booleans printed.
21fn text(v: &Value) -> Option<String> {
22    match v {
23        Value::String(s) => Some(s.clone()),
24        Value::Number(n) => Some(n.to_string()),
25        Value::Bool(b) => Some(b.to_string()),
26        _ => None,
27    }
28}
29
30/// A value as a number: numbers, or strings that parse as one.
31fn number(v: &Value) -> Option<f64> {
32    match v {
33        Value::Number(n) => n.as_f64(),
34        Value::String(s) => s.trim().parse().ok(),
35        _ => None,
36    }
37}
38
39/// A count or precision: a non-negative whole number.
40fn whole(v: &Value) -> Option<u32> {
41    number(v)
42        .filter(|n| n.is_finite() && *n >= 0.0 && n.fract() == 0.0 && *n <= u32::MAX as f64)
43        .map(|n| n as u32)
44}
45
46fn non_empty(s: Option<String>) -> Option<String> {
47    s.filter(|s| !s.trim().is_empty())
48}

50/// `getData.json`. `bins` allows 5-element array values to be read as
51/// `optimized` bins.
52pub(crate) fn data(body: &str, bins: bool) -> Result<Vec<TimeSeries>> {
53    if body.trim().is_empty() {
54        return Ok(Vec::new());
55    }
56    let items = match json(body)? {
57        Value::Array(items) => items,
58        // A single object rather than a one-element array.
59        v @ Value::Object(_) => vec![v],
60        Value::Null => Vec::new(),
61        _ => return Err(Error::Parse("expected a JSON array".into())),
62    };
63    Ok(items
64        .iter()
65        .filter_map(Value::as_object)
66        .map(|obj| series(obj, bins))
67        .collect())
68}

70fn meta(obj: Option<&Map<String, Value>>) -> Meta {
71    let mut m = Meta::default();
72    for (k, v) in obj.into_iter().flatten() {
73        match k.as_str() {
74            "name" => m.name = text(v).unwrap_or_default(),
75            "EGU" => m.egu = non_empty(text(v)),
76            "PREC" => m.prec = whole(v),
77            "LOPR" => m.lopr = number(v),
78            "HOPR" => m.hopr = number(v),
79            "DESC" => m.desc = non_empty(text(v)),
80            _ => {
81                if let Some(t) = text(v) {
82                    m.extra.insert(k.clone(), t);
83                }
84            }
85        }
86    }
87    m
88}

90/// What the `val`s say the series is, and whether the arrays are bins.
91fn classify(data: &[Value], meta: &Meta, bins: bool) -> (ValueKind, bool) {
92    let vals = || {
93        data.iter()
94            .filter_map(|s| s.get("val"))
95            .filter(|v| !v.is_null())
96    };
97    let kind = match vals().next() {
98        None | Some(Value::Number(_) | Value::Bool(_)) => ValueKind::Scalar,
99        Some(Value::String(_)) => ValueKind::Text,
100        Some(Value::Array(_)) => ValueKind::Waveform,
101        Some(_) => ValueKind::Other,
102    };
103    let waveform_hint = meta.extra.get("waveform").is_some_and(|w| w == "true");
104    let is_bin = |v: &Value| {
105        v.as_array().is_some_and(|a| {
106            a.len() == 5
107                && a[..4].iter().all(|x| x.is_null() || x.is_number())
108                && whole(&a[4]).is_some()
109        })
110    };
111    let binned = bins && kind == ValueKind::Waveform && !waveform_hint && vals().all(is_bin);
112    (if binned { ValueKind::Scalar } else { kind }, binned)
113}

115/// A JSON number (or numeric string) as f64; null and others are NaN.
116fn float(v: Option<&Value>) -> f64 {
117    match v {
118        Some(Value::Bool(b)) => f64::from(u8::from(*b)),
119        Some(v) => number(v).unwrap_or(f64::NAN),
120        None => f64::NAN,
121    }
122}

124fn series(obj: &Map<String, Value>, bins: bool) -> TimeSeries {
125    let meta = meta(obj.get("meta").and_then(Value::as_object));
126    let data: &[Value] = obj
127        .get("data")
128        .and_then(Value::as_array)
129        .map_or(&[], Vec::as_slice);
130    let (kind, binned) = classify(data, &meta, bins);
131    let mut samples: Vec<Sample> = Vec::with_capacity(data.len());
132    for s in data.iter().filter_map(Value::as_object) {
133        let Some(secs) = s.get("secs").and_then(number) else {
134            continue;
135        };
136        let nanos = s.get("nanos").and_then(number).unwrap_or(0.0);
137        let t = secs + nanos * 1e-9;
138        let severity = Severity::from_code(s.get("severity").and_then(Value::as_i64).unwrap_or(0));
139        let status = s
140            .get("status")
141            .and_then(Value::as_u64)
142            .map_or(0, |n| n.min(u16::MAX as u64) as u16);
143        let fields = s.get("fields").and_then(Value::as_object);
144        let field = |k: &str| fields.and_then(|f| f.get(k)).and_then(number);
145        let lost = field("cnxlostepsecs");
146
147        let val = s.get("val");
148        let mut sample = match (val, binned) {
149            (Some(Value::Array(a)), true) => {
150                let mean = float(a.first());
151                let (mut min, mut max) = (float(a.get(2)), float(a.get(3)));
152                if min.is_nan() || max.is_nan() {
153                    (min, max) = (mean, mean);
154                }
155                Sample {
156                    t,
157                    val: mean,
158                    min,
159                    max,
160                    count: a.get(4).and_then(whole).unwrap_or(0),
161                    severity,
162                    status,
163                }
164            }
165            (Some(Value::Array(a)), false) => Sample::raw(t, float(a.first())),
166            (Some(Value::String(_) | Value::Object(_)), _) => Sample::raw(t, f64::NAN),
167            (v, _) => Sample::raw(t, float(v)),
168        };
169        sample.severity = severity;
170        sample.status = status;
171
172        if severity == Severity::Invalid && lost.is_some() {
173            // The connection-lost event: the line breaks here.
174            sample = Sample {
175                status,
176                ..Sample::gap(t)
177            };
178        } else if let Some(lost) = lost {
179            // A first sample after reconnecting: break the line at the
180            // time the connection was lost, if nothing has yet.
181            let prev = samples.last();
182            if lost < t && prev.is_none_or(|p| p.t < lost && !p.is_gap()) {
183                samples.push(Sample::gap(lost));
184            }
185        }
186        samples.push(sample);
187    }
188    if !samples.is_sorted_by(|a, b| a.t <= b.t) {
189        samples.sort_by(|a, b| a.t.total_cmp(&b.t));
190    }
191    TimeSeries {
192        meta,
193        kind,
194        binned,
195        samples,
196    }
197}

199/// A JSON array of PV names.
200pub(crate) fn names(body: &str) -> Result<Vec<String>> {
201    match json(body)? {
202        Value::Array(items) => Ok(items.iter().filter_map(text).collect()),
203        Value::Null => Ok(Vec::new()),
204        _ => Err(Error::Parse("expected a JSON array of names".into())),
205    }
206}

208/// `getMetadata`: a flat object, values usually strings.
209pub(crate) fn metadata(body: &str) -> Result<PvInfo> {
210    let Value::Object(obj) = json(body)? else {
211        return Err(Error::Parse("expected a JSON object".into()));
212    };
213    let raw: BTreeMap<String, String> = obj
214        .iter()
215        .filter_map(|(k, v)| Some((k.clone(), text(v)?)))
216        .collect();
217    let get = |keys: &[&str]| {
218        keys.iter()
219            .find_map(|k| raw.get(*k).filter(|v| !v.trim().is_empty()))
220            .cloned()
221    };
222    let num = |keys: &[&str]| get(keys).and_then(|v| v.trim().parse::<f64>().ok());
223    let whole = |keys: &[&str]| {
224        num(keys)
225            .filter(|n| *n >= 0.0 && n.fract() == 0.0 && *n <= u32::MAX as f64)
226            .map(|n| n as u32)
227    };
228    Ok(PvInfo {
229        dbr_type: get(&["DBRType"]),
230        element_count: whole(&["elementCount"]),
231        units: get(&["units", "EGU"]),
232        precision: whole(&["precision", "PREC"]),
233        lower_display_limit: num(&["lowerDisplayLimit", "LOPR"]),
234        upper_display_limit: num(&["upperDisplayLimit", "HOPR"]),
235        raw,
236    })
