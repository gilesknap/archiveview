1//! A client for the EPICS Archiver Appliance retrieval API.
2//!
3//! The pieces are split so that a browser build could reuse everything but
4//! the HTTP: [`ArchiverClient`] builds the request URLs (`*_url`), the
5//! `parse_*` functions decode the bodies, and on native targets the
6//! blocking `get_data`/`search`/`metadata` methods do both over `ureq`.
7//!
8//! Times are UTC seconds since the Unix epoch as `f64` (sub-microsecond
9//! resolution for current dates), the same as bobview's trend and
10//! `archiveview`'s plot.
11//!
12//! See `DESIGN.md` at the repository root for the wire format and its
13//! surprises (bare `NaN` tokens, string `PREC`, `[ ]` for no data, binned
14//! `[mean, std, min, max, count]` values from `optimized_N`).
15

38/// Why a request failed.
39#[derive(Clone, Debug, PartialEq)]
40pub enum Error {
41    /// The archiver URL given could not be used.
42    BadUrl(String),
43    /// The server answered with a non-2xx status. `message` is the first
44    /// non-empty line of the body, with HTML tags stripped.
45    Http { status: u16, message: String },
46    /// No answer: connection refused, DNS, timeout, TLS, ...
47    Transport(String),
48    /// A 2xx answer that is not the expected JSON.
49    Parse(String),
50}
51
52impl Error {
53    /// True for the archiver's "PV is not archived" answer (HTTP 404).
54    pub fn is_not_archived(&self) -> bool {
55        matches!(self, Error::Http { status: 404, .. })
56    }
57}
58
59impl fmt::Display for Error {
60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        match self {
62            Error::BadUrl(m) => write!(f, "bad archiver URL: {m}"),
63            Error::Http { status, message } if message.is_empty() => write!(f, "HTTP {status}"),
64            Error::Http { status, message } => write!(f, "HTTP {status}: {message}"),
65            Error::Transport(m) => write!(f, "{m}"),
66            Error::Parse(m) => write!(f, "bad response: {m}"),
67        }
68    }
69}
70
71impl std::error::Error for Error {}
72
73pub type Result<T> = std::result::Result<T, Error>;

79/// An archived alarm severity (the `severity` field: 0..=3).
80#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
81pub enum Severity {
82    #[default]
83    NoAlarm,
84    Minor,
85    Major,
86    Invalid,
87}
88
89impl Severity {
90    /// From the archiver's number; anything above 3 counts as `Invalid`.
91    pub fn from_code(code: i64) -> Self {
92        match code {
93            ..=0 => Severity::NoAlarm,
94            1 => Severity::Minor,
95            2 => Severity::Major,
96            _ => Severity::Invalid,
97        }
98    }
99}

101/// One plotted point of archived data.
102///
103/// A raw sample has `min == max == val`. A bin from `optimized_N` has `val`
104/// = the bin mean, `min`/`max` its extremes and `count` its sample count;
105/// `t` is then the bin centre. `val` is NaN at a gap (a disconnect, a
106/// severity-INVALID connection-lost event, or an archived NaN), where the
107/// plot breaks the line.
108#[derive(Clone, Copy, Debug, PartialEq)]
109pub struct Sample {
110    /// UTC seconds since the epoch (`secs + nanos * 1e-9`).
111    pub t: f64,
112    pub val: f64,
113    pub min: f64,
114    pub max: f64,
115    /// Samples in the bin; 1 for a raw sample.
116    pub count: u32,
117    pub severity: Severity,
118    pub status: u16,
119}
120
121impl Sample {
122    /// A raw sample.
123    pub fn raw(t: f64, val: f64) -> Self {
124        Self {
125            t,
126            val,
127            min: val,
128            max: val,
129            count: 1,
130            severity: Severity::NoAlarm,
131            status: 0,
132        }
133    }
134
135    /// A line break at `t`.
136    pub fn gap(t: f64) -> Self {
137        Self {
138            severity: Severity::Invalid,
139            ..Self::raw(t, f64::NAN)
140        }
141    }
142
143    pub fn is_gap(&self) -> bool {
144        self.val.is_nan()
145    }
146}

148/// What the archived values were, as seen in the response.
149#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
150pub enum ValueKind {
151    /// Scalar numbers (doubles, integers or enum indices): plottable.
152    #[default]
153    Scalar,
154    /// Arrays: element 0 is plotted. Always fetch these raw.
155    Waveform,
156    /// Strings: not plotted in v1 (samples carry NaN).
157    Text,
158    /// PVA structures and anything else: not plotted.
159    Other,
160}

162/// The `meta` object of a response, with the common fields decoded. Values
163/// may arrive as strings or numbers; both are accepted.
164#[derive(Clone, Debug, Default, PartialEq)]
165pub struct Meta {
166    pub name: String,
167    /// `EGU`, absent when the PV has no units.
168    pub egu: Option<String>,
169    /// `PREC`.
170    pub prec: Option<u32>,
171    /// `LOPR`/`HOPR` (only with `fetchLatestMetadata=true`).
172    pub lopr: Option<f64>,
173    pub hopr: Option<f64>,
174    /// `DESC` (only with `fetchLatestMetadata=true`).
175    pub desc: Option<String>,
176    /// Every other key, as text.
177    pub extra: BTreeMap<String, String>,
178}

180/// One PV's archived data for a request.
181#[derive(Clone, Debug, Default, PartialEq)]
182pub struct TimeSeries {
183    pub meta: Meta,
184    pub kind: ValueKind,
185    /// True when the samples are `optimized_N` bins (`min`/`max` envelope
186    /// meaningful), false for raw samples.
187    pub binned: bool,
188    /// In time order. The first may be before the requested `from`: it is
189    /// the value that held at `from`.
190    pub samples: Vec<Sample>,
191}

197/// Server-side processing for a data request.
198#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
199pub enum Operator {
200    /// Every archived sample.
201    #[default]
202    Raw,
203    /// `optimized_N`: about N points; raw when there are at most N samples,
204    /// else `[mean, std, min, max, count]` bins.
205    Optimized(u32),
206    /// `mean_N`: the mean of each N-second bin.
207    Mean(u32),
208    /// `nth_N`: every Nth raw sample.
209    Nth(u32),
210}
211
212impl Operator {
213    /// The `pv` parameter for `pv`: `optimized_1200(SR-DI-DCCT-01:SIGNAL)`.
214    pub fn wrap(&self, pv: &str) -> String {
215        match self {
216            Operator::Raw => pv.to_string(),
217            Operator::Optimized(n) => format!("optimized_{n}({pv})"),
218            Operator::Mean(n) => format!("mean_{n}({pv})"),
219            Operator::Nth(n) => format!("nth_{n}({pv})"),
220        }
221    }
222
223    /// The operator to plot `width_px` device pixels: `optimized` with two
224    /// bins per pixel, at most [`MAX_BINS`].
225    pub fn for_width(width_px: f32) -> Self {
226        let n = (2.0 * width_px.max(1.0)).round() as u32;
227        Operator::Optimized(n.clamp(1, MAX_BINS))
228    }
229}

231/// A `getData.json` request.
232#[derive(Clone, Debug, PartialEq)]

241}
242
243impl Query {

255/// How a search pattern is matched.
256#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
257pub enum SearchMode {
258    /// `*` and `?` wildcards, sent as `pv=`.
259    #[default]
260    Glob,
261    /// A Java regex matching the whole name, sent as `regex=`.
262    Regex,
263}
264
265/// A `getMatchingPVs` request.
266#[derive(Clone, Debug, PartialEq)]
267pub struct Search {
268    pub pattern: String,
269    pub mode: SearchMode,
270    /// Names wanted at most; -1 for all.
271    pub limit: i32,
272}
273
274impl Search {
275    /// What the search box sends for `text`: a glob, wrapped as `*text*`
276    /// for a substring match when it has no `*` or `?`. Surrounding
277    /// whitespace is trimmed.
278    pub fn from_user(text: &str, limit: i32) -> Self {
279        let text = text.trim();
280        let pattern = if text.contains(['*', '?']) {
281            text.to_string()
282        } else {
283            format!("*{text}*")
284        };
285        Self {
286            pattern,
287            mode: SearchMode::Glob,
288            limit,
289        }
290    }
291
292    pub fn regex(re: impl Into<String>, limit: i32) -> Self {
293        Self {
294            pattern: re.into(),
295            mode: SearchMode::Regex,
296            limit,
297        }
298    }
299}

301/// What `getMetadata` says about a PV (all optional: servers differ).
302#[derive(Clone, Debug, Default, PartialEq)]
303pub struct PvInfo {
304    /// e.g. `DBR_SCALAR_DOUBLE`, `DBR_WAVEFORM_DOUBLE`, `DBR_SCALAR_ENUM`.
305    pub dbr_type: Option<String>,
306    pub element_count: Option<u32>,
307    pub units: Option<String>,
308    pub precision: Option<u32>,
309    pub lower_display_limit: Option<f64>,
310    pub upper_display_limit: Option<f64>,
311    /// Every key as returned.
312    pub raw: BTreeMap<String, String>,
313}
314
315impl PvInfo {
316    /// True when `optimized_N` makes sense: a scalar number (or enum).
317    /// Unknown types count as numeric.
318    pub fn is_scalar_numeric(&self) -> bool {
319        let scalar = self.element_count.is_none_or(|n| n <= 1);
320        let numeric = self
321            .dbr_type
322            .as_deref()
323            .is_none_or(|t| !t.contains("STRING") && !t.contains("V4"));
324        scalar && numeric
325    }
326
327    /// True when the plot should ask for `optimized_N` bins: a scalar
328    /// number that is not an enum. Bin means of enum state indices (1.63)
329    /// name no state, so enums are fetched raw, as are waveforms and text.
330    pub fn binnable(&self) -> bool {
331        self.is_scalar_numeric() && self.dbr_type.as_deref().is_none_or(|t| !t.contains("ENUM"))
332    }
333}

339/// An Archiver Appliance retrieval service. Cheap to clone; holds no
340/// connection state, so it can be shared freely between threads and
341/// between several plots.
342#[derive(Clone, Debug, PartialEq, Eq)]
343pub struct ArchiverClient {
344    /// The normalised base, ending in `/retrieval` with no trailing slash.
345    base: String,
346    /// Per-request timeout in seconds (native only).
347    timeout_secs: u64,
348}
349
350impl ArchiverClient {
351    /// From the URL the user gave, e.g. `http://host:17668` or
352    /// `http://host:17668/retrieval/`: strips trailing slashes and appends
353    /// `/retrieval` unless it already ends so. Only `http` and `https` are
354    /// accepted; a bare `host:port` gets `http://`.
355    pub fn new(url: &str) -> Result<Self> {
356        let mut url = url.trim().trim_end_matches('/').to_string();
357        if url.is_empty() {
358            return Err(Error::BadUrl("empty".into()));
359        }
360        if !url.contains("://") {
361            url = format!("http://{url}");
362        }
363        if !(url.starts_with("http://") || url.starts_with("https://")) {
364            return Err(Error::BadUrl(format!(
365                "{url}: only http and https are supported"
366            )));
367        }
368        if !url.ends_with("/retrieval") {
369            url.push_str("/retrieval");
370        }
371        Ok(Self {
372            base: url,
373            timeout_secs: 30,
374        })
375    }
376
377    pub fn with_timeout_secs(mut self, secs: u64) -> Self {
378        self.timeout_secs = secs;
379        self
380    }
381
382    /// The normalised retrieval base, e.g. `http://host:17668/retrieval`.
383    pub fn base(&self) -> &str {
384        &self.base
385    }
386
387    /// `getData.json` with `from`/`to` as `YYYY-MM-DDTHH:MM:SS.mmmZ` and
388    /// every parameter percent-encoded.
389    pub fn data_url(&self, q: &Query) -> String {
390        let mut url = format!(
391            "{}/data/getData.json?pv={}&from={}&to={}",
392            self.base,
393            encode(&q.op.wrap(&q.pv)),
394            encode(&format_time(q.from)),
395            encode(&format_time(q.to)),
396        );
397        if q.fetch_latest_metadata {
398            url.push_str("&fetchLatestMetadata=true");
399        }
400        url
401    }
402
403    /// `bpl/getMatchingPVs?pv=...&limit=N` or `?regex=...&limit=N`.
404    pub fn search_url(&self, s: &Search) -> String {
405        let key = match s.mode {
406            SearchMode::Glob => "pv",
407            SearchMode::Regex => "regex",
408        };
409        format!(
410            "{}/bpl/getMatchingPVs?{key}={}&limit={}",
411            self.base,
412            encode(&s.pattern),
413            s.limit
414        )
415    }
416
417    /// `bpl/getMetadata?pv=...`.
418    pub fn metadata_url(&self, pv: &str) -> String {
419        format!("{}/bpl/getMetadata?pv={}", self.base, encode(pv))
420    }
421}

423#[cfg(not(target_arch = "wasm32"))]
424impl ArchiverClient {
425    /// Fetch and decode one PV's data. `Ok(None)` when the archiver has no
426    /// data in the range (a `[ ]` response); `Err(Http{404})` when the PV is
427    /// not archived. Blocking: call from a worker thread.
428    pub fn get_data(&self, q: &Query) -> Result<Option<TimeSeries>> {
429        let body = self.get(&self.data_url(q))?;
430        let bins = matches!(q.op, Operator::Optimized(_));
431        Ok(decode::data(&body, bins)?.into_iter().next())
432    }
433
434    /// The archived PV names matching `s`, sorted. Blocking.
435    pub fn search(&self, s: &Search) -> Result<Vec<String>> {
436        let mut names = parse_names(&self.get(&self.search_url(s))?)?;
437        names.sort();
438        names.dedup();
439        Ok(names)
440    }
441
442    /// The PV's archive type info. Blocking.
443    pub fn metadata(&self, pv: &str) -> Result<PvInfo> {
444        parse_metadata(&self.get(&self.metadata_url(pv))?)
445    }
446}

452/// Decode a `getData.json` body: one [`TimeSeries`] per PV, empty for
453/// `[ ]`. Tolerates bare `NaN`/`Infinity`/`-Infinity`, string or numeric
454/// `PREC`/`LOPR`/`HOPR`, unknown keys, and `fields` objects. A severity-3
455/// sample whose `fields` hold `cnxlostepsecs` becomes a [`Sample::gap`].
456/// A 5-element array `val` on a scalar PV is an `optimized` bin
457/// (`binned = true`); other arrays are waveforms (element 0 plotted).
458///
459/// Without knowing the request, a series whose every array value has five
460/// elements with a whole, non-negative count is taken as bins;
461/// [`ArchiverClient::get_data`] only does that for `optimized` requests.
462pub fn parse_data(body: &str) -> Result<Vec<TimeSeries>> {
463    decode::data(body, true)
464}
465
466/// Decode a JSON array of PV names (`getMatchingPVs`, `getAllPVs`).
467pub fn parse_names(body: &str) -> Result<Vec<String>> {
468    decode::names(body)
469}
470
471/// Decode a `getMetadata` body.
472pub fn parse_metadata(body: &str) -> Result<PvInfo> {
473    decode::metadata(body)
474}

476/// Percent-encode a query parameter value (everything but RFC 3986
477/// unreserved characters).
478fn encode(s: &str) -> String {
479    let mut out = String::with_capacity(s.len() * 3 / 2);
480    for b in s.bytes() {
481        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
482            out.push(b as char);
483        } else {
484            out.push_str(&format!("%{b:02X}"));
485        }
486    }
487    out
488}
