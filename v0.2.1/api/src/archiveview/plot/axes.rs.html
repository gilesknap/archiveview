1//! Value axes: which axis a new trace joins, and what each axis is called.
2//!
3//! A trace settles on an axis once, when its first data arrives (see
4//! DESIGN.md "Value axes"): it joins the existing axis whose units and
5//! values suit it best, or gets a new one. Nothing here moves a trace
6//! again; only the user does, from its chip.
7//!
8//! Pure functions over [`Stats`] and names, so the rules are table-tested.
9
10/// Past this many axes of values (axes of states aside) a new trace joins
11/// the closest axis it may, rather than get an axis of its own. Soft: with
12/// none it may join, it still gets one.
13pub const SOFT_LIMIT: usize = 4;
14
15/// What a trace's (or an axis's) values look like, for choosing an axis.
16#[derive(Clone, Debug, PartialEq)]
17pub struct Stats {
18    /// Units (`EGU`), empty when it has none.
19    pub units: String,
20    /// The bulk of the values: the 5th and 95th percentiles, so a spike
21    /// or a glitch to zero does not decide the scale.
22    pub lo: f64,
23    pub hi: f64,
24    /// The smallest and largest values.
25    pub min: f64,
26    pub max: f64,
27    /// Every value is above zero (a log scale can show them all).
28    pub positive: bool,
29    /// Whole numbers: an enum or binary PV, drawn as states.
30    pub whole: bool,
31    /// An enum's state names, when known (from the live PV).
32    pub states: Vec<String>,
33}
34
35impl Stats {
36    /// The stats of `values` (finite ones; `None` with none).
37    pub fn of(values: impl Iterator<Item = f64>, units: &str, whole: bool) -> Option<Self> {
38        let mut v: Vec<f64> = values.filter(|v| v.is_finite()).collect();
39        if v.is_empty() {
40            return None;
41        }
42        v.sort_by(f64::total_cmp);
43        let at = |q: f64| v[((v.len() - 1) as f64 * q).round() as usize];
44        Some(Self {
45            units: units.to_string(),
46            lo: at(0.05),
47            hi: at(0.95),
48            min: v[0],
49            max: v[v.len() - 1],
50            positive: v[0] > 0.0,
51            whole,
52            states: Vec::new(),
53        })
54    }
55
56    /// The stats of an axis holding traces with stats `members`.
57    pub fn merge<'a>(members: impl IntoIterator<Item = &'a Stats>) -> Option<Self> {
58        members.into_iter().cloned().reduce(|a, b| Self {
59            // An axis keeps its first units; mixed only by hand.
60            units: if a.units.is_empty() { b.units } else { a.units },
61            lo: a.lo.min(b.lo),
62            hi: a.hi.max(b.hi),
63            min: a.min.min(b.min),
64            max: a.max.max(b.max),
65            positive: a.positive && b.positive,
66            whole: a.whole && b.whole,
67            states: a.states,
68        })
69    }
70
71    /// The typical size of the values: how far from zero the scale must
72    /// reach. Zero for values that are all zero.
73    pub fn magnitude(&self) -> f64 {
74        self.lo.abs().max(self.hi.abs()).max(self.hi - self.lo)
75    }
76
77    /// Best on a log scale: positive values in units of pressure (vacuum
78    /// spans decades, if not always in one window), or spanning several
79    /// decades: two in the bulk of the values, or three in all.
80    pub fn wants_log(&self) -> bool {
81        self.positive
82            && !self.whole
83            && self.min > 0.0
84            && (quantity(&self.units) == Some("Pressure")
85                || self.hi / self.lo >= 100.0
86                || self.max / self.min >= 1000.0)
87    }
88}

90/// How well a trace with stats `t` fits an axis with stats `a`: `None`
91/// when it never may (different units, states among values, states of
92/// another enum), else a score where up to 1 is a match and lower is
93/// better.
94///
95/// The score is how many decades apart the two typical sizes are, over
96/// the decades allowed: one, and a little more for matching units and for
97/// ranges that overlap. Enums share only with enums of the same states.
98pub fn score(t: &Stats, a: &Stats) -> Option<f64> {
99    if t.whole || a.whole {
100        let same = t.whole && a.whole && !t.states.is_empty() && t.states == a.states;
101        return same.then_some(0.0);
102    }
103    let units = !t.units.is_empty() && !a.units.is_empty();
104    if units && t.units != a.units {
105        return None;
106    }
107    let (mt, ma) = (t.magnitude(), a.magnitude());
108    let decades = match (mt > 0.0, ma > 0.0) {
109        (true, true) => (mt / ma).log10().abs(),
110        (false, false) => 0.0,
111        // All zero (0 % in a shutdown) says nothing of the scale.
112        _ => f64::INFINITY,
113    };
114    let overlap = t.lo <= a.hi && a.lo <= t.hi;
115    let allowed = 1.0 + if units { 0.3 } else { 0.0 } + if overlap { 0.3 } else { 0.0 };
116    Some(decades / allowed)
117}

119/// Where a trace settles.
120#[derive(Clone, Copy, Debug, PartialEq)]
121pub enum Choice {
122    /// The existing axis with this id.
123    Join(u32),
124    /// An axis of its own.
125    New,
126}
127
128/// The axis for a trace with stats `t`, among `axes` (id and stats of
129/// each axis with traces on it); `unknown` more axes have traces but no
130/// stats yet (never shown), and count toward the limit.
131///
132/// The best match within its score, else a new axis. At the
133/// [`SOFT_LIMIT`] of axes of values, the closest axis it may join even
134/// when not a match; never one it may not (other units, states): then a
135/// new axis after all.
136pub fn choose(t: &Stats, axes: &[(u32, Stats)], unknown: usize) -> Choice {
137    let values = axes.iter().filter(|(_, a)| !a.whole).count() + unknown;
138    let best = axes
139        .iter()
140        .filter_map(|(id, a)| score(t, a).map(|s| (*id, s)))
141        .min_by(|a, b| a.1.total_cmp(&b.1));
142    match best {
143        Some((id, s)) if s <= 1.0 => Choice::Join(id),
144        Some((id, _)) if !t.whole && values >= SOFT_LIMIT => Choice::Join(id),
145        _ => Choice::New,
146    }
147}

149/// What an axis is named from: its traces' units, names and descriptions.
150#[derive(Clone, Debug, Default)]
151pub struct Members<'a> {
152    pub units: Vec<&'a str>,
153    pub names: Vec<&'a str>,
154    pub descs: Vec<&'a str>,
155    /// An axis of states (enums).
156    pub states: bool,
157}

159/// The quantity units measure, from a small table ("mA" → "Current").
160pub fn quantity(units: &str) -> Option<&'static str> {
161    Some(match units.trim() {
162        "A" | "mA" | "uA" | "µA" | "nA" | "pA" | "kA" => "Current",
163        // "C" is degrees at facilities far more often than coulombs.
164        "°C" | "degC" | "C" | "C°" | "K" | "mK" | "°F" | "degF" => "Temperature",
165        "mbar" | "mBar" | "bar" | "Torr" | "torr" | "mTorr" | "Pa" | "kPa" | "hPa" | "MPa" => {
166            "Pressure"
167        }
168        "m" | "mm" | "um" | "µm" | "nm" | "cm" => "Position",
169        "V" | "mV" | "kV" | "uV" | "µV" | "MV" => "Voltage",
170        "Hz" | "kHz" | "MHz" | "GHz" => "Frequency",
171        "mC" | "uC" | "µC" | "nC" | "pC" | "fC" => "Charge",
172        "s" | "ms" | "us" | "µs" | "ns" | "min" | "h" | "hr" | "hours" | "d" => "Time",
173        "W" | "mW" | "kW" | "MW" => "Power",
174        "deg" | "°" | "rad" | "mrad" | "urad" | "µrad" => "Angle",
175        "T" | "mT" | "G" | "gauss" => "Field",
176        "eV" | "keV" | "MeV" | "GeV" | "J" | "mJ" | "uJ" | "µJ" => "Energy",
177        _ => return None,
178    })
179}

181/// Names for `axes`, built in layers, each used only when it says more:
182///
183/// 1. the quantity and units: "Current (mA)"; unknown units as they are,
184///    "(counts)";
185/// 2. for axes that share units, the part of their PV names that tells
186///    them apart: "Position X (mm)", "Position Y (mm)";
187/// 3. without units, the PVs' description, when known;
188/// 4. else "Axis N".
189///
190/// An axis of states is named after its PV, or "State" with several.
191pub fn names(axes: &[Members]) -> Vec<String> {
192    let units: Vec<String> = axes.iter().map(axis_units).collect();
193    axes.iter()
194        .enumerate()
195        .map(|(i, m)| {
196            if m.states {
197                return match m.names.as_slice() {
198                    [one] => one.to_string(),
199                    _ => "State".to_string(),
200                };
201            }
202            let u = &units[i];
203            // Layer 2: the others with these units, this one among them.
204            let group: Vec<usize> = (0..axes.len()).filter(|&j| units[j] == *u).collect();
205            let apart = if group.len() > 1 && !u.is_empty() {
206                let others: Vec<&str> = group
207                    .iter()
208                    .filter(|&&j| j != i)
209                    .flat_map(|&j| axes[j].names.iter().copied())
210                    .collect();
211                distinguishing(&m.names, &others)
212            } else {
213                None
214            };
215            let quantity = quantity(u).map(str::to_string);
216            let base = match (quantity, apart) {
217                (Some(q), Some(a)) => Some(format!("{q} {a}")),
218                (q, a) => q.or(a),
219            };
220            if !u.is_empty() {
221                return match base {
222                    Some(b) => format!("{b} ({u})"),
223                    None => format!("({u})"),
224                };
225            }
226            if let Some(b) = base {
227                return b;
228            }
229            // Layer 3: one description shared by its traces.
230            match m.descs.first() {
231                Some(d) if !d.is_empty() && m.descs.iter().all(|x| x == d) => d.to_string(),
232                _ => format!("Axis {}", i + 1),
233            }
234        })
235        .collect()
236}

238/// An axis's units: its traces', distinct and in order ("degC · V" for a
239/// mix the user made); empty with none.
240fn axis_units(m: &Members) -> String {
241    let mut seen: Vec<&str> = Vec::new();
242    for u in &m.units {
243        let u = u.trim();
244        if !u.is_empty() && !seen.contains(&u) {
245            seen.push(u);
246        }
247    }
248    seen.join(" · ")
249}

251/// The words that set `mine` (an axis's PV names) apart from `others`
252/// (the names on the axes it is told from): those every one of `mine`
253/// has and none of `others` has, in the first name's order; at most
254/// three, or it is no name. `None` when no word is left.
255///
256/// `SR01C-DI-EBPM-01:SA:X` against `SR01C-DI-EBPM-01:SA:Y` is "X". Plain
257/// string logic: it suits `DOMAIN-TECH-COMP-NN:SIGNAL` names without
258/// depending on them.
259fn distinguishing(mine: &[&str], others: &[&str]) -> Option<String> {
260    let (first, rest) = mine.split_first()?;
261    let rest: Vec<Vec<&str>> = rest.iter().map(|n| words(n)).collect();
262    let other: Vec<&str> = others.iter().flat_map(|n| words(n)).collect();
263    let mut kept: Vec<&str> = Vec::new();
264    for w in words(first) {
265        if rest.iter().all(|r| r.contains(&w)) && !other.contains(&w) && !kept.contains(&w) {
266            kept.push(w);
267        }
268    }
269    (!kept.is_empty() && kept.len() <= 3).then(|| kept.join(" "))
270}
271
272/// The words of a PV name, split at `-`, `:`, `.`, `_`, `/` and spaces.
273fn words(name: &str) -> Vec<&str> {
274    name.split(['-', ':', '.', '_', '/', ' '])
275        .filter(|w| !w.is_empty())
276        .collect()
277}

407    #[test]
408    fn traces_choose_an_axis_and_respect_the_soft_limit() {
409        let current = (0, stats("mA", 0.0, 300.0));
410        let x = stats("mm", 0.0, 0.04);
411        assert_eq!(choose(&x, &[], 0), Choice::New);
412        assert_eq!(choose(&x, std::slice::from_ref(&current), 0), Choice::New);
413        let axes = [current.clone(), (1, x.clone())];
414        assert_eq!(choose(&stats("mm", 0.0, 0.6), &axes, 0), Choice::Join(1));
415        // The better of two matches.
416        let axes = [
417            (0, stats("", 0.0, 1.0)),
418            (1, stats("", 0.0, 5.0)),
419            (2, stats("", 0.0, 50.0)),
420        ];
421        assert_eq!(choose(&stats("", 0.0, 8.0), &axes, 0), Choice::Join(1));
422        // Four axes: same units join the closest even when far apart.
423        let full = [
424            current.clone(),
425            (1, stats("h", 5.0, 13.0)),
426            (2, stats("nC", 0.0, 0.01)),
427            (3, stats("mm", 0.0, 1.0)),
428        ];
429        assert_eq!(choose(&stats("mm", 50.0, 90.0), &full, 0), Choice::Join(3));
430        // Other units never do: the limit is soft, a fifth axis.
431        assert_eq!(choose(&stats("V", 9.0, 11.0), &full, 0), Choice::New);
432        assert_eq!(choose(&stats("%", 0.0, 0.0), &full, 0), Choice::New);
433        // Axes not yet shown count; axes of states do not.
434        let three = &full[..3];
435        assert_eq!(choose(&stats("mm", 50.0, 90.0), three, 0), Choice::New);
436        let with_state = [
437            three,
438            &[(4, states(&["Off", "On"])), (5, stats("x", 50.0, 90.0))],
439        ]
440        .concat();
441        assert_eq!(
442            choose(&stats("x", 1000.0, 2000.0), &with_state, 0),
443            Choice::Join(5),
444            "four axes of values"
445        );
446        assert_eq!(
447            choose(&stats("x", 1000.0, 2000.0), &with_state[..4], 1),
448            Choice::New,
449            "three of values and one of states, plus one unknown, is four"
450        );
451        // States never join values, or other states: always their own.
452        assert_eq!(
453            choose(&states(&["Shutdown", "User"]), &with_state, 0),
454            Choice::New
455        );
456        assert_eq!(
457            choose(&states(&["Off", "On"]), &with_state, 0),
458            Choice::Join(4)
459        );
460    }

462    #[test]
463    fn stats_use_the_bulk_of_the_values() {
464        // A glitch to zero and a spike do not set the scale.
465        let mut v: Vec<f64> = vec![300.0; 98];
466        v.extend([0.0, 5000.0]);
467        let s = Stats::of(v.into_iter(), "mA", false).unwrap();
468        assert_eq!((s.lo, s.hi), (300.0, 300.0));
469        assert!(!s.positive);
470        assert!(Stats::of([f64::NAN].into_iter(), "", false).is_none());
471        // Vacuum over decades wants a log scale; a lifetime does not.
472        let p = Stats::of(
473            (0..100).map(|i| 1e-10 * 10f64.powf(i as f64 / 25.0)),
474            "mbar",
475            false,
476        )
477        .unwrap();
478        assert!(p.wants_log());
479        assert!(!stats("h", 5.2, 12.7).wants_log());
480        assert!(!stats("mA", -0.1, 300.0).wants_log());
481        // A gauge in mbar is on a log scale even in a quiet week; a vent
482        // spike in the full range counts too.
483        assert!(stats("mbar", 2e-10, 3e-10).wants_log());
484        assert!(!stats("mbar", 0.0, 3e-10).wants_log(), "not with a zero");
485        let spike = Stats {
486            max: 3e-7,
487            ..stats("", 2e-10, 3e-10)
488        };
489        assert!(spike.wants_log());
490        let both = Stats::merge([&stats("mm", 0.0, 0.04), &stats("mm", -0.1, 0.6)]).unwrap();
491        assert_eq!((both.lo, both.hi, both.units.as_str()), (-0.1, 0.6, "mm"));
492    }

529                members(&["mA"], &["BR01C-DI-DCCT-01:SIGNAL"]),
530            ]),
531            ["Current SR (mA)", "Current BR01C (mA)"]
532        );

534        assert_eq!(
535            names(&[
536                members(

538                    &["SR01C-DI-EBPM-01:SA:X", "SR02C-DI-EBPM-01:SA:X"]

540                members(&["mm"], &["SR01C-DI-EBPM-01:SA:Y"]),
541            ]),

545        let desc = Members {
