1//! Whether the archiver can be reached, for the archiver pill.
2//!
3//! A `bpl/getVersion` probe runs when an archiver is set (at start-up, or
4//! after a change). After that the state is passive: every data fetch and
5//! search says how the archiver answered. A success turns it reachable. A
6//! failure that looks like the archiver's (no connection, a gateway error,
7//! a wrong URL) is checked with a probe before the state turns
8//! unreachable, as one slow query timing out is not the archiver down.
9//! While unreachable it is probed again every [`REPROBE`] seconds, so it
10//! turns reachable again when the archiver comes back. A 404 for one PV
11//! (not archived) says nothing about the archiver.
12
13use archiver::Error;
14
15/// While the archiver is unreachable, probe it this often (seconds).
16pub const REPROBE: f64 = 60.0;
17
18/// A probe gives up after this long (seconds): a host that drops packets
19/// should not leave the pill checking for the fetch timeout.
20pub const PROBE_TIMEOUT: u64 = 10;
21
22/// A probe not answered this long after it was sent (seconds) is given up
23/// on, and another may be sent: its answer was lost.
24const PROBE_LOST: f64 = PROBE_TIMEOUT as f64 + 5.0;
25
26/// What is known of the archiver.
27#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
28pub enum ArchiverState {
29    /// None is set: the plot is live only.
30    #[default]
31    None,
32    /// Set, and not yet heard from.
33    Checking,
34    /// It answered last time.
35    Reachable,
36    /// It failed last time.
37    Unreachable,
38}
39
40/// The archiver's state as the pill shows it ([`crate::Plot::archiver_status`]).
41#[derive(Clone, Debug, Default, PartialEq)]
42pub struct ArchiverStatus {
43    pub state: ArchiverState,
44    /// The appliance's version, from `bpl/getVersion`.
45    pub version: Option<String>,
46    /// The last archiver error and when it happened (seconds since 1970).
47    pub last_error: Option<(String, f64)>,
48    /// When the archiver last answered (seconds since 1970).
49    pub last_ok: Option<f64>,
50}
51
52/// The status, and when to probe.
53#[derive(Debug)]
54pub struct Monitor {
55    pub status: ArchiverStatus,
56    /// Counts archiver changes: a probe's answer about an earlier archiver
57    /// is dropped.
58    pub epoch: u64,
59    /// A probe should be sent (an archiver was just set).
60    pub due: bool,
61    /// A probe is in flight.
62    pub probing: bool,
63    /// When the last probe was sent (seconds since 1970).
64    pub probed_at: f64,
65    /// The re-probe interval while unreachable ([`REPROBE`]; shorter in
66    /// tests).
67    pub reprobe: f64,
68}
69
70impl Default for Monitor {
71    fn default() -> Self {
72        Self {
73            status: ArchiverStatus::default(),
74            epoch: 0,
75            due: false,
76            probing: false,
77            probed_at: f64::NEG_INFINITY,
78            reprobe: REPROBE,
79        }
80    }
81}
82
83impl Monitor {
84    /// A new archiver (or none): forget what was known, and probe it.
85    pub fn reset(&mut self, archiver: bool) {
86        self.epoch += 1;
87        self.status = ArchiverStatus {
88            state: if archiver {
89                ArchiverState::Checking
90            } else {
91                ArchiverState::None
92            },
93            ..Default::default()
94        };
95        self.due = archiver;
96        self.probing = false;
97    }
98
99    /// Whether to send a probe now: one is due, or the archiver has been
100    /// unreachable since the last one for [`Monitor::reprobe`] seconds.
101    /// A probe in flight for longer than it can take has been lost (its
102    /// answer overwritten, say), and no longer holds the next one back.
103    pub fn should_probe(&self, now: f64) -> bool {
104        if self.status.state == ArchiverState::None {
105            return false;
106        }
107        let lost = self.probing && now - self.probed_at > PROBE_LOST;
108        if lost {
109            return true;
110        }
111        !self.probing
112            && (self.due
113                || (self.status.state == ArchiverState::Unreachable
114                    && now - self.probed_at >= self.reprobe))
115    }
116
117    /// A probe was sent.
118    pub fn sent(&mut self, now: f64) {
119        self.due = false;
120        self.probing = true;
121        self.probed_at = now;
122    }
123
124    /// The archiver answered. True if it was unreachable until now, so
125    /// that what failed can be asked for again.
126    pub fn ok(&mut self, now: f64) -> bool {
127        if self.status.state == ArchiverState::None {
128            return false;
129        }
130        let back = self.status.state == ArchiverState::Unreachable;
131        self.status.state = ArchiverState::Reachable;
132        self.status.last_ok = Some(now);
133        back
134    }
135
136    /// A fetch or search failed with `e`, which looks like the
137    /// archiver's fault: check with a probe before saying so (unless it is
138    /// known to be unreachable already).
139    pub fn suspect(&mut self, e: &Error, now: f64) {
140        if self.status.state == ArchiverState::None {
141            return;
142        }
143        self.status.last_error = Some((e.to_string(), now));
144        if self.status.state != ArchiverState::Unreachable {
145            self.due = true;
146        }
147    }
148
149    /// The archiver failed with `e`.
150    fn failed(&mut self, e: &Error, now: f64) {
151        if self.status.state == ArchiverState::None {
152            return;
153        }
154        self.status.state = ArchiverState::Unreachable;
155        self.status.last_error = Some((e.to_string(), now));
156    }
157
158    /// A probe's answer: the version (`None` when the archiver answers but
159    /// not to `getVersion`), or why it failed. Answers about an earlier
160    /// archiver are dropped; true when the archiver is back (as
161    /// [`Monitor::ok`]).
162    pub fn probed(&mut self, epoch: u64, result: Result<Option<String>, Error>, now: f64) -> bool {
163        if epoch != self.epoch {
164            return false;
165        }
166        self.probing = false;
167        match result {
168            Ok(version) => {
169                if version.is_some() {
170                    self.status.version = version;
171                }
172                self.ok(now)
173            }
174            Err(e) => {
175                self.failed(&e, now);
176                false
177            }
178        }
179    }
180}
181
182/// Whether `e`, from a data request or a search, is the archiver's fault
183/// rather than the PV's (or the search pattern's): no answer, a wrong URL,
184/// or a gateway or availability error. A 404 is the archiver's fault only
185/// for a search, which any archiver answers; for data it means "not
186/// archived" (a wrong URL is told apart before it gets here, as
187/// [`Error::BadUrl`]). Other HTTP errors are about the request (a 400 for
188/// an odd name, a 500 for a bad regex).
189pub fn archiver_fault(e: &Error, search: bool) -> bool {
190    match e {
191        Error::Transport(_) | Error::BadUrl(_) => true,
192        Error::Http { status, .. } => matches!(status, 502..=504) || (search && *status == 404),
193        Error::Parse(_) => false,
194    }
195}

197#[cfg(test)]
198mod tests {
199    use super::*;
200
201    fn down() -> Error {
202        Error::Transport("connection refused".into())
203    }
204
205    #[test]
206    fn a_new_archiver_is_checked_then_answers_decide() {
207        let mut m = Monitor::default();
208        assert_eq!(m.status.state, ArchiverState::None);
209        assert!(!m.should_probe(0.0));
210        // Without an archiver nothing changes it.
211        m.ok(1.0);
212        m.suspect(&down(), 1.0);
213        assert_eq!(m.status.state, ArchiverState::None);
214        assert!(!m.should_probe(100.0));
215
216        m.reset(true);
217        assert_eq!(m.status.state, ArchiverState::Checking);
218        assert!(m.should_probe(0.0));
219        m.sent(0.0);
220        assert!(!m.should_probe(0.0), "one probe at a time");
221        assert!(!m.probed(m.epoch, Ok(Some("v1".into())), 1.0));
222        assert_eq!(m.status.state, ArchiverState::Reachable);
223        assert_eq!(m.status.version.as_deref(), Some("v1"));
224        assert!(!m.should_probe(1000.0), "no polling while reachable");
225
226        // Passive: a failure is checked with a probe, which turns it red
227        // if the archiver is down; then it is probed again slowly.
228        m.suspect(&down(), 9.0);
229        assert_eq!(m.status.state, ArchiverState::Reachable, "not yet");
230        assert!(m.should_probe(9.0));
231        m.sent(9.0);
232        assert!(!m.probed(m.epoch, Ok(None), 9.5), "fine after all");
233        assert_eq!(m.status.state, ArchiverState::Reachable);
234        m.suspect(&down(), 10.0);
235        m.sent(0.0);
236        m.probed(m.epoch, Err(down()), 10.0);
237        assert_eq!(m.status.state, ArchiverState::Unreachable);
238        assert_eq!(
239            m.status.last_error,
240            Some(("connection refused".to_string(), 10.0))
241        );
242        assert!(!m.should_probe(30.0), "probed less than a minute ago");
243        assert!(m.should_probe(REPROBE), "a minute since the last probe");
244        // Failures while unreachable add no probes.
245        m.suspect(&down(), 30.0);
246        assert!(!m.should_probe(30.0));
247        m.sent(10.0);
248        assert!(!m.probed(m.epoch, Err(down()), 11.0));
249        assert!(!m.should_probe(30.0));
250        assert!(m.should_probe(10.0 + REPROBE));
251        // Any success brings it back.
252        assert!(m.ok(40.0));
253        assert_eq!(m.status.state, ArchiverState::Reachable);
254        assert_eq!(m.status.version.as_deref(), Some("v1"), "kept");
255    }
256
257    #[test]
258    fn a_lost_probe_answer_does_not_stop_probing() {
259        let mut m = Monitor::default();
260        m.reset(true);
261        m.sent(0.0);
262        // Its answer never comes (another's overwrote it).
263        assert!(!m.should_probe(PROBE_LOST - 1.0));
264        assert!(m.should_probe(PROBE_LOST + 1.0), "given up on");
265        m.sent(PROBE_LOST + 1.0);
266        assert!(!m.probed(m.epoch, Ok(Some("v".into())), PROBE_LOST + 2.0));
267        assert_eq!(m.status.state, ArchiverState::Reachable);
268    }
269
270    #[test]
271    fn answers_about_an_earlier_archiver_are_dropped() {
272        let mut m = Monitor::default();
273        m.reset(true);
274        m.sent(0.0);
275        let old = m.epoch;
276        m.reset(true);
277        assert!(!m.probed(old, Err(down()), 1.0));
278        assert_eq!(m.status.state, ArchiverState::Checking);
279        m.reset(false);
280        assert_eq!(m.status.state, ArchiverState::None);
281        assert!(!m.should_probe(0.0));
282    }

284    #[test]
285    fn only_the_archivers_faults_count() {
286        let http = |status| Error::Http {
287            status,
288            message: String::new(),
289        };
290        assert!(archiver_fault(&down(), false));
291        assert!(archiver_fault(&Error::BadUrl("x".into()), false));
292        assert!(archiver_fault(&http(503), false));
293        assert!(!archiver_fault(&http(404), false), "not archived");
294        assert!(archiver_fault(&http(404), true), "no search: a wrong URL");
295        assert!(!archiver_fault(&http(400), false));
296        assert!(!archiver_fault(&http(500), true), "a bad regex");
297        assert!(!archiver_fault(&Error::Parse("x".into()), false));
298    }
