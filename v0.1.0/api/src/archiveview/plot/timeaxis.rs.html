1//! The time axis: ticks on whole steps of local time, adaptive labels, and
2//! the span text the range controls read and show. `time_ticks` (made
3//! daylight-saving aware) and `parse_span` come from bobview's `trend.rs`.
4
5use chrono::{DateTime, Datelike, Local, NaiveDate, NaiveDateTime, NaiveTime, TimeZone, Timelike};
6
7const DAY: f64 = 86_400.0;
8const WEEK: f64 = 7.0 * DAY;
9/// A month as a step: nominal, for choosing it; its ticks are on the 1st.
10const MONTH: f64 = 30.0 * DAY;
11/// A year as a step: its ticks are on 1 January.
12const YEAR: f64 = 365.0 * DAY;
13
14/// Steps for the time axis, in seconds: whole seconds, minutes, hours,
15/// days, weeks (from Mondays), calendar months and years.
16const TIME_STEPS: &[f64] = &[
17    1.0,
18    2.0,
19    5.0,
20    10.0,
21    15.0,
22    30.0,
23    60.0,
24    120.0,
25    300.0,
26    600.0,
27    900.0,
28    1800.0,
29    3600.0,
30    7200.0,
31    10_800.0,
32    21_600.0,
33    43_200.0,
34    DAY,
35    2.0 * DAY,
36    WEEK,
37    2.0 * WEEK,
38    MONTH,
39    2.0 * MONTH,
40    3.0 * MONTH,
41    6.0 * MONTH,
42    YEAR,
43];
44
45/// Ticks from `from` to `to` (seconds since 1970) at the smallest step
46/// giving at most `max_ticks`, on whole steps of wall-clock time in `tz`
47/// (each tick's own offset, so day lines stay at midnight across a
48/// daylight-saving change). Week ticks are on Mondays, month ticks on the
49/// 1st and year ticks on 1 January. Returns the ticks and the step.
50pub fn time_ticks<Tz: TimeZone>(from: f64, to: f64, max_ticks: usize, tz: &Tz) -> (Vec<f64>, f64) {
51    let span = to - from;
52    if span <= 0.0 || !span.is_finite() {
53        return (Vec::new(), 0.0);
54    }
55    let max_ticks = max_ticks.max(1) as f64;
56    let step = TIME_STEPS
57        .iter()
58        .copied()
59        .find(|s| span / s <= max_ticks)
60        .unwrap_or_else(|| YEAR * (span / YEAR / max_ticks).ceil());
61    let Some(start) = DateTime::from_timestamp(from.floor() as i64, 0) else {
62        return (Vec::new(), step);
63    };
64    let local = start.with_timezone(tz).naive_local();
65    // The wall-clock time `n` and its instant (the earlier of a repeated
66    // hour; none in the hour skipped in spring).
67    let at = |n: NaiveDateTime| {
68        tz.from_local_datetime(&n)
69            .earliest()
70            .map(|d| d.timestamp() as f64)
71    };
72    let mut ticks: Vec<f64> = Vec::new();
73    let push = |ticks: &mut Vec<f64>, t: f64| {
74        if t >= from && t <= to && ticks.last().is_none_or(|l| t > *l) {
75            ticks.push(t);
76        }
77    };
78    if step >= MONTH {
79        // The 1st of every `months`th month (counted from January 1970),
80        // or 1 January of every `years`th year.
81        let months = if step >= YEAR {
82            12 * (step / YEAR).round() as i32
83        } else {
84            (step / MONTH).round() as i32
85        };
86        let (mut y, mut m) = (local.year(), local.month0() as i32);
87        for _ in 0..4000 {
88            if ticks.len() >= 1000 {
89                break;
90            }
91            let index = (y - 1970) * 12 + m;
92            if index.rem_euclid(months) == 0
93                && let Some(d) = NaiveDate::from_ymd_opt(y, m as u32 + 1, 1)
94                && let Some(t) = at(d.and_time(NaiveTime::MIN))
95            {
96                if t > to {
97                    break;
98                }
99                push(&mut ticks, t);
100            }
101            m += 1;
102            if m == 12 {
103                (y, m) = (y + 1, 0);
104            }
105        }
106    } else if step >= DAY {
107        // Local midnights: every `days`th day counted from 1970-01-01, or
108        // for weeks every `days`th day from Monday 1970-01-05.
109        let days = (step / DAY).round() as i64;
110        let epoch = if step >= WEEK {
111            NaiveDate::from_ymd_opt(1970, 1, 5)
112        } else {
113            NaiveDate::from_ymd_opt(1970, 1, 1)
114        }
115        .unwrap_or_default();
116        let mut date = local.date();
117        for _ in 0..4000 {
118            let Some(t) = at(date.and_time(NaiveTime::MIN)) else {
119                date = date.succ_opt().unwrap_or(date);
120                continue;
121            };
122            if t > to || ticks.len() >= 1000 {
123                break;
124            }
125            if (date - epoch).num_days().rem_euclid(days) == 0 {
126                push(&mut ticks, t);
127            }
128            match date.succ_opt() {
129                Some(d) => date = d,
130                None => break,
131            }
132        }
133    } else {
134        // Whole steps of the local day (every step divides a day).
135        let step_s = step as i64;
136        let into_day = local.time().num_seconds_from_midnight() as i64;
137        let mut n = local.date().and_time(NaiveTime::MIN)
138            + chrono::Duration::seconds(into_day / step_s * step_s);
139        for _ in 0..4000 {
140            if let Some(t) = at(n) {
141                if t > to || ticks.len() >= 1000 {
142                    break;
143                }
144                push(&mut ticks, t);
145            }
146            n += chrono::Duration::seconds(step_s);
147        }
148    }
149    (ticks, step)
150}

152/// A time axis label in local time, as precise as the step: "14:30:15",
153/// "14:30", "Tue 17" at midnight or for day steps, "Oct" at the start of a
154/// month. Week ticks read "13 Oct" (with the year on the first of a
155/// year), month ticks "Oct" ("Jan 2027" in January) and year ticks "2027".
156/// The `first` tick carries its context ("Tue 17 14:00", "Oct 2026").
157pub fn time_label(t: f64, step: f64, first: bool) -> String {
158    let Some(d) = Local.timestamp_opt(t.round() as i64, 0).single() else {
159        return String::new();
160    };
161    let midnight = d.hour() == 0 && d.minute() == 0 && d.second() == 0;
162    let format = if step < 60.0 {
163        if first { "%a %-d %H:%M:%S" } else { "%H:%M:%S" }
164    } else if step < DAY {
165        if first {
166            "%a %-d %H:%M"
167        } else if midnight {
168            "%a %-d"
169        } else {
170            "%H:%M"
171        }
172    } else if step < WEEK {
173        if first {
174            "%-d %b"
175        } else if d.day() == 1 {
176            "%b"
177        } else {
178            "%a %-d"
179        }
180    } else if step < MONTH {
181        // The first tick of a new year carries it.
182        let new_year = d.month() == 1 && f64::from(d.day() - 1) * DAY < step;
183        if new_year { "%-d %b %Y" } else { "%-d %b" }
184    } else if step < YEAR {
185        if first || d.month() == 1 {
186            "%b %Y"
187        } else {
188            "%b"
189        }
190    } else {
191        "%Y"
192    };
193    d.format(format).to_string()
194}

196/// A time in full for the crosshair readout: "Tue 17 Oct 14:32:05.250"
197/// (milliseconds only when the span shown is under a minute).
198pub fn readout_time(t: f64, span: f64) -> String {
199    let ms = (t * 1000.0).round() as i64;
200    let Some(d) = Local.timestamp_millis_opt(ms).single() else {
201        return String::new();
202    };
203    let format = if span < 60.0 {
204        "%a %-d %b %H:%M:%S%.3f"
205    } else {
206        "%a %-d %b %H:%M:%S"
207    };
208    d.format(format).to_string()
209}

211/// A time for the crosshair's tag on the time axis: "14:32:05", with the
212/// day ("Tue 17 14:32") over spans of a day or more, and milliseconds
213/// over spans under a minute.
214pub fn tag_time(t: f64, span: f64) -> String {
215    let ms = (t * 1000.0).round() as i64;
216    let Some(d) = Local.timestamp_millis_opt(ms).single() else {
217        return String::new();
218    };
219    let format = if span < 60.0 {
220        "%H:%M:%S%.3f"
221    } else if span < 86_400.0 {
222        "%H:%M:%S"
223    } else {
224        "%a %-d %H:%M"
225    };
226    d.format(format).to_string()
227}

229/// A span typed into the custom range field, such as `90m`, `1 hour 30 min`
230/// or `2 days`, in seconds (bobview's strip chart `parse_span`).
231pub fn parse_span(text: &str) -> Option<f64> {
232    let mut total = 0.0;
233    let mut words = text.split_whitespace().peekable();
234    let mut any = false;
235    while let Some(word) = words.next() {
236        // A number and its unit, apart (`10 min`) or together (`10min`).
237        let split = word
238            .find(|c: char| !(c.is_ascii_digit() || c == '.'))
239            .unwrap_or(word.len());
240        let (num, unit) = word.split_at(split);
241        let n: f64 = num.parse().ok()?;
242        let unit = if unit.is_empty() {
243            words.next().unwrap_or("s")
244        } else {
245            unit
246        };
247        let secs = match unit.to_ascii_lowercase().trim_end_matches('s') {
248            "" | "sec" | "second" => 1.0,
249            "m" | "min" | "minute" => 60.0,
250            "h" | "hr" | "hour" => 3600.0,
251            "d" | "day" => 86_400.0,
252            "w" | "wk" | "week" => 7.0 * 86_400.0,
253            _ => return None,
254        };
255        total += n * secs;
256        any = true;
257    }
258    (any && total > 0.0).then_some(total)
259}

261/// A span as short text: "45s", "3h 20m", "1d", "7d 12h".
262pub fn format_span(secs: f64) -> String {
263    let s = secs.round().max(1.0) as u64;
264    let (d, h, m, sec) = (s / 86_400, s % 86_400 / 3600, s % 3600 / 60, s % 60);
265    let parts: Vec<String> = [(d, "d"), (h, "h"), (m, "m"), (sec, "s")]
266        .iter()
267        .filter(|(n, _)| *n > 0)
268        .map(|(n, u)| format!("{n}{u}"))
269        .collect();
270    // The two largest units are plenty.
271    parts.into_iter().take(2).collect::<Vec<_>>().join(" ")
272}

274/// A local date and time as typed: `2026-10-17 14:30`, with optional
275/// seconds, or a `T` between date and time; or just a date (midnight).
276pub fn parse_local(text: &str) -> Option<f64> {
277    let text = text.trim().replace('T', " ");
278    let naive = ["%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M"]
279        .iter()
280        .find_map(|f| NaiveDateTime::parse_from_str(&text, f).ok())
281        .or_else(|| {
282            chrono::NaiveDate::parse_from_str(&text, "%Y-%m-%d")
283                .ok()
284                .and_then(|d| d.and_hms_opt(0, 0, 0))
285        })?;
286    let local = Local.from_local_datetime(&naive).earliest()?;
287    Some(local.timestamp() as f64)
288}

290/// `t` as local `YYYY-MM-DD HH:MM:SS`, as [`parse_local`] reads it.
291pub fn format_local(t: f64) -> String {
292    Local
293        .timestamp_opt(t.floor() as i64, 0)
294        .single()
295        .map(|d| d.format("%Y-%m-%d %H:%M:%S").to_string())
296        .unwrap_or_default()
297}

305    /// UTC, or an hour ahead of it, as `FixedOffset`s.
306    fn utc() -> FixedOffset {
307        FixedOffset::east_opt(0).unwrap()
308    }
309
310    /// A UK-like zone: GMT, then BST (an hour ahead) from `change`
311    /// (seconds since 1970) on: clocks go 01:00 -> 02:00.
312    #[derive(Clone, Copy)]
313    struct Spring {
314        change: i64,
315    }
316
317    impl TimeZone for Spring {
318        type Offset = FixedOffset;
319        fn from_offset(_: &FixedOffset) -> Self {
320            unreachable!()
321        }
322        fn offset_from_local_date(&self, _: &NaiveDate) -> LocalResult<FixedOffset> {
323            unreachable!()
324        }
325        fn offset_from_local_datetime(&self, l: &NaiveDateTime) -> LocalResult<FixedOffset> {
326            let as_utc = l.and_utc().timestamp();
327            let (gmt, bst) = (utc(), FixedOffset::east_opt(3600).unwrap());
328            if as_utc < self.change {
329                LocalResult::Single(gmt)
330            } else if as_utc < self.change + 3600 {
331                LocalResult::None
332            } else {
333                LocalResult::Single(bst)
334            }
335        }
336        fn offset_from_utc_date(&self, _: &NaiveDate) -> FixedOffset {
337            unreachable!()
338        }
339        fn offset_from_utc_datetime(&self, u: &NaiveDateTime) -> FixedOffset {
340            let off = if u.and_utc().timestamp() < self.change {
341                0
342            } else {
343                3600
344            };
345            FixedOffset::east_opt(off).unwrap().fix()
346        }
347    }

349    #[test]
350    fn time_ticks_fall_on_whole_local_steps() {
351        let (t, step) = time_ticks(1000.0, 1600.0, 6, &utc());
352        assert_eq!(step, 120.0);
353        assert_eq!(t, [1080.0, 1200.0, 1320.0, 1440.0, 1560.0]);
354        // Half an hour ahead of UTC, hours are whole in local time.
355        let half = FixedOffset::east_opt(1800).unwrap();
356        let (t, step) = time_ticks(0.0, 4.0 * 3600.0, 4, &half);
357        assert_eq!(step, 3600.0);
358        assert_eq!(t[0], 1800.0);
359        // A week over 4 ticks: 2-day steps; 60 days over 3: 30-day steps.
360        assert_eq!(time_ticks(0.0, 7.0 * 86_400.0, 4, &utc()).1, 2.0 * 86_400.0);
361        assert_eq!(
362            time_ticks(0.0, 60.0 * 86_400.0, 3, &utc()).1,
363            30.0 * 86_400.0
364        );
365        // Longer still: calendar months, then years.
366        let (_, step) = time_ticks(0.0, 400.0 * 86_400.0, 5, &utc());
367        assert_eq!(step, 3.0 * MONTH);
368        let (t, step) = time_ticks(0.0, 20.0 * YEAR, 5, &utc());
369        assert_eq!(step, 4.0 * YEAR);
370        let years: Vec<i32> = t
371            .iter()
372            .map(|t| DateTime::from_timestamp(*t as i64, 0).unwrap().year())
373            .collect();
374        assert_eq!(years, [1970, 1974, 1978, 1982, 1986]);
375        assert!(time_ticks(5.0, 5.0, 5, &utc()).0.is_empty());
376    }

399    fn week_month_and_year_ticks_are_on_calendar_boundaries() {
400        // 60 days from 20 Aug 2026 over 8 ticks: weekly, on Mondays.
401        let from = utc_at("2026-08-20 10:00");
402        let (t, step) = time_ticks(from, from + 60.0 * DAY, 9, &utc());

426    #[test]
427    fn day_ticks_stay_at_midnight_across_a_clock_change() {
428        // Clocks go forward at 01:00 GMT on day 10.
429        let change = 10 * 86_400 + 3600;
430        let tz = Spring { change };
431        let (t, step) = time_ticks(5.0 * 86_400.0, 15.0 * 86_400.0, 12, &tz);
432        assert_eq!(step, 86_400.0);
433        // Midnights before the change are at 00:00 UTC, after it at 23:00
434        // UTC the day before (00:00 BST): never an hour off in local time.
435        assert_eq!(t[0], 5.0 * 86_400.0);
436        assert!(t.contains(&(10.0 * 86_400.0)));
437        assert!(t.contains(&(11.0 * 86_400.0 - 3600.0)));
438        assert!(!t.contains(&(11.0 * 86_400.0)));
439        // Hour ticks skip the missing hour and stay whole.
440        let (t, step) = time_ticks(change as f64 - 7200.0, change as f64 + 7200.0, 6, &tz);
441        assert_eq!(step, 3600.0);
442        assert!(t.windows(2).all(|w| w[1] - w[0] == 3600.0), "{t:?}");
443    }
