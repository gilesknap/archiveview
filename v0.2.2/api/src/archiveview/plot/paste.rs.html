1//! Adding PVs from pasted text: the primary selection (middle-click, the
2//! EDM/MEDM way) or the clipboard (Ctrl+V over the plot).
3//!
4//! The widget never touches a clipboard itself, so it needs no clipboard
5//! dependency: it reports middle-clicks ([`Plot::take_middle_click`]) for
6//! the host to answer with the selection's text ([`Plot::paste_pvs`],
7//! [`Plot::paste_into_search`]), and reports the search box's selected
8//! text for the host to make the primary selection
9//! ([`Plot::take_search_selection`]). Ctrl+V needs no host: egui delivers
10//! the clipboard as an `Event::Paste`.
11//!
12//! [`Plot::take_middle_click`]: super::Plot::take_middle_click
13//! [`Plot::paste_pvs`]: super::Plot::paste_pvs
14//! [`Plot::paste_into_search`]: super::Plot::paste_into_search
15//! [`Plot::take_search_selection`]: super::Plot::take_search_selection

17/// Where a middle-click landed, so where the selection's text goes.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum MiddleClick {
20    /// The plot area or the legend: add the PVs named in it
21    /// ([`Plot::paste_pvs`](super::Plot::paste_pvs)).
22    Plot,
23    /// The search box: paste the text into it at the cursor
24    /// ([`Plot::paste_into_search`](super::Plot::paste_into_search)).
25    Search,
26}

28/// What pasting text onto the plot did.
29#[derive(Clone, Debug, Default, PartialEq, Eq)]
30pub struct Pasted {
31    /// The PVs added.
32    pub added: Vec<String>,
33    /// PVs named that were plotted already.
34    pub present: Vec<String>,
35    /// The text, when it named no PV.
36    pub text: String,
37    /// It named more than [`MAX_PASTE`] PVs, this many, so none were
38    /// added: most likely a screenful pasted by accident.
39    pub too_many: usize,
40}
41
42/// The most PVs one paste adds. A paste naming more adds none (most
43/// likely a screenful selected by accident), and says so.
44pub const MAX_PASTE: usize = 20;
45
46impl Pasted {
47    /// A short message saying so, for a toast.
48    pub fn message(&self) -> String {
49        if self.too_many > 0 {
50            return format!(
51                "The selection names {} PVs: select at most {MAX_PASTE}",
52                self.too_many
53            );
54        }
55        match (self.added.as_slice(), self.present.as_slice()) {
56            ([one], _) => format!("Added {one}"),
57            ([], []) => {
58                let text = self.text.split_whitespace().collect::<Vec<_>>().join(" ");
59                if text.is_empty() {
60                    "Nothing to add: the selection is empty".into()
61                } else {
62                    let shown: String = text.chars().take(40).collect();
63                    let more = if shown.len() < text.len() { "…" } else { "" };
64                    format!("No PV names in “{shown}{more}”")
65                }
66            }
67            ([], [one]) => format!("{one} is plotted already"),
68            ([], all) => format!("{} PVs are plotted already", all.len()),
69            (added, _) => format!("Added {} PVs", added.len()),
70        }
71    }
72}

93/// The PV names in `text`, in order and once each. It is split into words
94/// on whitespace, commas and semicolons, except inside the `(arguments)`
95/// of a `loc://` or `sim://` name closed on the same line
96/// (`sim://sine(0, 10, 0.1)` is one word), and quotes and brackets around
97/// a word are dropped, as is a lone bracket left by a list in brackets
98/// (`(A:B, C:D)`).
99///
100/// The schemes taken are those the host's [`pv::PvManager`] serves, as
101/// [`pv::PvName::parse`] reads them: `ca://` and `pva://` (in either case),
102/// and `loc://` and `sim://`, which no archiver has, so they are plotted
103/// live only. Formulas (`=expr`, `eq://expr`) are not: their text has
104/// spaces and operators, and cannot be told from prose. Any other scheme
105/// (`https://`, `sys://`) is not a PV.
106///
107/// After `ca://` or `pva://`, or with no scheme, a word can be a name if it
108/// has only characters EPICS record names use (letters, digits,
109/// `_ - + : [ ] < > { } $`, and `.` before a field), at least one letter,
110/// and does not start with `-` or `+` (a command-line flag) or start or
111/// end with `.` or `:`. After `loc://` such a name can be followed by a
112/// `<VType>` and then `(arguments)`, which end it (`loc://x<VLong>(7)`);
113/// after `sim://` it is one of bobview-pv's functions (`sine`,
114/// `sawtooth`, `ramp`, `noise`, `gaussianNoise`, `flipflop`, `const`), then
115/// optionally `(arguments)` (`sim://ramp`, `sim://sine(0, 10, 0.1)`). Unless it has a scheme, shapes that
116/// are something else are left out: times and dates (`12:30:00`,
117/// `2026-10-17T13:41:52Z`), `host:port` (two parts, a host name with a
118/// `.`, `localhost` or lower case, then digits: not `SR:BPM:01` or
119/// `DLS:01`), `a::b` paths, and in running text Python log lines
120/// (`INFO:root:Connected`; `WARN:LIMIT` is a name).
121///
122/// A selection of one word is a name if it can be one, so names with no
123/// structure at all (`TEMP1`, `BEAMCURRENT`) work. In several words only
124/// those with a scheme or some structure are kept (a `:`, `-`, `_`,
125/// braces, a digit or a `.FIELD`), so prose, shell commands and code
126/// selected with them add nothing.
127pub fn pv_names(text: &str) -> Vec<String> {
128    let words: Vec<&str> = words(text)
129        .into_iter()
130        .map(unwrap)
131        .filter(|w| !w.is_empty())
132        .collect();
133    let one = words.len() == 1;
134    let mut names: Vec<String> = Vec::new();
135    for word in words {
136        if looks_like_pv(word, one) && !names.iter().any(|n| n == word) {
137            names.push(word.to_string());
138        }
139    }
140    names
141}

143/// `text` split on whitespace, commas and semicolons, except inside the
144/// parentheses of a word that starts (after any quotes or brackets) with
145/// `loc://` or `sim://`: the arguments of `sim://sine(0, 10, 0.1)` or
146/// `loc://s<VString>("a b")`. A `(` not closed on its line within
147/// [`MAX_ARGS`] bytes is an ordinary character, so it cannot swallow the
148/// rest, and the work stays linear in a large selection.
149fn words(text: &str) -> Vec<&str> {
150    let sep = |c: char| c.is_whitespace() || c == ',' || c == ';';
151    let mut words = Vec::new();
152    // Where the word started, and whether its arguments are kept whole.
153    let mut start: Option<(usize, bool)> = None;
154    let mut i = 0;
155    while let Some(c) = text[i..].chars().next() {
156        if sep(c) {
157            if let Some((s, _)) = start.take() {
158                words.push(&text[s..i]);
159            }
160        } else {
161            let (_, glue) = start.get_or_insert_with(|| (i, has_arguments(&text[i..])));
162            if c == '(' && *glue {
163                if let Some(len) = closing(&text[i..]) {
164                    i += len;
165                    continue;
166                }
167                // Not a name's arguments, so neither is a later `(`.
168                *glue = false;
169            }
170        }
171        i += c.len_utf8();
172    }
173    if let Some((s, _)) = start {
174        words.push(&text[s..]);
175    }
176    words
177}

179/// Whether `text`, from a word's start, names a `loc://` or `sim://` PV,
180/// whose `(arguments)` may hold spaces and commas.
181fn has_arguments(text: &str) -> bool {
182    let quote_or_bracket = |c| {
183        matches!(
184            c,
185            '"' | '\'' | '`' | '“' | '”' | '‘' | '’' | '(' | '[' | '{' | '<'
186        )
187    };
188    let name = text.trim_start_matches(quote_or_bracket);
189    name.starts_with("loc://") || name.starts_with("sim://")
190}
191
192/// The longest `(arguments)` kept whole, in bytes: bobview's names are
193/// one short line.
194const MAX_ARGS: usize = 256;
195
196/// The length of `text`, which starts with `(`, up to and including the
197/// `)` that closes it, passing over parentheses in `"strings"`; `None`
198/// when it is not closed before a new line or within [`MAX_ARGS`] bytes.
199fn closing(text: &str) -> Option<usize> {
200    // Bytes: the delimiters are ASCII, which no other character's UTF-8
201    // contains.
202    let (mut depth, mut quoted, mut escaped) = (0usize, false, false);
203    for (i, &b) in text.as_bytes().iter().enumerate().take(MAX_ARGS) {
204        if b == b'\n' {
205            return None;
206        }
207        if quoted {
208            match b {
209                _ if escaped => escaped = false,
210                b'\\' => escaped = true,
211                b'"' => quoted = false,
212                _ => {}
213            }
214            continue;
215        }
216        match b {
217            b'"' => quoted = true,
218            b'(' => depth += 1,
219            b')' => {
220                depth = depth.checked_sub(1)?;
221                if depth == 0 {
222                    return Some(i + 1);
223                }
224            }
225            _ => {}
226        }
227    }
228    None
229}

231/// `word` without the quotes and brackets around it: `"A:B"`, `(A:B)`,
232/// `['A:B']`, and a leading `(`/`[` or trailing `)`/`]` the word has one
233/// too many of, as a list in brackets splits into (`(A:B,` `C:D)`). A bracket
234/// that belongs to the name (`A:B[1]`, `loc://x(1)`) stays.
235fn unwrap(mut word: &str) -> &str {
236    loop {
237        let quotes = |c| matches!(c, '"' | '\'' | '`' | '“' | '”' | '‘' | '’');
238        let trimmed = word.trim_matches(quotes);
239        let inner = [('(', ')'), ('[', ']'), ('{', '}'), ('<', '>')]
240            .iter()
241            .find_map(|&(open, close)| {
242                trimmed
243                    .strip_prefix(open)
244                    .and_then(|w| w.strip_suffix(close))
245            })
246            .unwrap_or(trimmed);
247        let inner = [('(', ')'), ('[', ']')]
248            .iter()
249            .fold(inner, |w, &(open, close)| {
250                let count = |c| w.chars().filter(|&x| x == c).count();
251                let (opens, closes) = (count(open), count(close));
252                match (w.strip_prefix(open), w.strip_suffix(close)) {
253                    (Some(rest), _) if opens > closes => rest,
254                    (_, Some(rest)) if closes > opens => rest,
255                    _ => w,
256                }
257            });
258        if inner == word {
259            return word;
260        }
261        word = inner;
262    }
263}

267    "TRACE", "DEBUG", "INFO", "WARN", "WARNING", "ERROR", "CRITICAL", "FATAL",

270/// Whether `word` can be a PV name; `alone` when it is the whole
271/// selection (see [`pv_names`]).
272fn looks_like_pv(word: &str, alone: bool) -> bool {
273    let Some((scheme, body)) = word.split_once("://") else {
274        return record_name(word) && !something_else(word, alone) && (alone || structured(word));
275    };
276    // A scheme says it is a PV, whatever it looks like, if the host's
277    // PvManager serves it.
278    match pv::PvName::parse(word).scheme {
279        pv::Scheme::Ca | pv::Scheme::Pva => record_name(body),
280        pv::Scheme::Loc => host_name(body, false),
281        pv::Scheme::Sim => host_name(body, true),
282        // Not told from prose (see `pv_names`).
283        pv::Scheme::Formula => false,
284        // `CA://`, `PVA://`, as some lists write them.
285        pv::Scheme::Unsupported => {
286            (scheme.eq_ignore_ascii_case("ca") || scheme.eq_ignore_ascii_case("pva"))
287                && record_name(body)
288        }
289    }
290}

292/// Only characters EPICS record names use, at least one letter, and no
293/// flag's `-`/`+` start or `.`/`:` at either end.
294fn record_name(body: &str) -> bool {
295    let chars_ok = body.chars().all(|c| {
296        c.is_ascii_alphanumeric()
297            || matches!(c, '_' | '-' | '+' | ':' | '[' | ']' | '<' | '>')
298            || matches!(c, '.' | '{' | '}' | '$')
299    });
300    let ends_ok = !body.starts_with(['.', ':', '-', '+']) && !body.ends_with(['.', ':']);
301    chars_ok && ends_ok && body.chars().any(|c| c.is_ascii_alphabetic())
302}

304/// The functions bobview-pv's `sim://` PVs know (its `sim::run`, which
305/// does not export them): any other is "unknown simulated PV".
306const SIM_FUNCTIONS: [&str; 7] = [
307    "sine",
308    "sawtooth",
309    "ramp",
310    "noise",
311    "gaussianNoise",
312    "flipflop",
313    "const",
314];
315
316/// A `loc://` name after its scheme: a record name, then optionally a
317/// `<VType>`, then optionally `(arguments)` that end it. `sim`: a
318/// `sim://` name, one of [`SIM_FUNCTIONS`] with no `<VType>`.
319fn host_name(body: &str, sim: bool) -> bool {
320    let (base, mut rest) = body.split_at(body.find(['<', '(']).unwrap_or(body.len()));
321    if sim && !SIM_FUNCTIONS.contains(&base) {
322        return false;
323    }
324    if let Some(after) = rest.strip_prefix('<')
325        && !sim
326    {
327        let Some((vtype, after)) = after.split_once('>') else {
328            return false;
329        };
330        if vtype.is_empty() || !vtype.chars().all(|c| c.is_ascii_alphanumeric()) {
331            return false;
332        }
333        rest = after;
334    }
335    let args_ok = rest.is_empty() || (rest.starts_with('(') && closing(rest) == Some(rest.len()));
336    record_name(base) && args_ok
337}

339/// A time, date, `host:port`, `a::b` path or log line, not a PV;
340/// `alone` when it is the whole selection.
341fn something_else(body: &str, alone: bool) -> bool {
342    let digits = |s: &str| !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit());
343    let parts: Vec<&str> = body.split(':').collect();
344    let b = body.as_bytes();
345    let date = b.len() >= 10
346        && digits(&body[..4])
347        && b[4] == b'-'
348        && digits(&body[5..7])
349        && b[7] == b'-'
350        && digits(&body[8..10]);
351    // 12:30, 13:41:52Z: digits before every colon.
352    let time = parts.len() > 1 && parts[..parts.len() - 1].iter().all(|p| digits(p));
353    // localhost:5064, ioc.example.org:5075, gateway:5064; not the
354    // numbered names everywhere (SR:BPM:01, DLS:01).
355    let port = parts.len() == 2 && digits(parts[1]) && {
356        let host = parts[0];
357        host.contains('.') || host == "localhost" || !host.chars().any(|c| c.is_ascii_uppercase())
358    };
359    // Python's LEVEL:logger:message, in running text only: WARN:LIMIT
360    // alone is a PV.
361    let log = !alone
362        && parts.len() >= 3
363        && LOG_LEVELS.contains(&parts[0])
364        && parts[1].chars().any(|c| c.is_ascii_lowercase());
365    date || time || port || log || body.contains("::")
366}

368/// Has some structure, as a PV name in running text would: a `:`, `-`,
369/// `_`, braces, a digit, or a `.FIELD` (upper case) at its end.
370fn structured(body: &str) -> bool {
371    let field = body.rsplit_once('.').is_some_and(|(_, f)| {
372        f.starts_with(|c: char| c.is_ascii_uppercase())
373            && f.chars()
374                .all(|c| c.is_ascii_uppercase() || c.is_ascii_digit())
375    });
376    field
377        || body
378            .chars()
379            .any(|c| matches!(c, ':' | '-' | '_' | '{' | '}') || c.is_ascii_digit())
380}

388        assert_eq!(pv_names("BL16I-EA-DET-01:COUNT"), ["BL16I-EA-DET-01:COUNT"]);

465            ("sim://ramp", &["sim://ramp"]),

479            ),

481            (

485        ] {

487        }
488        for text in [
489            "loc://",
490            "loc://(1)",
491            "sim://42",
492            "loc://x(1)junk",
493            "loc://x<>(1)",
494            "loc://x<V Long>(1)",
495            "loc://x<VLong",
496            // Never closed: an ordinary character, so not a name.
497            "sim://sine(0, 10",
498            // Formulas are not told from prose.
499            "eq://a+b",
500            "=a+b",
501            // Parentheses without a scheme still split and are still not names.
502            "A:B(1, 2)",
503        ] {
504            assert!(pv_names(text).is_empty(), "{text:?}: {:?}", pv_names(text));
505        }
506        // An unclosed bracket does not swallow what follows.
507        assert_eq!(pv_names("sim://sine(0, SR:X"), ["SR:X"]);
508        // Nor does one closed lines later, or far on.
509        assert_eq!(
510            pv_names("loc://x(1 or\n\nSR:A1 SR:B1\nlater)"),
511            ["SR:A1", "SR:B1"]
512        );
513        let far = format!("loc://x(1 {} SR:A1)", "word ".repeat(60));
514        assert_eq!(pv_names(&far), ["SR:A1"]);
515        // Only loc:// and sim:// arguments are kept whole: the URL's
516        // words split, and the name after its comma is found.
517        assert_eq!(pv_names("https://x/Foo_(SR:A1, SR:B1)"), ["SR:B1"]);
518        // sim:// names only bobview-pv's functions, with no type.
519        for text in [
520            "sim://foo",
521            "sim://Sine",
522            "sim://a:b<VLong>",
523            "sim://ramp<VLong>(1)",
524        ] {
525            assert!(pv_names(text).is_empty(), "{text:?}");
526        }
527        for f in SIM_FUNCTIONS {
528            let name = format!("sim://{f}(1, 2)");
529            assert_eq!(pv_names(&name), [name.as_str()]);
530        }
531    }

534    fn a_large_selection_is_parsed_in_linear_time() {

584    #[test]
585    fn the_message_says_what_happened() {
586        let p = |added: &[&str], present: &[&str], text: &str| Pasted {
587            added: added.iter().map(|s| s.to_string()).collect(),
588            present: present.iter().map(|s| s.to_string()).collect(),
589            text: text.into(),
590            too_many: 0,
591        };
592        assert_eq!(p(&["A:B"], &[], "").message(), "Added A:B");
593        assert_eq!(p(&["A:B", "C:D"], &["E:F"], "").message(), "Added 2 PVs");
594        assert_eq!(p(&[], &["A:B"], "").message(), "A:B is plotted already");
595        assert_eq!(
596            p(&[], &["A:B", "C:D"], "").message(),
597            "2 PVs are plotted already"
598        );
599        assert_eq!(
600            p(&[], &[], "12:30\n  42").message(),
601            "No PV names in “12:30 42”"
602        );
603        assert!(p(&[], &[], &"1".repeat(100)).message().ends_with("1…”"));
604        assert!(p(&[], &[], " ").message().contains("empty"));
605        let many = Pasted {
606            too_many: 143,
607            ..Pasted::default()
608        };
609        assert_eq!(
610            many.message(),
611            "The selection names 143 PVs: select at most 20"
612        );
613    }
